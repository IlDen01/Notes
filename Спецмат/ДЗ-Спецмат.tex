    \subsection*{Поле $\mathbb{F}_8$}
    
    \begin{example}[Построение $\mathbb{F}_8$]
    	Многочлен $f(x) = x^3+x+1$ неприводим над $\mathbb{F}_2$: $f(0)=1$, $f(1)=1$, корней нет, а степень $3$~--- значит, разложение на множители меньшей степени невозможно. $\mathbb{F}_8 := \mathbb{F}_2[x]/(f)$, $a := \bar x$, тогда $a^3 = a+1$.
    
    	Проверим, что $a$~--- примитивный элемент, то есть $\ord(a) = 7 = |\mathbb{F}_8^*|$:
    	\[
    		a^0=1,\quad a^1=a,\quad a^2=a^2,\quad a^3=a+1,\quad a^4=a^2+a,
    	\]
    	\[
    		a^5=a\cdot a^4=a^3+a^2=a^2+a+1,\quad a^6=a\cdot a^5=a^3+a^2+a=a^2+1,\quad a^7=a\cdot a^6=a^3+a=1.
    	\]
    	Все восемь степеней $a^0,\dots,a^6$ различны и исчерпывают все ненулевые элементы поля, значит $a$~--- примитивный элемент.
    \end{example}

    \subsection*{Поле $\mathbb{F}_9$}
    
    \begin{example}[Построение $\mathbb{F}_9$]
    	Многочлен $g(x) = x^2+1$ неприводим над $\mathbb{F}_3$: квадраты в $\mathbb{F}_3$~--- это $0^2=0$, $1^2=1$, $2^2=1$, значит $-1=2$ не является квадратом, и уравнение $x^2=-1$ решений не имеет. $\mathbb{F}_9 := \mathbb{F}_3[x]/(g)$, $i := \bar x$, тогда $i^2 = -1 = 2$; элементы поля~--- $a+bi$, $a,b \in \{0,1,2\}$.
    
    	В качестве примитивного элемента возьмем $u := 1+i$ (порядок группы $\mathbb{F}_9^*$ равен $8$):
    	\[
    		u^2 = (1+i)^2 = 1+2i+i^2 = 2i,\qquad u^3 = u^2 \cdot u = 2i(1+i) = 2i+2i^2 = 1+2i,
    	\]
    	\[
    		u^4 = u^3\cdot u = (1+2i)(1+i) = 1+3i+2i^2 = 2,\qquad u^5 = 2u = 2+2i,
    	\]
    	\[
    		u^6 = u^5\cdot u = (2+2i)(1+i)=2+4i+2i^2=i,\qquad u^7 = u^6 \cdot u = i(1+i)=i+i^2=2+i,\qquad u^8=1.
    	\]
    	Все семь степеней различны, значит $u=1+i$~--- примитивный элемент.
    \end{example}
