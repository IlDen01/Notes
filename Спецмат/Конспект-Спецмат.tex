	\section{Теория групп.}
	\begin{definition}
		\label{group-def}
		Группа~--- множество с одной операцией $*$: $G \times G \rightarrow G$ со следующими свойствами:
		\begin{enumerate}
			\item $a * (b * c) = (a * b) * c$
			\item $\exists e$: $a * e = e * a = a$
			\item $\forall a$ $\exists a^{-1}$: $a * a^{-1} = a^{-1} * a = e$
		\end{enumerate}
	\end{definition}
	\begin{example}
		\begin{itemize}
			\item $(\mathbb{Z}; +)$
			\item $(\mathbb{Q}; +)$
			\item $(\mathbb{R}; +)$
			\item $(\mathbb{C}; +)$
			\item $(V; +)$
			\item $(\mathbb{R}_+; \cdot)$
			\item $(\mathbb{R} \setminus \{ 0 \}; \cdot)$
			\item $(\mathbb{Z}_n; +)$
		\end{itemize}
	\end{example}
	\begin{definition}
		Пусть $G$~--- группа, $H \subset G$. Говорим, что $H$ является подгруппой (пишем $H < G$), если $H$ является группой относительно операции в $G$. Чтобы проверить, что $H$ является подгруппой, необходимо убедиться, что произведение двух элементов из $H$ принадлежит $H$, и элементы, обратные к $H$, тоже лежат в $H$.
	\end{definition}
	\begin{theorem}[Кэли]
		Любая группа $G$ является подгруппой в группе подстановок, а именно $S_G$.
	\end{theorem}
	\begin{definition}
		Абелева группа~--- \hyperref[group-def]{группа} с коммутативностью.
	\end{definition}
	\begin{definition}
		Говорят, что группа $G$ порождается элементами $\{ x_i \}$, если любой элемент из $G$ можно представить как произведение нескольких $x_i$ и обратных к ним. Группа называется циклической, если она порождена одним элементом.
	\end{definition}
	\begin{theorem}
		Конечная циклическая группа изоморфна $(\mathbb{Z}_n, +)$.
	\end{theorem}
	\begin{definition}
		Пусть $G$~--- группа, $H < G$. Введем два отношения эквивалентности на $G$: $x \sim_1 y$ если $xy^{-1} \in H$, $x \sim_2 y$ если $x^{-1}y \in H$.
	\end{definition}
	\begin{statement}
		$\sim_1$ и $\sim_2$ совпадают в абелевой группе.
	\end{statement}
	\begin{note}
		Классы эквивалентности по отношению $\sim_1$ называются левыми смежными классами; класс элемента $x$ обозначается $xH$. Классы эквивалентности по $\sim_2$~--- правые смежные классы, обозначаются $Hx$.
	\end{note}
	\begin{definition}
		Пусть теперь группа $G$ конечна, $H < G$. Количество классов эквивалентности $\sim_1$ называется индексом $G$ по $H$ и обозначается $[G : H]$.
	\end{definition}
	\begin{theorem}[Лагранжа]
		$|G| = |H| \cdot [G : H]$.
	\end{theorem}
	\begin{definition}
		Пусть $x \in G$. Порядком элемента $x$ называется наименьшее натуральное число $n$, такое что $x^n = e$, где $e$~--- нейтральный элемент. Обозначение: $ord(x)$. Если такого $n$ не существует, то пишем $ord(x) = + \infty$.
	\end{definition}
	\begin{definition}
		Пусть $X, Y \subset G$~--- подмножества группы. Их произведением будем называть множество $\{ xy | x \in X, y \in Y \}$.
	\end{definition}
	\begin{definition}
		Полная линейна группа $GL(n, F)$~--- это множество всех квадратных матриц размера $n \times n$ с элементами из поля $F$, которые являются обратимыми ($\det \not= 0$), вместе с операцией матричного умножения.
	\end{definition}
	\begin{definition}
		Специальная линейная группа $SL(n, F)$~--- это подгруппа полной линейной группы, состоящая из всех матриц с определителем, равным $1$. То есть это множество всех матриц $A$ размера $n \times n$ над полем $F$, таких что $\det(A) = 1$.
	\end{definition}
	\begin{definition}
		Специальная ортогональная группа $SO(n, F)$~--- группа из ортогональных матриц. Матрица $A$ называется ортогональной, если $A^T \times A = A \times A^T = E$.
	\end{definition}
	\begin{statement}
		Пусть $X$~--- произвольное множество. Тогда множество всех биекций $f: X \rightarrow X$ образуют группу относительно композиции. Эту группу обозначают $S_X$. Если $X = \{ 1, 2, \dots, n \}$, то $S_X$ обозначают $S_n$~--- группа подстановок.
	\end{statement}
	\subsection{Таблица Кэли.}
	\begin{definition}
		Пусть $G$~--- конечная группа порядка $n$. Её таблицей Кэли (таблицей умножения) будем называть таблицу $(n + 1) \times (n + 1)$ (левый столбец и левая строка считаются нулевыми и служат лишь для нумерации). В нулевом столбце и в нулевой строке стоят все элементы группы в одном и том же порядке. На пересечении строки и столбца этой таблицы будем ставить произведение соответствующих элементов в нулевом столбце и в нулевой строке (слева пишется элемент, задающий строку, справа — столбец).
	\end{definition}
	\begin{definition}
		Напомним, что перестановкой мы будем называть биекцию $f: \{ 1, 2, \dots, n \} \rightarrow \{ 1, 2, \dots, n \}$. Умножение перестановок — композиция биекций. Перестановки записываются в две строчки: в первой~--- числа от $1$ до $n$ (как правило, в порядке возрастания, но не обязательно), а во второй строчке под числом $k$ стоит число $f(k)$.
	\end{definition}
	\subsection{Группы.}
	\begin{theorem}
		$G$~--- группа; $H < G$. Равносильно:
		\begin{enumerate}
			\item $\forall x$ $xH = Hx$
			\item $\forall x$ $xH \subset Hx$
			\item $\forall x$ $xH \supset Hx$
			\item $\forall x$ $xHx^{-1} = H$
			\item $\forall x$ $xHx^{-1} \subset H$
			\item $\forall x$ $xHx^{-1} \supset H$
			\item $\forall x \in G$ $\forall h \in H$ $x^{-1}hx \in H$
		\end{enumerate}
	\end{theorem}
	\begin{definition}
		Такая $H$ называется нормальной подгруппой. Обозначается $H \vartriangleleft G$.
	\end{definition}
	\begin{definition}
		$xhx^{-1}$ называется сопряженным элементом $h$.
	\end{definition}
	\begin{statement}
		$h \sim x^{-1}hx$~--- отношение эквивалентности.
	\end{statement}
	\begin{definition}
		$H \vartriangleleft G$. Класс смежности по $H$ можно перемножать.
	\end{definition}
	\begin{definition}
		Множество классов смежности образуют группу. Это называется фактор-группой $G$ по $H$. $G/H$.
	\end{definition}
	\begin{definition}
		Простая группа~--- группа без нетривиальных нормальных подгрупп.
	\end{definition}
	\subsection{Перестановки.}
	\begin{definition}
		Перестановкой конечного множества $M$ называется биекция $\pi : M \rightarrow M $. Множество всех перестановок множества $M$ обозначается символом $S(M)$. Произведением перестановок $\pi$ и $\sigma$ называется перестановка $\sigma \cdot \pi$, соответствующая биекции $x \mapsto \sigma (\pi (x))$.
	\end{definition}
	\begin{definition}
		Пусть $\pi \in S(M)$. Орбитой элемента $a \in M$ называется множество $\orb_{\pi}(a) = \{ a, \pi(a), \pi^2(a), \dots \}$. Порядком элемента $a$ называется мощность его орбиты: $\ord(a) = |\orb(a)|$.
	\end{definition}
	\begin{definition}
		В предыдущей серии показано, что любая перестановка является произведением циклов. Количество циклов в перестановке обозначается $\cycl(\pi)$. Если перестановка разбивается на циклы, длины которых~--- $l_1, \dots, l_m$, то $(l_1, \dots,l_m)$ называется цикленным типом перестановки $\pi$. Перестановка, у которой цикленный тип $(2, 1, 1, \dots, 1)$, называется транспозицией.
	\end{definition}
	\begin{definition}
		Перестановка $\pi$ называется четной или нечетной в зависимости от четности числа $n + \cycl(\pi)$. Знаком перестановки называется число $\text{sign}(\pi) = (-1)^{n + \cycl(\pi)}$.
	\end{definition}
	\begin{definition}
		Хотим $\forall$ перестановки называть ее четной или нечетной: ЧЧ $=$ Ч, НН $=$ Ч, ЧН $=$ Н, НЧ $=$ Н и $\exists$ Ч и НЧ перестановки.
	\end{definition}
	\begin{definition}
		Инверсия: $\alpha = \begin{pmatrix}
			1 & 2 & \dots & i & \dots & j & \dots & n\\
			a_1 & a_2 & \dots & a_i & \dots & a_j & \dots & n\\
		\end{pmatrix}$. Пара $(i, j)$ называется инверсией, если $i < j$ и $a_i > a_j$.
	\end{definition}
	\begin{definition}
		Перестановка с четным количеством инверсий~--- четная, с нечетным~--- нечетная.
	\end{definition}
	\begin{note}
		$\forall$ перестановка~--- произведение транспозиций. Четное количество транспозиций $\Leftrightarrow$ четная перестановка.
	\end{note}
	\begin{definition}
		Четная перестановка $= (n + \text{количество циклов}) \mod{2}$.
	\end{definition}
	\begin{theorem}
		Перестановки сопряжены $\Leftrightarrow$ совпадает их циклический тип.
	\end{theorem}
	\subsection{Центр группы.}
	\begin{definition}
		$G$~--- группа. Центр $G$: $Z(G) := \{ g \in G | gh = hg, \forall g \in G \}$.
	\end{definition}
	\begin{statement}
		$Z(S_n) = \{ e \}$.
	\end{statement}
	\begin{theorem}
		$Z(G) \vartriangleleft G$.
	\end{theorem}
	\begin{statement}
		$Z(GL(\mathbb{R})) = \begin{pmatrix}
			\alpha & 0 & \dots & 0\\
			0 & \alpha & \dots & 0\\
			\vdots & \vdots & \ddots & \vdots\\
			0 & 0 & \dots & \alpha
		\end{pmatrix}$
	\end{statement}
	\begin{definition}[Коммутатор]
		$x, y \in G$. $[x, y] := xyx^{-1}y^{-1}$.
	\end{definition}
	\begin{statement}
		$[x, y]^{-1} = [y, x]$.
	\end{statement}
	\begin{note}
		Произведение коммутаторов не обязательно является коммутатором.
	\end{note}
	\begin{definition}[Коммутант]
		$[G, G] := \left< [x, y] \right>$.
	\end{definition}
	\begin{theorem}
		\begin{enumerate}
			\item $[G, G] \vartriangleleft G$.
			\item $G / [G; G]$~--- абелева.
			\item $G / H$~--- абелева $\Rightarrow H \supset [G, G]$.
		\end{enumerate}
	\end{theorem}
	\begin{statement}
		$\alpha^{-1} [x, y] \alpha = [\alpha^{-1} x \alpha, \alpha^{-1} y \alpha]$.
	\end{statement}
	\begin{statement}
		$\alpha^{-1} [x_1, y_1] [x_2, y_2] \dots [x_k, y_k] \alpha \in [G, G]$.
	\end{statement}
	\begin{theorem}[Галуа.]
		$A_n$ простая, если $n \geqslant 5$.
	\end{theorem}
	\subsection{Гомоморфизм группы.}
	\begin{definition}
		$(G, \cdot)$; $(H; *)$~--- группа. $f: G \rightarrow H$~--- отображение. $f$ называется гомоморфизмом, если $\forall a, b \in G$ $f(a \cdot b) = f(a) * f(b)$.
	\end{definition}
	\begin{definition}
		$f: G \rightarrow H$~--- гомоморфизм. $f$ называется:
		\begin{itemize}
			\item Эпиморфизмом, если $f$~--- сюръекция.
			\item Мономорфизм, если $f$~--- инъекция.
			\item Изоморфизм, если $f$~--- биекция.
		\end{itemize}
		$G \simeq H$, если существует изоморфизм $f: G \rightarrow H$.
	\end{definition}

	$\mathbb{H} = \{ \pm 1, \pm i, \pm j, \pm k \}$.
	\begin{lemma}
		$a_1, a_2, \dots, a_{2n}$~--- строка из символов $j$ и $k$. Если перевернуть строку и заменить $j \leftrightarrow k$, то произведение не изменится.
	\end{lemma}
	\subsection{Теоремы об изоморфизме.}
	\subsubsection{Первая теорема об изоморфизме.}
	\begin{theorem}
		Пусть $f : G \rightarrow H$~--- гомоморфизм групп. Тогда $\Img{f} \simeq G / \Ker{f}$.
	\end{theorem}
	\subsubsection{Вторая теорема об изоморфизме.}
	\begin{theorem}
		Пусть $G$~--- группа, $H$ и $K$~--- ее подгруппы, причем $K$ нормальна. Тогда $HK / K \simeq H / (H \cap K)$.
	\end{theorem}
	\subsubsection{Третья теорема об изоморфизме.}
	\begin{theorem}
		Пусть $G$~--- группа, $H$ и $K$~--- ее нормальные подгруппы, причем $K \subset N$. Тогда $(G / K) / (N / K) \simeq G / N$.
	\end{theorem}
	\subsection{Группы автоморфизмов графа.}
	\begin{definition}
		Автоморфизмом графа Г называется биекция множества его вершин на себя, так что пара вершин смежны тогда и только тогда, когда смежны их образы. Автоморфизмы графа образуют группу относительно композиции. Обозначается $\Aut{\text{Г}}$.
	\end{definition}
	\begin{theorem}[Фрухта]
		Для любой группы $G$ найдется граф, группа автоморфизмов которого в точности $G$.
	\end{theorem}
	\subsection{Действие группы на множестве.}
	\begin{definition}
		$G$~--- группа, $M$~--- множество. Действием (левое) называем $G$ на $M$ называем $\cdot$: $G \times M \rightarrow M$: $(g; x) \mapsto g \cdot x$ или $g(x)$.
	\end{definition}
	\begin{property}
		\begin{enumerate}
			\item $(g \cdot h) \cdot x = g \cdot (h \cdot x)$.
			\item $e \cdot x = x$.
		\end{enumerate}
	\end{property}
	\begin{corollary}
		$gx = y \Rightarrow g^{-1}y = x$.
	\end{corollary}
	\begin{definition}[2]
		Действие $G$ на $M$~--- это гомоморфизм $F: G \rightarrow S_M$ (группа перестановок $M$). $g \cdot x := F(g)(x)$. Наоборот: есть $\cdot$: $G \times M \rightarrow M$: $F(g)(x) := g(x)$.
	\end{definition}
	\begin{definition}
		$x$ неподвижен под действием $g$, если $g \cdot x = x$. Орбита $x$: $\orb(x) := \{ g \cdot x | g \in G \}$.
	\end{definition}
	\begin{statement}
		Орбиты либо совпадают, либо не пересекаются.
	\end{statement}
	\begin{definition}
		$x \in M$; стабилизатор $x$~--- это $\st{x} := \{ g \in G | gx = x \}$.
	\end{definition}
	\begin{statement}
		$\st{x}$~--- подгруппа в $G$.
	\end{statement}
	\begin{definition}
		$\st{g} := \{ x | gx = x \}$~--- стабилизатор $g$.
	\end{definition}
	\begin{lemma}[Бермсайда]
		$G$~--- конечная группа; $M$~--- конечная $\Rightarrow$ количество орбит $= \dfrac{1}{|G|} \sum\limits_{g \in G} |\st{g}| = \dfrac{1}{|G|} \sum\limits_{x \in M} |\st{x}|$, тк $\sum\limits_{g \in G} |\st{g}| = \sum\limits_{x \in M} |\st{x}| = \# \{ (g, x) | gx = x \}$.
	\end{lemma}
	
	\section{Метрические пространства.}
	
	\begin{definition}
		$M$~--- множество; $\rho$: $M \times M \rightarrow \mathbb{R}$.
		\begin{enumerate}
			\item $\rho(x, y) \geqslant 0$; $\rho(x, y) = 0 \Leftrightarrow x = y$
			\item $\rho(x, y) = \rho(y, x)$
			\item $\rho(x, z) \leqslant \rho(x, y) + \rho(y, z)$
		\end{enumerate}
		$(M, \rho)$~--- метрическое пространство.
	\end{definition}
	
	\begin{definition}
		$B(x_0, r) := \{ x \in M | \rho(x, x_0) < r \}$
	\end{definition}
	
	\begin{definition}
		\label{open-def}
		$A \subset M$. $A$ называется открытым, если $\forall x_0 \in A \, \exists \varepsilon: B(x_0, \varepsilon) \subset A$.
	\end{definition}
	
	\begin{definition}
		$f$ называется замкнутым, если $X \setminus f$ \hyperref[open-def]{открыто}.
	\end{definition}
	
	\begin{theorem}
		\begin{enumerate}
			\item $\{ U_i \}_{i \in I}$~--- открыто $\Rightarrow$ $\bigcup\limits_{i \in I}$~--- открыто.
			\item $\{ U_i \}_{i \in I}^{n}$~--- открыто $\Rightarrow$ $\bigcup\limits_{i \in I}^{n}$~--- открыто.
			\item $\varnothing; M$~--- открыты.
		\end{enumerate}
	\end{theorem}
	
	\begin{theorem}
		\begin{enumerate}
			\item $\{ F_i \}_{i \in I}$~--- замкнуто $\Rightarrow$ $\bigcap\limits_{i \in I} F_i$~--- замкнуто.
			\item $\{ F_i \}_{i \in I}^{n}$~--- замкнуто $\Rightarrow$ $\bigcap\limits_{i \in I}^{n} F_i$~--- замкнуто.
			\item $\varnothing; M$~--- замкнуто.
		\end{enumerate}
	\end{theorem}
	
	\begin{definition}[Топологическое пространство]
		Множество $X$ с множеством $\Omega \subset 2^X$ ($A \in \Omega$~--- открытые подмножества):
		\begin{enumerate}
			\item $\bigcup\limits_{i \in I} U_i \in \Omega$, если $\forall U_i \in \Omega$
			\item $U_1, \dots, U_n \in \Omega \Rightarrow \bigcap\limits_{i=1}^n U_i \in \Omega$
			\item $\varnothing, X \in \Omega$
		\end{enumerate}
	\end{definition}
	
	\begin{definition}[Граница множества $\partial A$]
		\label{border-def}
		Это множество точек, в любой окрестности (шаре) которых есть как точки из $A$, так и точки не из $A$. $x \in \partial A \Leftrightarrow \forall r > 0: B(x, r) \cap A \ne \varnothing$ и $B(x, r) \cap (X \setminus A) \ne \varnothing$
	\end{definition}
	
	\begin{definition}[Замыкание $\overline{A}$]
		Объединение множества и его \hyperref[border-def]{границы}: $\overline{A} = A \cup \partial A$. Замыкание всегда является замкнутым множеством.
	\end{definition}
	
	\begin{definition}[Канторово множество, $K$]
		Получается, если выкидывать каждый раз среднюю треть на $(0, 1)$.
	\end{definition}
	
	\begin{definition}[Непрерывность отображения]
		$f: X \rightarrow Y$ непрерывна, если $\forall U \subset Y \, f^{-1} (U)$~--- открыто в $X$.
	\end{definition}
	
	\begin{definition}[Гомеоморфизм]
		$X \simeq Y$, если $\exists f: X \rightarrow Y$:
		\begin{itemize}
			\item $f$~--- непрерывна.
			\item $f$~--- биекция.
			\item $f'$~--- непрерывна.
		\end{itemize}
	\end{definition}
	
	\begin{property}
		\begin{enumerate}
			\item $K$~--- замкнуто.
			\item Длина $K = 0$.
			\item $K$ несчетно.
			\item $K \simeq K \times K$
		\end{enumerate}
	\end{property}
	
	\section{Связность.}
	
	\begin{definition}[Несвязность]
		$X$~--- несвязное пространство, если $\exists$ открытые $U, V \subset X$:
		\begin{enumerate}
			\item $U, V \ne \varnothing$
			\item $U \cap V = \varnothing$
			\item $U \cup V = X$
		\end{enumerate}
	\end{definition}
	
	\begin{definition}[Несвязность]
		$X$~--- топологическое пространство; $A \ne \varnothing \subset X$. $A$ несвязно, если $\exists$ открытые $U, V \subset X$:
		\begin{itemize}
			\item $U \cap A \ne \varnothing$, $V \cap A \ne \varnothing$
			\item $U \cap V \cap A = \varnothing$
			\item $A \subset U \cup V$
		\end{itemize}
	\end{definition}
	
	\begin{definition}[Линейная связность]
		$X$~--- топологическое пространство. $X$ называется линейно связным, если $\forall a, b \in X$  $\exists$ непрерывное $f: [0, 1] \rightarrow X$: $f(0) = a, f(1) = b$.
	\end{definition}
	
	\begin{note}
		Линейная связность $\Rightarrow$ связность.
	\end{note}
	
	\begin{theorem}
		$I = [0, 1]$ связен.
	\end{theorem}
	
	\begin{proof}
		Допустим противное.
		
		$U, V$~--- открытые; $U \cap I \ne \varnothing$, $V \cap I \ne \varnothing$; $U \cap V \cap I = \varnothing$; $I \subset U \cap V$.
		
		НУО $0 \in U$.
		
		$x_0 = \inf \{ V \cap I \}$.
		
		Пусть $x_0 \in V \Rightarrow \exists \varepsilon > 0: B(x_0, \varepsilon) \subset V \Rightarrow (x_0 - \varepsilon; x_0 + \varepsilon) \subset V \Rightarrow x_0$~--- не инфимум.
		
		Пусть тогда $x_0 \in U \Rightarrow \exists \varepsilon > 0: B(x_0, \varepsilon) \subset U \Rightarrow (x_0 - \varepsilon; x_0 + \varepsilon) \subset U \Rightarrow x_0 + \dfrac{\varepsilon}{2}$~--- нижняя грань $V \cap I$.
		
		Если $x_0 = 0: [0, \varepsilon) \subset U \Rightarrow x_0 \geqslant \varepsilon$.
	\end{proof}
	
	\begin{theorem}
		$f: X \rightarrow Y$ непрерывно; $A \subset X$~--- связно $\Rightarrow f(A)$ связно в $Y$.
	\end{theorem}
	
	\begin{theorem}[Вейерштрасса]
		$X$~--- связен; $f: X \rightarrow R$; $f(x_1) = a$, $f(x_2) = b$, $a \leqslant c \leqslant b \Rightarrow \exists x_0: f(x_0) = c$.
	\end{theorem}
	
	\begin{definition}[Блин]
		Открытое (связное) ограниченное подмножество плоскости.
	\end{definition}
	
	\begin{statement}
		Блин одной прямой, параллельной данной, можно разделить на 2 равные части.
	\end{statement}
	
	\begin{statement}
		Любой блин можно разделить на две равновеликие части прямой, проходящей через заданную точку.
	\end{statement}
	
	\begin{statement}
		Два любых блина, каждый на две равновеликие части.
	\end{statement}
	
	\begin{statement}
		Один блин на 4 равновеликие части, двумя перпендикулярными разрезами.
	\end{statement}
	
	\section{Компактность.}
	
	\begin{definition}[Покрытие]
		$(X, \Omega)$~--- топологическое пространство; $\{ U_i \}$~--- открытое. Если $\bigcup\limits_i U_i = X$, то $\{ U_i \}$~--- открытое покрытие.
	\end{definition}
	
	\begin{definition}[Компактность]
		$X$ называется компактным, если $\forall$ покрытия $\{ U_i \}$ $\exists U_1, U_2, \dots, U_n$: $\{ U_k \}_{k = 1}^n$~--- покрытие.
	\end{definition}
	
	\begin{definition}
		$(X, \Omega)$~--- топологическое пространство, $A \subset X$. $A$ компактное, если $\forall \{ U_i \}_{i \in I}$: $\bigcup\limits_{i \in I} U_i \supset A \Rightarrow \exists U_1, \dots, U_n$: $U_1 \cup \dots \cup U_n \supset A$.
	\end{definition}
	
	\begin{theorem}
		$A \subset \mathbb{R}^n \Rightarrow A$~--- компактно $\Leftrightarrow A$~--- замкнуто и ограничено.
	\end{theorem}
	
	\begin{theorem}
		$A$ компактно, $f: A \rightarrow \mathbb{R}$ непрерывная функция $\Rightarrow f$ достигает свои max и min.
	\end{theorem}
	
	\begin{theorem}
		$[0, 1]$~--- компактно.
	\end{theorem}
	
	\begin{theorem}
		Замкнутое подмножество компактного~--- компактно.
	\end{theorem}

    \section{Поля}
    
    \begin{definition}\label{def:field}
    	Множество $K$ с двумя операциями $+$ и $\cdot$ называется \emph{полем}, если выполнены следующие аксиомы:
    	\begin{enumerate}
    		\item \label{ax:add-assoc} ассоциативность сложения: $(a+b)+c = a+(b+c)$;
    		\item \label{ax:zero} существует нейтральный элемент $0$: $a + 0 = a$;
    		\item \label{ax:add-inv} для любого $a$ существует противоположный $-a$: $a + (-a) = 0$;
    		\item \label{ax:add-comm} коммутативность сложения: $a + b = b + a$;
    		\smallskip
    		\item[] \quad (\ref{ax:add-assoc}--\ref{ax:add-comm}: $K$ является абелевой группой по сложению)
    		\smallskip
    		\item \label{ax:distrib} дистрибутивность: $a(b+c) = ab + ac$;
    		\smallskip
    		\item \label{ax:mul-assoc} ассоциативность умножения: $(ab)c = a(bc)$;
    		\item \label{ax:one} существует единица $1 \ne 0$: $a \cdot 1 = a$;
    		\item \label{ax:mul-inv} для любого $a \ne 0$ существует обратный $a^{-1}$: $a \cdot a^{-1} = 1$;
    		\item \label{ax:mul-comm} коммутативность умножения: $ab = ba$;
    		\smallskip
    		\item[] \quad (\ref{ax:mul-assoc}--\ref{ax:mul-comm}: $K \setminus \{0\}$ является абелевой группой по умножению)
    	\end{enumerate}
    \end{definition}
    
    \begin{example}
    	\[
    	\mathbb{Q}, \quad \mathbb{R}, \quad \mathbb{C}, \quad \mathbb{F}_p = \mathbb{Z}/p\mathbb{Z} \ (p \text{ --- простое}), \quad \mathbb{Q}(\sqrt{2}), \ \ldots
    	\]
    \end{example}
    
    \begin{definition}\label{def:extension}
    	Пусть $K \subset L$ --- поля (определение~\ref{def:field}), причём операции на $L$, ограниченные на $K$, совпадают с операциями на $K$ (т.е. $K$ замкнуто относительно $+, \cdot$ и содержит $0, 1$).
    	
    	Тогда $K$ называется \emph{подполем} $L$, а $L$ называется \emph{расширением} поля $K$. Обозначение: $L/K$.
    \end{definition}
    
    \begin{theorem}\label{thm:vector-space}
    	Если $L/K$ --- расширение полей (определение~\ref{def:extension}), то $L$ является линейным (векторным) пространством над $K$: элементы $L$ складываются как в поле, а умножение на скаляры $\lambda \in K$ --- это обычное умножение в $L$. Все аксиомы векторного пространства следуют из аксиом поля $L$ (определение~\ref{def:field}).
    \end{theorem}
    \begin{proof}
    	Сложение в $L$ уже задано (как в поле). Умножение на скаляры $K \times L \to L$ определим как обычное умножение в $L$ (это корректно, так как $K \subset L$). Проверим аксиомы векторного пространства:
    	\begin{itemize}
    		\item $(L, +)$ --- абелева группа (аксиомы~\ref{ax:add-assoc}--\ref{ax:add-comm} поля $L$, определение~\ref{def:field});
    		\item $\lambda(v+w) = \lambda v + \lambda w$ и $(\lambda+\mu)v = \lambda v + \mu v$ --- это в точности дистрибутивность (аксиома~\ref{ax:distrib}) в $L$;
    		\item $(\lambda\mu)v = \lambda(\mu v)$ --- ассоциативность умножения в $L$ (аксиома~\ref{ax:mul-assoc});
    		\item $1 \cdot v = v$ --- аксиома~\ref{ax:one}.
    	\end{itemize}
    	Все нужные тождества --- это частные случаи аксиом поля $L$ (определение~\ref{def:field}), применённые к $\lambda, \mu \in K \subset L$. Значит, $L$ --- векторное пространство над $K$.
    \end{proof}
    
    \begin{definition}\label{def:degree}
    	Пусть $L/K$ --- расширение полей (определение~\ref{def:extension}); напомним, что тогда $L$ --- векторное пространство над $K$ (теорема~\ref{thm:vector-space}). Если $\dim_K L < \infty$, то расширение $L/K$ называется \emph{конечным}, а само число
    	\[
    	[L:K] := \dim_K L
    	\]
    	называется \emph{степенью расширения}.
    \end{definition}
    
    \begin{definition}\label{def:algebraic}
    	Пусть $L/K$ --- расширение полей (определение~\ref{def:extension}) и $\alpha \in L$ (в общем случае $\alpha$ может лежать и в каком-то большем поле, содержащем $K$, --- сам $\alpha$ не обязан лежать в $K$).
    	
    	Элемент $\alpha$ называется \emph{алгебраическим над $K$}, если существует ненулевой многочлен $f \in K[x]$ такой, что $f(\alpha) = 0$.
    	
    	Если такого $f$ не существует, $\alpha$ называется \emph{трансцендентным над $K$}.
    
    	Наименьшая степень такого многочлена $f$ называется степенью $\alpha$. Сам многочлен $f$ называется \emph{минимальным} для $\alpha$.
    \end{definition}
    
    \begin{definition}\label{def:min-poly}
    	Пусть $\alpha$ алгебраичен над $K$ (определение~\ref{def:algebraic}). Многочлен $f \in K[x]$ наименьшей степени среди всех многочленов, обращающихся в $\alpha$ в ноль (обычно берут ещё нормированный, со старшим коэффициентом $1$), называется \emph{минимальным многочленом} $\alpha$ над $K$.
    	
    	Степень этого многочлена называется \emph{степенью элемента $\alpha$} над $K$.
    \end{definition}
    
    \begin{definition}\label{def:algebraic-extension}
    	Расширение $L/K$ (определение~\ref{def:extension}) называется \emph{алгебраическим}, если каждый элемент $\alpha \in L$ является алгебраическим над $K$ (определение~\ref{def:algebraic}).
    \end{definition}
    
    \begin{definition}\label{def:generated-field}
    	Пусть $\alpha$ --- произвольный элемент некоторого расширения $K$ (алгебраичность заранее не предполагается). Через $K(\alpha)$ обозначается наименьшее по включению поле, содержащее $K$ и $\alpha$.
    \end{definition}
    
    \begin{theorem}\label{thm:irreducible}
    	Пусть $\alpha$ алгебраичен над $K$ (определение~\ref{def:algebraic}), а $f$ --- его минимальный многочлен (определение~\ref{def:min-poly}). Тогда $f$ неприводим над $K$.
    \end{theorem}
    \begin{proof}
    	Предположим противное: $f = gh$, где $g, h \in K[x]$ и $\deg g, \deg h < \deg f$. Тогда $0 = f(\alpha) = g(\alpha)h(\alpha)$. Так как $\alpha$ лежит в поле (в частности, в области целостности), из $g(\alpha)h(\alpha)=0$ следует $g(\alpha)=0$ или $h(\alpha)=0$. Но $\deg g, \deg h < \deg f$, а $f$ по определению~\ref{def:min-poly} --- многочлен наименьшей степени, обращающийся в $\alpha$ в ноль. Противоречие. Значит, $f$ неприводим.
    \end{proof}
    
    \begin{theorem}\label{thm:basis}
    	Пусть $\alpha$ алгебраичен над $K$ (определение~\ref{def:algebraic}) и его степень равна $n$ (определение~\ref{def:min-poly}). Тогда $1, \alpha, \alpha^2, \ldots, \alpha^{n-1}$ образуют базис $K[\alpha]$ как векторного пространства над $K$.
    \end{theorem}
    \begin{proof}
    	\emph{Линейная независимость.} Если $c_0 + c_1\alpha + \cdots + c_{n-1}\alpha^{n-1} = 0$ для некоторых $c_i \in K$, не все из которых равны нулю, то многочлен $c_0 + c_1x + \cdots + c_{n-1}x^{n-1}$ --- ненулевой, имеет степень $<n$ и обращается в $\alpha$ в ноль. Это противоречит тому, что минимальный многочлен $f$ (определение~\ref{def:min-poly}) имеет наименьшую степень $n$ среди всех многочленов, обнуляющихся на $\alpha$. Значит, все $c_i = 0$.
    	
    	\emph{Порождающая система.} Пусть $\beta \in K[\alpha]$, то есть $\beta = g(\alpha)$ для некоторого $g \in K[x]$. Поделим $g$ с остатком на $f$ (в $K[x]$ это возможно, так как $f$ можно считать нормированным): $g = qf + r$, где $r = 0$ или $\deg r < n$. Тогда
    	\[
    	\beta = g(\alpha) = q(\alpha)f(\alpha) + r(\alpha) = q(\alpha)\cdot 0 + r(\alpha) = r(\alpha),
    	\]
    	а $r(\alpha)$ --- это $K$-линейная комбинация $1, \alpha, \ldots, \alpha^{n-1}$. Значит, эти элементы порождают $K[\alpha]$.
    	
    	Итак, $1, \alpha, \ldots, \alpha^{n-1}$ --- базис $K[\alpha]$ над $K$.
    \end{proof}
    
    \begin{theorem}\label{thm:Kalpha}
    	Пусть $\alpha$ алгебраичен над $K$ (определение~\ref{def:algebraic}). Тогда:
    	\begin{enumerate}
    		\item[(a)] \label{thm:Kalpha-a} расширение $K(\alpha)/K$ алгебраическое (определение~\ref{def:algebraic-extension});
    		\item[(b)] \label{thm:Kalpha-b} расширение $K(\alpha)/K$ конечное (определение~\ref{def:degree});
    		\item[(c)] \label{thm:Kalpha-c} $[K(\alpha):K]$ равно степени $\alpha$ (определение~\ref{def:min-poly}).
    	\end{enumerate}
    \end{theorem}
    \begin{proof}
    	\emph{(b), (c).} Пусть $f$ --- минимальный многочлен $\alpha$ (определение~\ref{def:min-poly}), $\deg f = n$. По теореме~\ref{thm:irreducible} $f$ неприводим над $K$, поэтому идеал $(f) \subset K[x]$ максимален (так как $K[x]$ --- кольцо главных идеалов), и, значит, $K[x]/(f)$ --- поле.
    	
    	Рассмотрим гомоморфизм колец $\varphi: K[x] \to K[\alpha]$, $\varphi(g) = g(\alpha)$. Он сюръективен по определению $K[\alpha]$, а его ядро --- это в точности многочлены, обращающиеся в $\alpha$ в ноль, то есть идеал $(f)$ (это множество --- идеал, а $f$ --- его элемент наименьшей степени, значит, он и порождает этот идеал, поскольку $K[x]$ --- область главных идеалов). По теореме о гомоморфизме,
    	\[
    	K[\alpha] \cong K[x]/(f),
    	\]
    	а значит, $K[\alpha]$ само является полем. Так как $K(\alpha)$ --- наименьшее поле, содержащее $K$ и $\alpha$ (определение~\ref{def:generated-field}), а $K[\alpha]$ уже поле, содержащее $K$ и $\alpha$, получаем $K(\alpha) = K[\alpha]$.
    	
    	По теореме~\ref{thm:basis} $1, \alpha, \ldots, \alpha^{n-1}$ --- базис $K[\alpha]$ над $K$, поэтому $[K(\alpha):K] = n$ и это конечно (определение~\ref{def:degree}). Это доказывает (b) и (c).
    	
    	\emph{(a).} Сначала докажем общий факт: \emph{если $M/K$ --- конечное расширение (определение~\ref{def:degree}), то оно алгебраическое (определение~\ref{def:algebraic-extension})}\label{lem:finite-algebraic}. Пусть $[M:K] = d$ и $\beta \in M$ произвольный. Тогда $1, \beta, \beta^2, \ldots, \beta^d$ --- это $d+1$ элементов в $d$-мерном пространстве $M$ над $K$, значит, они линейно зависимы: существуют $c_0, \ldots, c_d \in K$, не все нулевые, такие что $c_0 + c_1\beta + \cdots + c_d\beta^d = 0$. Это ненулевой многочлен из $K[x]$, обращающийся в $\beta$ в ноль, значит, $\beta$ алгебраичен над $K$ (определение~\ref{def:algebraic}). Так как $\beta$ был произвольным, $M/K$ алгебраическое.
    	
    	Применяя это к $M = K(\alpha)$ (которое конечно над $K$ по пункту~\ref{thm:Kalpha-b}), получаем, что $K(\alpha)/K$ алгебраическое.
    \end{proof}
    
    \begin{theorem}[мультипликативность степени]\label{thm:tower}
    	Пусть $M/L/K$ --- башня конечных расширений полей (определения~\ref{def:extension}, \ref{def:degree}). Тогда
    	\[
    	[M:K] = [M:L] \cdot [L:K].
    	\]
    \end{theorem}
    \begin{proof}
    	Пусть $e_1, \ldots, e_m$ --- базис $L$ над $K$ (так что $[L:K]=m$), а $f_1, \ldots, f_n$ --- базис $M$ над $L$ (так что $[M:L]=n$); такие базисы существуют, так как $L/K$ и $M/L$ конечны (определение~\ref{def:degree}). Докажем, что $mn$ элементов $\{e_i f_j\}_{1\leqslant i \leqslant m, 1 \leqslant j \leqslant n}$ образуют базис $M$ над $K$.
    	
    	\emph{Порождающая система.} Пусть $x \in M$. Так как $f_1,\ldots,f_n$ --- базис над $L$, найдутся $\lambda_1,\ldots,\lambda_n \in L$ такие, что $x = \sum_j \lambda_j f_j$. Так как $e_1,\ldots,e_m$ --- базис $L$ над $K$, каждое $\lambda_j = \sum_i c_{ij} e_i$ для некоторых $c_{ij} \in K$. Подставляя,
    	\[
    	x = \sum_j \Big(\sum_i c_{ij} e_i\Big) f_j = \sum_{i,j} c_{ij} \, e_i f_j,
    	\]
    	то есть $x$ --- это $K$-линейная комбинация элементов $e_i f_j$.
    	
    	\emph{Линейная независимость.} Пусть $\sum_{i,j} c_{ij} e_i f_j = 0$ для $c_{ij} \in K$. Перегруппируем по $j$:
    	\[
    	\sum_j \Big(\sum_i c_{ij} e_i\Big) f_j = 0.
    	\]
    	Каждый коэффициент $\sum_i c_{ij} e_i$ лежит в $L$, а $f_1,\ldots,f_n$ линейно независимы над $L$, поэтому $\sum_i c_{ij} e_i = 0$ для каждого $j$. Но $e_1,\ldots,e_m$ линейно независимы над $K$, значит, $c_{ij} = 0$ для всех $i,j$.
    	
    	Итак, $\{e_i f_j\}$ --- базис $M$ над $K$ из $mn$ элементов, откуда $[M:K] = mn = [M:L]\cdot[L:K]$.
    \end{proof}
    
    \begin{theorem}\label{thm:sum-product}
    	Сумма и произведение алгебраических над $K$ элементов (определение~\ref{def:algebraic}) также алгебраичны над $K$.
    \end{theorem}
    \begin{proof}
    	Пусть $\alpha, \beta$ алгебраичны над $K$. Так как $\alpha$ алгебраичен над $K$, расширение $K(\alpha)/K$ конечно (теорема~\ref{thm:Kalpha}, пункт~\hyperref[thm:Kalpha-b]{b}). Так как $\beta$ алгебраичен над $K$ (существует ненулевой $g \in K[x]$ с $g(\beta)=0$), то $\beta$ алгебраичен и над большим полем $K(\alpha) \supset K$ (тот же многочлен $g$ годится, ведь $K[x] \subset K(\alpha)[x]$). Значит, $K(\alpha)(\beta)/K(\alpha)$ тоже конечно (снова теорема~\ref{thm:Kalpha}, пункт~\hyperref[thm:Kalpha-b]{b}, применённая к базовому полю $K(\alpha)$).
    	
    	По теореме~\ref{thm:tower} о мультипликативности степени,
    	\[
    	[K(\alpha,\beta):K] = [K(\alpha,\beta):K(\alpha)]\cdot[K(\alpha):K] < \infty,
    	\]
    	где $K(\alpha,\beta) := K(\alpha)(\beta)$. Значит, $K(\alpha,\beta)/K$ --- конечное расширение, а по факту, доказанному выше (см.~\hyperref[lem:finite-algebraic]{общий факт} в доказательстве теоремы~\ref{thm:Kalpha}), конечное расширение алгебраично. Так как $\alpha+\beta, \ \alpha\beta \in K(\alpha,\beta)$, оба этих элемента алгебраичны над $K$.
    \end{proof}
    
    \begin{theorem}[избавление от иррациональности в знаменателе]\label{thm:rationalize}
    	Дана дробь, числитель которой равен $1$, а знаменатель --- конечная сумма корней из целых чисел. Тогда существует равная ей дробь, такая что её знаменатель --- целое число, а числитель --- сумма (с целыми коэффициентами) корней из целых чисел.
    \end{theorem}
    \begin{proof}
    	Пусть $D = r_1 + r_2 + \cdots + r_k \ne 0$, где $r_i = n_i^{1/m_i}$ --- некоторый вещественный корень $m_i$-й степени из целого числа $n_i$. Каждое $r_i$ алгебраично над $\mathbb{Q}$ (определение~\ref{def:algebraic}; корень многочлена $x^{m_i}-n_i$), поэтому по теореме~\ref{thm:sum-product} (применённой несколько раз подряд к суммам) $D$ алгебраично над $\mathbb{Q}$.
    	
    	Пусть $f(x) = x^N + c_{N-1}x^{N-1} + \cdots + c_1 x + c_0$ --- минимальный многочлен $D$ над $\mathbb{Q}$ (определение~\ref{def:min-poly}); домножив на общий знаменатель коэффициентов, можно считать все $c_i \in \mathbb{Z}$. Заметим, что $c_0 \ne 0$: иначе $f(x) = x\cdot(x^{N-1}+\cdots+c_1)$, и так как $f(D)=0$ и $D\ne 0$, получили бы $D^{N-1}+\cdots+c_1=0$ --- ненулевое соотношение меньшей степени, чем $\deg f$, что противоречит минимальности $f$ (определение~\ref{def:min-poly}).
    	
    	Из $f(D)=0$ получаем
    	\[
    	D\cdot\big(D^{N-1}+c_{N-1}D^{N-2}+\cdots+c_1\big) = -c_0,
    	\]
    	то есть
    	\[
    	\frac{1}{D} = \frac{-\big(D^{N-1}+c_{N-1}D^{N-2}+\cdots+c_1\big)}{c_0}.
    	\]
    	Знаменатель $c_0$ --- целое число. Осталось проверить, что числитель --- сумма корней из целых чисел. Числитель является $\mathbb{Z}$-линейной комбинацией степеней $D^j$, а каждая степень $D^j$ (после раскрытия по полиномиальной формуле) --- это $\mathbb{Z}$-линейная комбинация одночленов вида $r_1^{a_1}r_2^{a_2}\cdots r_k^{a_k}$.
    	
    	Пусть $L = \mathrm{lcm}(m_1,\ldots,m_k)$. Тогда каждый такой одночлен сам является корнем степени $L$ из целого числа:
    	\[
    	r_1^{a_1}\cdots r_k^{a_k} = n_1^{a_1/m_1}\cdots n_k^{a_k/m_k} = \Big(n_1^{a_1 L/m_1}\cdots n_k^{a_k L/m_k}\Big)^{1/L},
    	\]
    	а показатели $a_iL/m_i$ --- целые числа, так как $m_i \mid L$. Значит, числитель --- это целочисленная линейная комбинация корней степени $L$ из целых чисел, то есть сумма (с учётом знаков и повторений) корней из целых чисел, что и требовалось.
    \end{proof}
    
    \begin{theorem}[квадратичное расширение полей]\label{thm:quadratic}
    	Пусть $L/K$ --- расширение полей (определение~\ref{def:extension}), $\operatorname{char} K \ne 2$, и $\sqrt{\alpha} \notin K$ ни для какого $\alpha \in K$, не являющегося квадратом в $K$ (т.е. рассматриваем нетривиальный случай). Тогда
    	\[
    	L = K(\sqrt{\alpha}) \text{ для некоторого } \alpha \in K,\ \sqrt\alpha \notin K \quad \Longleftrightarrow \quad [L:K] = 2.
    	\]
    \end{theorem}
    \begin{proof}
    	\emph{($\Rightarrow$)} Пусть $L = K(\sqrt{\alpha})$ (определение~\ref{def:generated-field}), где $\alpha \in K$ не является квадратом в $K$. Тогда многочлен $x^2-\alpha$ не имеет корней в $K$, а значит (будучи степени 2) неприводим над $K$. Он же обнуляется на $\sqrt\alpha$, значит, он и есть минимальный многочлен $\sqrt\alpha$ над $K$ (определение~\ref{def:min-poly}), и его степень равна $2$. По теореме~\ref{thm:Kalpha}, пункт~\ref{thm:Kalpha-c}, $[K(\sqrt\alpha):K]$ равно степени $\sqrt\alpha$, то есть $[L:K]=2$.
    	
    	\emph{($\Leftarrow$)} Пусть $[L:K]=2$ (определение~\ref{def:degree}). Возьмём любой $\beta \in L \setminus K$. Так как $\dim_K L = 2$, а $1, \beta$ линейно независимы над $K$ (если бы $\beta = c\cdot 1$ для $c\in K$, то $\beta \in K$ --- противоречие), пара $\{1,\beta\}$ --- базис $L$ над $K$.
    	
    	Тогда $\beta^2 \in L$ раскладывается по этому базису: $\beta^2 = a + b\beta$ для некоторых $a,b \in K$, то есть $\beta$ --- корень многочлена $x^2 - bx - a \in K[x]$.
    	
    	Так как $\operatorname{char} K \ne 2$, элемент $2$ обратим в $K$, и можно выделить полный квадрат: положим $\gamma := \beta - \dfrac{b}{2} \in L$. Тогда
    	\[
    	\gamma^2 = \beta^2 - b\beta + \frac{b^2}{4} = (a+b\beta) - b\beta + \frac{b^2}{4} = a + \frac{b^2}{4} =: \alpha \in K.
    	\]
    	Значит, $\gamma = \sqrt{\alpha}$ для некоторого $\alpha \in K$. Так как $b/2 \in K$, то $K(\gamma) = K(\beta)$ (определение~\ref{def:generated-field}; каждое выражается через другое с коэффициентами из $K$), и $K(\beta) = L$ (так как $\beta \notin K$, а $[L:K]=2$, поле $K(\beta)$ строго больше $K$, значит, совпадает с $L$). Наконец, $\alpha$ не является квадратом в $K$: иначе $\gamma \in K$, откуда $\beta = \gamma + b/2 \in K$ --- противоречие.
    	
    	Итак, $L = K(\gamma) = K(\sqrt\alpha)$ для некоторого $\alpha \in K$, не являющегося квадратом в $K$.
    \end{proof}
