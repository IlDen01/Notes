	\begin{enumerate}
		\item $(x^2 - 1)y' + 2xy^2 = 0$
		
		$y' = - \dfrac{2xy^2}{x^2 - 1}, x \ne \pm 1$
		
		$\dfrac{dy}{y^2} = -\dfrac{2xdx}{x^2 - 1}$
		
		$- \dfrac{1}{y} = - \ln |x^2 - 1| + C$
		
		Итого решение: $\dfrac{1}{y} = \ln |x^2 - 1| + C$ и $y = 0$.
	\end{enumerate}
