	\section{Политология}
	\begin{definition}
		Политология~--- (с греческого) искусство управления полисом (городом). Человек не участвующий в жизни полиса~--- идиот.
	\end{definition}
	\noindent
	Политика строится вокруг государства.
	\subsection{Государство}
	\begin{definition}
		Государство~--- политический институт с научной точки зрения; политическая организация обладающая властью на определенной территории. С бытовой~--- страна, народ страны.
	\end{definition}
	\begin{person}[Никколо Макиавелли]
		Флорентийский (Итальянский) мыслитель $16$ века, впервые заговоривший о государстве.
	\end{person}
	\subsubsection{Признаки государства}
	\paragraph{Основные признаки государства}
	\begin{itemize}
		\item Суверинетет:
		\begin{itemize}
			\item Внутренний~--- полнота власти на своей территории; у государства есть право на легальное применение силы.
			\item Внешний~--- независимость от других государств.
		\end{itemize}
		У государства должен быть и внутренний и внешний суверенитет, иначе это не государство.
		\item Территория. У государства обязательно должна быть своя территория, иначе тяжело управлять государством, сбором налогов и тому подобное.
		\item Наличие государственной власти. Власть состоит из двух частей:
		\begin{itemize}
			\item Аппарат управления.
			\item Аппарат принуждения.
		\end{itemize}
		\item Регулярный сбор налогов.
		\item Законодательная деятельность.
	\end{itemize}
	\paragraph{Дополнительные признаки}
	\begin{itemize}
		\item Наличие собственной государственной символики (герб, гимн, флаг, особые атрибуты).
		\item Государственный язык.
		\item Гражданство.
		\item Обладание определенными материальными ресурсами.
	\end{itemize}
	\subsubsection{Функции государства}
	\paragraph{Внутренние функции государства}
	\begin{itemize}
		\item Правовая~--- обеспечение правопорядком, законности, борьба с преступностью.
		\item Экономическая~--- организация хозяйственной жизни страны.
		\item Социальная~--- выполнение обязательств перед обществом (система здравоохранения, система образования, поддержка материнства и детства, помощь малоимущим, выплата пенсий и так далее).
		\item Образовательная (иногда выделяется из социальной).
		\item Экологическая функция.
	\end{itemize}
	\paragraph{Внешние функции государства}
	\begin{itemize}
		\item Оборонная~--- защита границ и противодействие внешним угрозам.
		\item Внешне-политическая (дипломатическая).
		\item Международное сотрудничество.
	\end{itemize}
	\subsection{Политический режим}
	\begin{definition}
		Политический режим~--- то, как совершается управление государством.
	\end{definition}
	\subsubsection{Основные типы политических режимов}
	\begin{enumerate}
		\item Демократический~--- власть в основном договаривается с обществом.
		\item Тоталитарный~--- метод подчинения, навязывания, насилия.
		\item Авторитарный~--- комбинация демократического и тоталитарного режимов.
	\end{enumerate}
	\subsubsection{Тоталитарный режим}
	\begin{definition}
		От латинского totalize~--- полный; вся полнота власти над обществом сосредоточена в руках государства, отсутствие предела контроля.
	\end{definition}
	\paragraph{Признаки тоталитаризма}
	\begin{itemize}
		\item Наличие единственной господствующей идеологии, остальные являются враждебными и преследуются.
		\item Однопартийность.
		\item Экономика подчинена государству, свободного рынка нет.
		\item Опора на насилие, массовый террор.
	\end{itemize}
	\subsubsection{Демократический режим}
	\begin{definition}
		Противоположность тоталитаризма. Государство существует для народа. Основной источник власти~--- народ.
	\end{definition}
	\paragraph{Признаки демократии}
	\begin{itemize}
		\item Многопартийность.
		\item Свободный рынок.
	\end{itemize}
	\paragraph{Типы демократии}
	\begin{itemize}
		\item Прямая~--- граждане принимают непосредственное участие в управлении государством. Референдум~--- всенародное голосование по особо важным вопросом; пример современной прямой демократии. Особенность в том, что референдум не имеет обязательной юридической силы, в отличии от плебисцита.
		\item Представительная~--- выбор представителей в органы государственной власти.
	\end{itemize}
	\subsubsection{Авторитарный режим}
	\begin{definition}
		Авторитарный режим~--- тип не демократического политического режима, основанного на несменяемой централизованной власти одного лица или группы лиц при сохранении в стране определенного уровня экономических, гражданских и идейных свобод; осуществляется полный контроль, зачастую, только власти. Самый распространенный тип за всю историю.
	\end{definition}
	\subsection{Форма государства}
	\begin{definition}
		Форма государства~--- особенность внутренней организации государства, которая охватывает форму правления, государственное устройство и политический режим.
	\end{definition}
	\subsubsection{Формы государства}
	\begin{enumerate}
		\item Монархия.
		\item Республика.
	\end{enumerate}
	\subsubsection{Монархия}
	\begin{definition}[Монархия]
		Власть принадлежит одному лицу.
	\end{definition}
	\paragraph{Признаки монархии}
	\begin{itemize}
		\item Власть передается по наследству.
		\item Власть бессрочна.
		\item Власть неподвластна населению.
	\end{itemize}
	\begin{note}
		Самая древняя форма правления.
	\end{note}
	\paragraph{Древние формы монархий}
	\begin{enumerate}
		\item Деспотия~--- власть правителя схожа с властью хозяина над рабами. Примеры: государства древнего Востока, древний Египет, древний Китай.
		\item Теократия~--- монарх представляется как живой бог (или наместник бога).
		\item Абсолютная монархия~--- вся власть (законодательная, исполнительная, судебная, военная и так далее) сконцентрирована в руках одного человека. Примеры: государства Ближнего Востока, Франция при Людовике XIV.
		\item Просвещенная монархия~--- предполагает определенные обязанности государственной власти. Монарх~--- не просто единоличный правитель, а образованный, просвещенный государь, который обязан заботится о просвещении своего народа и благе государства. Пример: Россия при Екатерине Великой. Важный шаг к демократии.
	\end{enumerate}
	\paragraph{Современные формы монархий}
	\begin{enumerate}
		\item Абсолютная монархия~--- вся полнота власти в руках одного человека.
		\item Конституционная монархия~--- полномочия монарха ограничены конституцией.
		\begin{enumerate}
			\item Дуалистическая монархия~--- у монарха обширная власть. У парламента мало власти.
			\item Парламентская монархия~--- у монарха минимальная власть. По большей части функции монарха~--- символические. Законы принимает парламент. В некоторых случаях у монарха есть право вето, но этим право они не пользуются.
		\end{enumerate}
	\end{enumerate}
	\subsubsection{Республика}
	\begin{definition}[Республика]
		Законодательная власть принадлежит выборному представительному органу (парламенту). Глава государства (президент) либо избирается всем населением, либо специальной избирательной комиссией.
	\end{definition}
	\paragraph{Признаки республики}
	\begin{enumerate}
		\item Выборность власти.
		\item Срочность власти.
		\item Зависимость от воли избирателей.
	\end{enumerate}
	\paragraph{Типы республики}
	\begin{enumerate}
		\item Президентская. Президент~--- глава исполнительной власти и государства. Избирается, как правило, прямым голосованием всех граждан. У президента большая власть и большой объем полномочий, но несмотря на все это он~--- не монарх, и его возможности ограничены парламентом.
		\item Парламентская~--- правительство формируется парламентом, и перед парламентом же несет ответственность. Парламент может выразить недовольство правительству или отдельным министрам, что может спровоцировать правительственный кризис. Глава государства~--- тоже президент, но полномочия весьма скромные, в отличии от полномочий в президентской республике; в основном представительные функции (мало чем отличаются от функций монарха в парламентской монархии).
		\item Смешанная~--- высшие властные полномочия распределенны между президентом и парламентом.
	\end{enumerate}
	\subsection{Формы государственного устройства}
	\begin{definition}
		Государственное устройство~--- способ взаимодействия частей государства.
	\end{definition}
	\begin{enumerate}
		\item Унитарное государство~--- части такого государства подчинены центру. Единая конституция, единое право. Примеры: Великобритания, Польша, Финляндия, Китай.
		\item Федерация~--- объединение, союз. Части федерации обладают признаками суверенитет (имеют право на собственную конституцию, законодательство, судебную систему и тому подобное). Два уровня власти: федеральная (высшая) и власть субъектов. Примеры: США, Германия, Индия, Канада, Российская Федерация.
		\item Конфедерация~--- не государство, а объединение разных, абсолютно независимых политических субъектов (возможно государств). Примеры: Речь Посполитая.
		\item Содружество государств~--- новоя форма, которая частично заменили федерацию. Примеры: ЕС, СНГ, Британское содружество наций.
	\end{enumerate}
	\subsection{Политические партии и организации}
	\begin{definition}
		Партия~--- группа лиц, которых объединяют общие интересы.
	\end{definition}
	\begin{definition}
		Политическая партия~--- политическая организация, которая выражает интересы определенных групп и слоев населения, которая борется за власть и осуществляет государственную власть в своих интересах.
	\end{definition}
	\begin{note}
		Первые полит кружки появились в начале 19 века~--- кружки декабристов. Но они не были партиями, так как не боролись за власть.
	\end{note}
	\subsubsection{Функции политических партий}
	\begin{enumerate}
		\item Организационная (завлечение людей в партию).
		\item Представительная (выражение и представление определенных слоев).
		\item Идеологическая (пропаганда своих идей).
	\end{enumerate}
	\subsubsection{Признаки политических партий}
	\begin{enumerate}
		\item Стремление к завоеванию власти.
		\item Наличие идеологии.
		\item Устойчивость.
		\item Наличие организации (устав, программа и членство). Программа~--- что партия будет делать, когда придет к власти. Устав~--- документ. в котором прописаны нормы внутрипартийной жизни. Членство~--- партия структурно состоит из руководства (люди, профессионально занятые политикой), передовых людей и социальной базы.
	\end{enumerate}
	\subsubsection{Виды политических партий}
	\begin{enumerate}
		\item По идеологическому признаку.
		\begin{itemize}
			\item Либерализм. Свобода и права человека. Появились в $17 - 18$ веке, раньше консерваторов. \\
			Консерватизм. Сохранение старого и традиционных ценностей; людей надо научить пользоваться свободой, и лучше всего их этому научит государство. \\
			Обе идеологии выражали интересы собственников.
			\item Социал-демократизм. За социальную справедливость. чем беднее человек, тем меньше он платит налогов; социальная политика. Но при всем этом сохранение частной собственности. \\
			Коммунизм. Частная собственность~--- основа для эксплуатации человека человеком, поэтому ее надо ликвидировать. \\
			Выражали интересы трудящихся, в первую очередь рабочего класса.
			\item Националистические партии.
			\begin{itemize}
				\item Умеренный национализм. Регулирование потоков миграции.
				\item Радикальный национализм. Отмена миграции, этнические чистки. Такие националистические партии под запретом.
			\end{itemize}
		\end{itemize}
		\item По шкале политического спектра.
		\begin{itemize}
			\item Правые. Те, кто отстаивает силы государства, охраняющие частную собственность.
			\item Центристские. За компромис, соединение государственных и общественных интересов.
			\item Левые. Те, кто больше ориентирован на общество и общественные интересы.
		\end{itemize}
		\item По отношению к другим элементам политической системы.
		\begin{itemize}
			\item Демократические. Терпимость по отношению к другим партия, сотрудничество и ведения диалогов с ними.
			\item Не демократические. Не намерены вести диалоги, стремление подчинить себе другие партии, институты, государство.
		\end{itemize}
		\item По положению в отношении правления.
		\begin{itemize}
			\item Правящие. Те, кто победили на выборах и правят.
			\item Оппозиция. Те, кто не у власти. Как правило критикуют правящую партию, ждут их ошибки и пользуются этим, чтобы в будущем самим прийти к власти.
			\begin{itemize}
				\item Системная оппозиция. Представлена в парламенте.
				\item Несистемная оппозиция. Легальная в парламенте не представлена, но и не запрещена. Нелегальная~--- законом запрещена.
			\end{itemize}
		\end{itemize}
		\item По численности.
		\begin{itemize}
			\item Массовые. Заинтересована в максимальном количестве своих членов. Главный источник дохода таких партий~--- членские взносы.
			\item Кадровые. Имеют другое финансирование. Существуют за счет финансовой помощи спонсоров.
		\end{itemize}
	\end{enumerate}
	\subsubsection{Политические движения}
	\begin{definition}
		Политические движения~--- отличаются от партий тем, что не участвуют в борьбе за власть. Также отсутствуют членство и организационное единство. В политической жизни участвуют, пытаясь воздействовать на власть и заявить о себе.
	\end{definition}
	\subsection{Партийные системы и их виды}
	\begin{definition}
		Партийная система~--- система взаимодействия между партиями.
	\end{definition}
	\subsubsection{Виды партийных систем}
	\begin{itemize}
		\item Однопартийная. Характерна для стран с не демократическими политическими режимами. При этом может сохраняться видимость многопартийности. Таким образом однопартийные политические системы делятся на:
		\begin{itemize}
			\item Реально однопартийные. Пример: СССР (партия КПСС).
			\item Формально многопартийные. Не смотря на существование нескольких партий власть контролируется одной партией. Пример: Китай, Северная Корея.
		\end{itemize}
		\item Двухпартийная. Характерна для стабильных, устоявшихся политических режимов. Пример: США, Великобритания, Австралия. Каждая из двух партия способна прийти к власти и сформировать правительство. При этом партий может быть больше. Причины двухпартийности:
		\begin{itemize}
			\item Бипартизм всегда имеет исторические корни.
			\item Психологическое значение.
		\end{itemize}
		Сильная сторона двухпартийной системы~--- обеспечение стабильности и порядка (одна партия правит, другая ее критикует, а потом они меняются местами). Но слабая сторона~--- неспособность такой системы выразить все многообразие мнений и взглядов народа.
		\item Многопартийная. Характеризует ситуаций когда партий не просто много/мало, а когда не одна из существующих партий не располагает достаточной поддержкой избирателей, чтобы самостоятельно прийти к власти. Для того, чтобы организовать правительство партиям необходимо объединяться и вступать в коалиции. Преимущества такой системы в том, что она лучше отображает интересы разных социальных групп. Но ее главный недостаток в том, что она не обеспечивает такой уровень стабильности как двухпартийная система.
		\item Двух с половиной партийная. Ни одна из соперничающих партий оказывается не в состоянии победить на выборах, набрать необходимое для организации правительства голосов. В такой ситуации несколько процентов голосов, которые может им дать какая-нибудь малозначительная партия, принесут им победу; и тогда эти партии объединяются и маленькая партия получает большое политическое значение.
		\item С доминирующей партией. Суть в том, что одна и та же партия в течении длительного времени побеждает всех своих конкурентов и находится у власти. Это может длиться годами, десятилетиями. Например либерально-демократическая партия Японии находилась у власти без перерыва $38$ лет, причем в стране не однопартийная система. Похожая ситуация была в Индии. Причин может быть несколько: люди просто искренне поддерживают партию, неосознанность населения ("лишь бы не стало хуже, поэтому менять ничего не будем").
	\end{itemize}
	\subsection{Разбор политики на примерах}
	\subsubsection{Великобритания}
	\begin{note}
		В Великобритании парламентская монархия. Король/королева несет только символическую роль. Основная власть принадлежит премьер-министру. Зачастую он~--- представитель партии, победившей на выборах в нижней палате. Нижняя палата является главной в английском парламенте, так как она общая и избирается на $5$ лет народным голосованием. Также есть верхняя палата Лордов, которая не избирается. До недавнего времени она формировалась по наследственному признаку. Сейчас же наследственным путем формируется $60\%$ палаты (наследственные пэры), а остальные получают место в палате за выдающиеся достижения (пожизненные пэры). Во главе верховной палаты~--- лорд-канцлер, который назначается монархом на 5 лет.
	\end{note}
	\begin{note}
		По форме государственного устройства Великобритания~--- унитарное государство, которое состоит из Англии, Уэльса, Шотландия и Северная Ирландия. Хотя все ее части претендуют на самостоятельность.
	\end{note}
	\begin{note}
		В Великобритании двухпартийная система. Партии~--- лейбористская и консервативная.
	\end{note}
	\subsubsection{США}
	\begin{note}
		Президент представляет верховную власть. В его руках исполнительная власть, он руководит деятельностью правительства, внутренней и внешней политикой. Избирается на $4$ года, но не по средствам прямых выборов, а коллегией выборщиков. То есть население голосует не за кандидатов на пост президента, а за выборщиков. А сами выборщики уже голосуют за республиканцев или демократов. Так исторически сложилось, так как при всеобщих выборах: голоса малых штатов будут не сильно учитываться; когда принималось это законодательство в США не было сильных политических штатов, СМИ, и народ, в основной своей массе, был мало образован и мало грамотен, и тогда симпатией избирателей могли бы завладеть популисты, демагоги, болтуны и тому подобное, и поэтому было поручено оставить выборы президента уважаемым людям~--- выборщикам. Характерной чертой политической системы США является ее двухпартийность~--- демократы и республиканцы. Американский парламент называется \textbf{Конгресс}. Он состоит из двух палат: верхней палаты (сенат) и нижней (палаты представителей). В сенат входят по два представителя от каждого штата. Нижняя палат представляет американское общество в целом.
	\end{note}
	\begin{note}
		США~--- федеративной государство. Представляет собой $50$ штатов и один федеральный округ. У каждого штата свои законы, которые могут сильно различаться.
	\end{note}
	\subsubsection{Германия}
	\begin{note}
		Парламентская республика. Глава государства~--- федеральный президент. Он избирается специально созываемым для этой цели органом~--- федеральным собранием, сроком на $5$ лет и реальной властью не обладает. Играет такую же символическую роль, как и Английская королева. Он формально назначает и увольняет министров, но предлагает кандидатуры министров канцлер~--- глава правительства. Президент теоретически может не подписать федеральный закон (но за всю историю Германии такое было только $8$ раз). Также он может объявить помилование (помиловать преступников, осуществить акт государственного милосердия). Президент, в основном, фигура символическая, надпартийная, он олицетворяет все немецкое общество, представляет все государство. Пока он находится в должности президента, он прекращает партийную деятельность. Реальной властью обладает канцлер~--- глава правительства. Он избирается на $4$-х летний срок бундестагом (парламент). И у него огромная власть. Он определяет внешнюю и внутреннюю политику. А Бундестаг издает федеральные законы и контролирует правительство. На ряду с Бундестагом есть еще Бундесрат~--- орган представления земель Германии ($16$ федеральных частей).
	\end{note}
	\begin{note}
		Многопартийная система. Крупнейшие партии: христианский-демократический союз и христианский-социальный союз. ХСС действует в основном на территории Баварии, а ХДС действует в остальных землях. ХСС и ХСД~--- блок умеренных консерваторов. Важны христианские ценности. Одна из старейших и влиятельных партий~--- Социал-демократическая партия германии. Известна антифашисткой борьбой.
	\end{note}
	\subsection{Избирательные системы}
	\begin{definition}
		Избирательная система~--- то, как организованы выборы в той или иной стране. Включает в себя теоретическую часть (избирательное право) и практическую часть (избирательный процесс). Избирательное право делится на: объективное избирательное право (законы, на основе которых осуществляются выборы) и субъективное право (право избирать и право избираться). Субъективное право имеет возрастные ограничения: избирать (в России) можно только с $18$ лет, а избираться можно в зависимости от уровня выборов в разном возрасте (например в депутаты разных уровней можно избираться с $21$ года; на должность главы субъекта можно претендовать с $30$ лет; а на должность президента с $35$ лет).
	\end{definition}
	\subsubsection{Принципы демократической избирательной системы}
	\begin{enumerate}
		\item Принцип равенства.
		\item Всеобщность. Отстранение от выборов после преступления. Ограничение, накладываемые на принцип всеобщности~--- ценз (основные цензы~--- возрастной, гражданский; также есть половой (чистая дискриминация), имущественный (люди с имуществом заинтересованы в стабильности), военный (в некоторых государствах не допускаются военные), грамотности).
		\item Принцип тайного голосования. Голосовать надо тайно, чтобы на избирателей нельзя было повлиять.
		\item Принцип состязательности. Выбор должен быть по крайней мере из двух кандидатов.
		\item Принцип гласности. Выборы должны быть открытыми и публичными. Можно наблюдать и за процессом выборов и за процессом подсчетов голосов.
		\item Принцип прямого голосования.
	\end{enumerate}
	\subsection{Системы выборов}
	\subsubsection{Мажоритарная система}
	\begin{definition}[Мажоритарная система (фр. Majoritaro~--- большинство)]
		При такой системе можно избрать только одного человека, набравшее установленное большинство. Кандидаты могут представлять партии, а могут быть самовыдвиженцами. Большинство бывает \textbf{относительными} (больше, чем остальные) и \textbf{абсолютным} (более $50 \%$).
	\end{definition}
	\paragraph{Плюсы мажоритарной системы}
	\begin{enumerate}
		\item Хорошая связь между избирателями и депутатами.
	\end{enumerate}
	\paragraph{Минусы мажоритарной системы}
	\begin{enumerate}
		\item Низкий уровень легитимности.\\Рассмотрим ситуацию, при которой за победителя проголосовали  $51\%$, тогда оставшиеся $49\%$ голосовали за других кандидата.
		\item Не ограничивает популизм. Кандидат может многое пообещать и ничего не исполнить.
		\item Не способствует развитию партийной системы.
	\end{enumerate}
	\subsubsection{Пропорциональная система}
	\begin{definition}[Пропорциональная система]
		Вид выборов, при котором партии получают места в представительном органе пропорционально числу голосов, отданых за них.
	\end{definition}
	\paragraph{Плюсы пропорциональной системы}
	\begin{enumerate}
		\item Более справедливая система. Партии выбираются в соответствие с их популярностью. Каждый избиратель найдет себе партию, будет минимум пропавших голосов.
		\item Способствует развитию партийной системы.
		\item Защита от популизма. Депутатов могут призвать к ответу внутри партии.
	\end{enumerate}
	\paragraph{Минусы пропорциональной системы}
	\begin{enumerate}
		\item В парламент могут попасть мелкие партии, выражающие интересы очень маленькой группы. Для этого существует \textbf{\textit{избирательный порог}}. Избирательный порог в России~--- $5\%$.
		\item Выбираются не люди, а партии. Партия формирует списки депутатов, которые обычно заранее не оглашаются.
	\end{enumerate}
	\subsubsection{Смешанная система}
	\begin{definition}[Смешанная система]
		Способ проведения выборов, при котором часть депутатов (или представителей) избирается по одной системе, а другая часть — по другой. Чаще всего это комбинация мажоритарной и пропорциональной систем.
	\end{definition}
	\paragraph{Плюсы смешанной системы}
	\begin{enumerate}
		\item Сбалансированность: Сочетает в себе преимущества обеих систем.
		\item Связь с избирателями: Депутаты, избранные по округам, представляют интересы конкретных территорий и должны работать с запросами своих избирателей.
		\item Представительство партий: Пропорциональная часть позволяет отразить в парламенте весь спектр политических предпочтений общества, давая шанс малым партиям.
		\item Стабильность и разнообразие: Система может способствовать как формированию стабильного большинства (за счет мажоритарной части), так и плюрализму мнений (за счет пропорциональной).
	\end{enumerate}
	\paragraph{Минусы смешанной системы}
	\begin{enumerate}
		\item Сложность для избирателя: Гражданину бывает сложно разобраться в двойном механизме голосования.
		\item Риск конфликта интересов: В парламенте могут возникнуть две ``касты'' депутатов: ``окружники'' и ``списочники'', которые могут по-разному понимать свою ответственность.
		\item В несвязанной системе: Результаты могут быть непропорциональными. Крупная партия, выиграв много округов, может получить и так большое количество мест по спискам, что ее общее представительство будет превышать ее реальную поддержку в стране.
		\item Манипуляции: Партии могут использовать ``технических'' кандидатов в округах или сложные схемы формирования списков для увеличения своего представительства.
	\end{enumerate}
	\subsection{Политическая власть}
	\begin{definition}[Власть]
		Способность осуществлять свою волю в отношении кого-то или чего-то. Власть это всегда отношение двух сторон: субъекта (того, кто осуществляет власть) и объекта (того, на кого власть направлена).
	\end{definition}
	\subsubsection{Признаки политической власти}
	\begin{enumerate}
		\item Верховенство власти: ее решение обязательно для всех.
		\item Легальность (законность): политическая власть должна быть законна.
		\item Централизованность (моноцентричность): существование общего государственного центра принятия решений.
		\item Многообразие ресурсов.
		\item Публичность: она действует от имени всего общества.
	\end{enumerate}
	\subsubsection{Природа власти}
	\begin{enumerate}
		\item Теологическая теория: власть дана от бога, всякий правитель, осуществляющий ее, осуществляет волю бога.
		\item Биологическая теория: стремление к власти~--- проявление инстинкта выживания, который характерен для всего живого.
		\item Элитарная теория: происхождение политической власти из общественного неравенства.
		\item Классовая теория (Марксистская концепция): связывает власть с отношениями собственности, те политическая власть с точки зрения этой теории~--- порождение экономической власти.
		\item Системная теория: понимание общества как сложной саморегулируемой системы, которая стремиться к своей стабильности, и власть обеспечивает стабильность.
	\end{enumerate}
	\subsubsection{Власть по способам осуществления власти}
	\begin{enumerate}
		\item Тоталитарная: насилие.
		\item Авторитарная: насилие и убеждение.
		\item Демократическая: убеждение.
	\end{enumerate}
	\subsubsection{Власть по субъекту}
	\begin{enumerate}
		\item Автократия: неограниченное и бесконтрольное правление одного человека.
		\item Аристократия: власть ``лучших''.
		\item Меритократии: власть ``достойных''.
		\item Олигополия: власть ``немногих''.
		\item Демократия: власть народа.
	\end{enumerate}
	\subsubsection{Структура политической власти}
	\begin{itemize}
		\item Субъект: то, что совершает власть.
		\item Объект: то, на что власть направлена.
		\item Ресурсы: то, посредствам чего власть достигает своих целей. Ресурсы бывают:
		\begin{itemize}
			\item Экономические: материальные ценности и блага.
			\item Социальные: возможность воздействовать на человека повышая или понижая его социальный статус.
			\item Информационные: информация и средства ее распространения.
			\item Силовые: средства принуждения и специально обученные и подготовленные люди для применения этих средств.
		\end{itemize}
		\item Методы: поощрение, наказание, убеждение и принуждение. Самый хороший метод~--- убеждение.
	\end{itemize}
	\subsubsection{Легальность и легитимность власти}
	\begin{definition}[Легальность]
		Действие по четко прописанным законам.
	\end{definition}
	\begin{definition}[Легитимность]
		Доверие народа к власти.
	\end{definition}
	\paragraph{Три типа легитимности власти}
	\begin{enumerate}
		\item Легальная легитимность: в этом случае легальность и легитимность совпадают; главным мотивом подчинения власти является интерес. В политической системе, построенной на легальной легитимности, люди подчиняются не какой-то конкретной личности, а подчиняются законам, при чем подчиняются все: и общество, и сама власть.
		\item Традиционная легитимность: основана не на интересе, а на вере; люди верят, что существующие традиции и порядки являются правильными и справедливыми. Роль закона очень не велика.
		\item Харизматическая легитимность: особое политическое обаяние, которое заставляет людей следовать за данным человеком.
		\item Персональная легитимность: основана на вере людей в личные качества лидеров.
		\item Идеологическая: основана на религии.
	\end{enumerate}
	\section{Идеология}
	\begin{definition}
		Идеология~--- система взглядов, представлений людей, выражающих интересы того или иного общества, или той или иной социальной группы.
	\end{definition}
	\begin{person}[Антуан Дестют де Траси, 1754 -- 1836]
		Создатель слова ``идеология'', лидер движения ``идеологов''. Пытался создать науку идей, те как они возникают, рождаются, как связаны с жизнью людей.
	\end{person}
	\subsection{Идеи Маркса}
	\begin{itemize}
		\item Буржуазная идеология (болтовня; обман трудящихся, так как надо было оправдать свое превосходство).
		\item Идеология пролетариата (научная и хорошая).
	\end{itemize}
	\begin{note}
		После него идеологию стали наполнять политическим смыслом.
	\end{note}
	\subsection{Функции идеологий}
	\begin{enumerate}
		\item Ориентационная: цель.
		\item Мобилизационная: привидение в людей действие.
		\item Интеграционная: идеология способна объединить людей.
		\item Социально-классовая: идеология порождает и защищает идеи определенных социальных групп.
	\end{enumerate}
	\subsection{Идеологии по признакам}
	\subsubsection{По шкале политического спектра}
	\begin{enumerate}
		\item Оценка роли государства, усиление его роли~--- правая.
		\item Роль общества должна повышаться, роль государства должна уменьшаться~--- левая.
	\end{enumerate}
	\subsubsection{По глубине преобразований в обществе}
	\begin{enumerate}
		\item Радикальные: резкое изменение, разрушение.
		\begin{itemize}
			\item Анархизм: резкое уничтожение государства.
			\item Марксизм: постепенное упразднение государства.
		\end{itemize}
		\item Умеренные: постепенное улучшение жизни общества путем осторожных реформ.
		\begin{itemize}
			\item Либерализм.
			\item Социал-демократизм.
		\end{itemize}
		\item Консерваторы: не допускаются никакие изменения.
		\begin{itemize}
			\item Монархизм.
			\item Социализм.
		\end{itemize}
	\end{enumerate}
	\subsection{Три основные идеологии современности}
	\subsubsection{Либерализм}
	\paragraph{Классический либерализм}
	\begin{note}
		Основу классического либерализма заложили такие мыслители, как Адам Смит, Вольтер, Монтескье, Франклин, Джеферсон в 17 -- 18 века. Право на жизнь, свободу, личную неприкосновенность, собственность. Принцип индивидуализма, реализуемый в социальный, экономической, политической и духовной сферах. В социальной сфере: не может быть преимущества у одних людей, над другими, все люди равны, <<Свобода, равенство, братство!>> В экономической сфере: каждый в праве владеть собственностью, преумножать, распоряжаться по своей воле; идея свободного рынка и ничем не ограниченной конкуренции; ограничение вмешательства государства в экономику. В политической сфере была сформулирована идея народа как источника власти, все могут учавствовать в совершении власти; также либерализм ограничивает возможность государственного вмешательства в частную жизнь людей; народный суверенитет. Идея твердого индивидуализма: на помощь государства в плохой ситуации человек расчитывать не должен; человек сам хозяин своей судьбы. В духовной сфере либерализм проводит политику культурного легитимизма: все культуры являются разными, но равными.
	\end{note}
	\paragraph{Нео-либерализм}
	\begin{definition}[Нео-либерализм]
		В 20 веке либерализм подвергся изменениям. Возникла идея необходимости социальной поддержки государства, но сохранения главного посыла либерализма о необходимости соблюдать вечные свободы и права человека. Возникла такая идея из-за большого количества кризисов, великой депрессии.
	\end{definition}
	\subsubsection{Консерватизм}
	\paragraph{Классический консерватизм}
	\begin{note}[Зарождение]
		Зарождение консерватизма связывают с публикацией работы Эдмунда Берка <<Размышления о революции во Франции>> в 1790 году. Его идеи были связаны с тем, что все вернется в конечном итоге к началу. Другие представители консерватизма: Жозеф де Местр.
	\end{note}
	\begin{note}[Главные идеи]
		От латинского conservo~--- сохранять. Их взгляды полностью противоположны взглядам либералов. Консерваторы считали, что государство должно вмешиваться в экономику. В политической сфере: государство первично. За людьми надо присматривать, и надо учить их распоряжаться свободами, а сделать это может только государство. Власть должна находить только в руках элиты, так как если народу отдать власть, то будет только хаос. В социальной сфере консерваторы сторонники неравенства, так как в природе равенства нет, оно противоестественно, а человеческое общество~--- часть природы. В духовной сфере консерваторы отрицают культурный легитимизм, и выступают за культурную иерархи. Человечество может существовать и нормально развивать только опираясь на незыблемые ценности, такие как: государство, семья, церковь, мораль; эти ценности формировались веками и их нельзя отрицать.
	\end{note}
	\paragraph{Нео-консерватизм}
	\begin{note}[Появление]
		Появился в конце 20 века.
	\end{note}
	\begin{note}[Идеи]
		Государство должно уйти из экономики. В обществе должно существовать равенство по признаку рассы и пола, но не должно быть социального равенства, так как если оно будет, то прекратиться развитие общества.
	\end{note}
	\subsubsection{Социализм}
	\paragraph{Классический марксизм}
	\begin{note}
		Появился как выражение взглядов и интересов дворянства. Политическая идеология, сторонники которой хотели бы заменить капиталистическое общество, основанной на частной собственности, угнетении, другим, более справедливым общественным устройством. Наиболее распространенным социалистическим учением в 19 веке стал <<Марксизм>>. Путь к коммунизму лежит через революцию, так как буржуазия просто так не отдаст свои богатства, и диктатуру пролетариата, так как буржуазия будет оказывать сопротивление, а пролетариат должен быть сильным, чтоб справиться с ним, а для этого на какое-то время надо сохранить государство как аппарат принуждения. Только после окончательно победы над буржуазией можно будет установить коммунистическое общество, в котром не будет ничего частного.
	\end{note}
	\begin{note}
		На рубеже 19 -- 20 веков классический марксизм разделился на два больших направления:
		\begin{itemize}
			\item Коммунистическое: марксизм, который завещал Маркс, полностью реализованный большевиками.
			\item Социал-демократическое: идея мирного перехода, возможности слияния (конвергенции) капитализма и социализма. Личности, связанные с социал-демократизмом: Георгий Плеханов, Мартов. Например в Швеции <<Социал-демократическая рабочая партия Швеции>> провела важные реформы, является старейшей и самой многочисленной партией.
		\end{itemize} 
	\end{note}
	
	\subsection{Другие идеологии}
	
	\subsubsection{Радикальные идеологии}
	\begin{itemize}
		\item Крайне левые: полное отсутствие государства. Например: анархизм.
		\item Крайне правые: государство~--- все. Например: фашизм.
	\end{itemize}
	
	\subsubsection{Крайне правые идеологии}
	\paragraph{Фашизм}
	\begin{note}
		Присущ этнический национализм. Существует достаточно давно. Его считают как единым явлением, так и индивидуальным в каждой стране. Классические образцы фашизма: Итальянский фашизм и Немецкий национал-социализм. Изначально расизм был не свойствен Итальянскому фашизму, только Немецкому. Но позже проводилась антисемитская политика под влиянием Гитлера. Человек для государства. Был вызван реалиями 20 века: формирование идеологи толпы (быстрое развитие капитализма породило процесс урбанизации; в свою очередь в города перебирается сельское население, увеличивается численность городского населения; но урбанизация не сопровождалась улучшением жизни людей, наоборот, стали частыми экономические кризисы, что сильно ухудшало уровень жизни; это и привело к образованию идеологии толпы), появление новых средств массовой агитации, массовой информации и массового контроля. Был запрещен после второй мировой войны.
	\end{note}
	
	\begin{note}
		Особенности фашизма:
		\begin{enumerate}
			\item Прежде всего фашизм получает развитие в побежденных странах и странах с нерешенными национальными проблемы.
			\item Возникает во время экономических и социальных кризисов.
			\item Возникает под консервативными лозунгами.
		\end{enumerate}
	\end{note}
	
	\subsubsection{Крайне левые идеологии}
	
	\paragraph{Анархизм}
	
	\begin{definition}
		Левые~--- больше возможностей влиять на политический процесс обществу, соответственно крайне левые~--- полное отсутствие государства. Такая идеология называется анархизм.
	\end{definition}
	
	\begin{note}[Взгляды]
		Государство основано на принуждении. Надо убрать принуждение из жизни людей и строить отношения на солидарности, так как люди по природе не испорчены. Анархисты против частной собственности , так как это такое же средство угнетения. Также анархизм отвергает марксизм, потому что марксизм говорит о необходимости пролетарского государства, но с точки зрения монархизма любое государство~--- зло. Первый мыслитель-анархист~--- Пьер Прудон (18 век). Русские мыслители-анархисты: Бакунин (разработал теорию анархо-коллективизма: частной собственности нет, но работники получают заработную плату), Петр Кропоткин (анархо-коммунизм: все общее и каждый получает столько, сколько хочет). Анархист вне зависимости от видов остаются самой левой радикальной идеологии, с отрицательным отношением к государству.
	\end{note}
	
	\begin{note}[Современный анархизм]
		Отличаются большим разнообразием: зеленый анархизм (делает акцент на проблемах окружающей среды), анархо-феминизм (против всех типов угнетения), анархо-капитализм.
	\end{note}
	
	\paragraph{Экологизм}
	\begin{definition}
		Основная идея~--- экоцентризм, те природа оказывается в центре миропонимания. Уникальна, так как все другие идеологии антропоцентричны. Главная задача~--- сохранение природы. Появился в конце 19, начале 20 века из-за промышленного переворота и изменения природы человеком.
	\end{definition}
	\begin{note}
		Задачи:
		\begin{enumerate}
			\item Отказ от политики насилия, войн, ядерного оружия, атомной энергетики.
			\item Децентрализация власти, поскольку людям на местах виднее проблемы их региона и, соответственно, решения.
		\end{enumerate}
	\end{note}
	
	\begin{note}
		Современные течения:
		\begin{enumerate}
			\item Эко-социализм. Капиталистическое устройство~--- главная угроза природы.
			\item Эко-феминизм. Женщины более чутки к природе, и обеспечить защиту природы можно только через повышения роли женщин.
			\item Эко-консерваторы. С их точки зрения человек может спасти природу возвращаясь к традиционным ценностям.
		\end{enumerate}
	\end{note}
	
	\begin{note}
		Направления:
		\begin{enumerate}
			\item Умеренные: гармонию с природой можно достичь только возрождая традиционные ценности, развивать идею заботы о природе с ранних лет.
			\item Радикальные: различные нео-анархические группы.
		\end{enumerate}
	\end{note}
	
	\paragraph{Феминизм}
	
	\begin{note}
		Зародился в 18 веке, но активизировалась в 19 в процессе политической борьбы. Далее добавилось требование доступа к образованию, социального равенства, доступа к профессиям. Затем и полное равенство.
	\end{note}
	
	\begin{note}
		Ключевые идеи:
		
		\begin{itemize}
			\item \textbf{Равноправие:} Политическое, экономическое, культурное и социальное равенство женщин и мужчин.
			\item \textbf{Критика патриархата:} Отрицание системы, в которой доминирующая роль в семье и обществе принадлежит мужчинам.
			\item \textbf{Гендер как социальный конструкт:} Различение биологического пола и гендера (социальной роли). Идея о том, что ``женщиной не рождаются, а становятся'' под влиянием воспитания и общества.
			\item \textbf{Борьба со стереотипами:} Отказ от навязанных социальных ролей (``хранительница очага'', ``слабый пол'').
			\item \textbf{Экономическая независимость:} Требование равной оплаты за равный труд и преодоление «стеклянного потолка» (невидимого барьера, мешающего женщинам занимать высшие должности).
			\item \textbf{Телесная автономия:} Право женщины самостоятельно распоряжаться своим телом (репродуктивные права, защита от насилия).
			\item \textbf{Личное — это политическое:} Проблемы частной жизни (домашнее насилие, распределение бытовых обязанностей) являются общественными проблемами, требующими решения на уровне государства.
		\end{itemize}
	\end{note}
	
	\begin{note}
		Задачи:
		\begin{enumerate}
			\item Понять природу неравенства (анализ причин угнетения).
			\item Объяснить обществозначимость проблемы (вывести проблему из ``тени'').
			\item Изменить социальные институты и законодательство.
			\item Бороться за конкретные права женщин (защита от дискриминации).
		\end{enumerate}
	\end{note}
	
	\section{Политическая культура}
	
	\begin{definition}[Политическая культура]
		Совокупность политических знаний, убеждений и ценностей, которые разделяются большинством граждан и находят выражение в их политической деятельности и отношении к политической жизни.
	\end{definition}
	
	\begin{note}
		Формируется на двух уровнях:
		\begin{itemize}
			\item Целенаправленно (с участием государства, через системы образования).
			\item Стихийно (через общение людей, СМИ, интернет).
		\end{itemize}
	\end{note}
	
	\subsection{Функции}
	\begin{enumerate}
		\item \textbf{Познавательная}.
		\item \textbf{Объединяющая/разъединяющая} (интегрирующая/дис-интегрирующая). Политическая культура может как объединять людей на основе, например, либеральных взглядов, так и разъединять.
		\item \textbf{Коммуникационная}. Помогает людям коммуницировать между собой и с государством.
		\item \textbf{Ценностная}. Помогает людям выработать системы ценностей и приоритетов.
	\end{enumerate}
	
	\subsection{Типы политической культуры}
	\begin{enumerate}
		\item \textbf{Патриархальный}. Характерная для отсталых государств и для политических режимов тоталитарного типа. Носители такой культуры политикой просто не интересуются, ориентированы на семейные и хозяйственные дела.
		\item \textbf{Подданническая}. В отличии от патриархальной уже ориентирована на политическую систему. Граждане имеют предстваление о политике, деятельности властей, но активное участие не принимают.
		\item \textbf{Культура участия}. Присуща современным демократическим системам. Носители культуры не просто ориентированы и осведомленны, но и стремятся стать активными участниками. Они подчиняются власти, но сами воздействуют на нее.
		\item \textbf{Доминирующая}. Культура, нормы и ценности которой разделяются большинством граждан.
		\item \textbf{Политическая субкультура}, те культура тех или иных слоев населения.
		\item \textbf{Политическая контркультура}. Противостоит доминирующей культуре.
	\end{enumerate}
	
	\begin{note}
		В реальной жизни по отдельности не встречаются, а смешиваются (с возможным преобладанием).
	\end{note}
	
	\subsection{Составляющие политической культуры}
	\begin{enumerate}
		\item \textbf{Политические знания}.
		\item \textbf{Политические ценности}. Есть у каждого человека, они помогают людям так или иначе понять людям политику. Выражаются в простых и ясных понятиях, как правило разделяются большинство людей и не порождают споров.
		\item \textbf{Политические убеждения}. Вызывают споры и конфликты, так как это представления людей о том, как и каким образом политические ценности могут быть воплощены в жизнь.
	\end{enumerate}
	
	\subsection{Политическое поведение}
	
	\subsubsection{По степени вовлеченности граждан}
	\begin{itemize}
		\item \textbf{Политический активизм}.
		\item \textbf{Конформизм}. Приспособление людей к политическим условиям; не желая выделятся, действуют так, как все.
		\item \textbf{Индифферентизм}. Люди не интересуются политикой и заняты собственными делами.
		\item \textbf{Абсентеизм}. Сознательный отказ от участия в политической жизни.
	\end{itemize}
	
	\subsection{Этапы политической социализации}
	\begin{enumerate}
		\item \textbf{Детство и юношеские годы}. Первоначальные политические ``взгляды'', представления; не столько на уровне знаний, сколько на уровне эмоций. В первую очередь исходят от семьи.
		\item \textbf{Период обучения в старших классах и вузах}. В этот период появляются знания.
		\item \textbf{Начало активной социальной деятельности}. Участие в выборах, разбирательства в программах кандидатов.
		\item \textbf{Вся последующая жизнь человека}. Ре-социализация~--- процесс неизбежных изменений в политической культуре человека.
	\end{enumerate}
	
	\section{Гражданское общество и правовое государство}
	
	\begin{definition}[Гражданское общество]
		Часть общественной жизни, которая не контролируется государством. В ней люди независимы.
	\end{definition}
	
	Пример: Общество защиты прав потребителя~--- готово вступиться за тех, чьи права потребителя были нарушены.
	
	\begin{definition}[Гражданское общество как форма демократии]
		Включает в себя следующие пункты:
		\begin{itemize}
			\item Независимые СМИ важны для демократии.
			\item Частный бизнес гарантирует экономическую независимость.
			\item Органы местного самоуправления помогают гражданам осуществлять власть.
		\end{itemize}
	\end{definition}
	
	\noindent \textbf{Демократия невозможна без правового государства}
	
	\begin{definition}[Правовое государство]
		Государство, где нет возможности злоупотребления властью. Все равны перед законом
	\end{definition}
	
	\subsection{Принципы правового государства}
	
	\begin{enumerate}
		\item Верховенство права и закона в обществе.
		\item Незыблемость прав и свобод.
		\item Равенство.
		\item Взаимно ответственность человека и государства.
		\item Принцип разделения власти по горизонтали.
	\end{enumerate}
	
	\section{СМИ в обществе}
	
	\begin{definition}[СМИ]
		Средства тиражирования и распространения информации.
	\end{definition}
	
	\subsection{История}
	\begin{itemize}
		\item Первые СМИ относят к XVIII в.~--- первые газеты (печатные СМИ).
		\item В XX веке появились аудиовизуальные СМИ (радио и телевидение).
		\item В XXI веке появился интернет СМИ.
		\item Кинематограф и реклама~--- тоже своего рода СМИ.
	\end{itemize}
	
	\subsection{Признаки СМИ}
	\begin{enumerate}
		\item Периодичность издания.
		\item Наличие постоянного названия.
		\item Массовость и общедоступность.
	\end{enumerate}
	
	\subsection{Политические функции СМИ}
	\begin{enumerate}
		\item Информационная.
		\item Политическая социализация.
		\item Критика и контроль.
		\item Мобилизационная.
	\end{enumerate}
	
	\subsection{Положительные и негативные стороны СМИ}

	\subsection{Разделение власти}
	
	\subsubsection{Горизонтальное разделение власти}
	
	Это распределение функций между ветвями власти на одном уровне (как правило, общенациональном).
	
	Включает три классические ветви:
	\begin{itemize}
		\item \textbf{Законодательная} (Парламент): создает и принимает законы.
		\item \textbf{Исполнительная} (Правительство, Президент): реализует законы и управляет государственными делами.
		\item \textbf{Судебная} (Суды): осуществляет правосудие и разрешает споры.
	\end{itemize}
	
	\textbf{Цель}: создать систему «сдержек и противовесов», чтобы ни одна из ветвей не могла узурпировать власть.
	
	\subsubsection{Вертикальное разделение власти}
	
	Это распределение полномочий между разными иерархическими уровнями власти (сверху вниз).
	
	Основные уровни:
	\begin{itemize}
		\item \textbf{Федеральный/Центральный уровень}: решает вопросы общегосударственного значения (оборона, внешняя политика, валюта).
		\item \textbf{Региональный уровень} (субъекты федерации, штаты, области): решают вопросы своего региона.
		\item \textbf{Местный уровень} (муниципалитеты): решают локальные вопросы (благоустройство, ЖКХ).
	\end{itemize}
	
	\textbf{Цель}: децентрализовать управление, приблизить власть к народу и учитывать интересы конкретных территорий.

			\item Право на обращение в суд для защиты своих прав и свобод.

		\end{enumerate}
	\end{itemize}
	
	\section{Экономика}
	
	\begin{definition}[Экономика]
		В переводе~--- искусство ведения домашнего хозяйства.
		
		В современно языке употребляется в двух значениях:
		
		\begin{itemize}
			\item Как хозяйственная деятельность.
			\item Как наука, изучающая эту деятельность.
		\end{itemize}
	\end{definition}
	
	\subsection{Функции экономики, как науки}
	
	\begin{enumerate}
		\item \textbf{Познавательная}.
		\item \textbf{Практическая}: подтверждает полезность экономических знаний.
		\item \textbf{Прогностическая}: определяет перспективы социально-экономического развития на будущее.
		\item \textbf{Методологическая}: не только наука, но и метод достижения поставленных задач, и методологическая функция показывает, как решения будут правильными, а от каких стоит воздержаться.
	\end{enumerate}
	
	\subsection{Этапы становления экономической науки}
	
	\begin{definition}[Экономическая мысль и наука]
		Экономическая мысль~--- это размышления людей об их хозяйственной деятельности. Появилась вместе с хозяйством.
		
		Экономическая наука~--- это отслеживание закономерностей явлений, вскрытие причин, определение последствий. Появилась намного позднее мысли, когда для этого созрели условия.
	\end{definition}
	
	\begin{person}[Аристотель]
		Основатель экономической науки. Грамотное распределил хозяйство. Выделил собственность и богатство.
	\end{person}
	
	\begin{note}
		Экономика долгое время не становилась самостоятельной наукой, так как в этом небыло необходимость из-за того, что долгое время было актуально натуральное хозяйство. Самостоятельное значения стала приобретать только в новое время, когда появилось капиталистическое хозяйство (16--17 века). В это время стали появляться первые экономические школы~--- Меркантилисты (от итальянского~--- торговец) и Физиократы. Меркантилисты выступали за необходимость поддержки торговли. Физиократы считали, что богатство создается не в сфере торговли, а в сфере производства; называли природу <<матерью богатств>>.
	\end{note}
	
	\subsection{Типы экономических систем}
	
	\begin{definition}[Экономическая система]
		Способ организации хозяйственной жизни и общества.
	\end{definition}
	
	\paragraph{Основные вопросы экономики}
	
	\begin{itemize}
		\item Что производить.
		\item Как производить.
		\item Для кого производить.
	\end{itemize}
	
	\paragraph{Типы систем}
	
	\begin{enumerate}
		\item \textbf{Традиционная}. Самая древняя. Для нее характерен низкий уровень производства, аграрная экономика и сельское хозяйство (там где возможно).
		
		Что и как производить определяется обычаями и традициями. Сама система меняется очень медлено. В основном господство натурального хозяйства (производства продукта для потребления). Большая часть населения живет крайне бедно. Излишек (добавочный продукт) невелик и достается верхушке общества, остальное распределяется по традициям.
		
		Достоинства такой системы:
		\begin{enumerate}
			\item Стабильность, предсказуемость, отсутствие кризисов.
			\item Высокое качество производимой продукции.
		\end{enumerate}
		
		Недостатки же перекрывают достоинства:
		\begin{enumerate}
			\item Отсутствие технического прогресса: такая система консервативна.
			\item Ограниченность производимых продуктов: их производится очень мало и производятся они не для продажи.
		\end{enumerate}
		
		\item \textbf{Рыночная}. Основа~--- экономическая свобода, право частной собственности и конкуренция.
		
		Что производить и как продавать решает сам производитель. Также потребитель тоже решает что и по какой цене ему покупать. Существует суверенитет производителя и потребителя. 
		
		Плюсы:
		\begin{enumerate}
			\item Эффективность.
			\item Гибкая реакция на спрос.
			\item Применяет достижения научно-технического прогресса.
		\end{enumerate}
		
		Минусы:
		\begin{enumerate}
			\item Нестабильность развития: рынок развивается от кризиса к кризису.
			\item Не обеспечивает достаточную степень социальной защиты: сильные участники рынка процветают, а слабые вымываются.
		\end{enumerate}
		
		Способы государственного вмешательства:
		\begin{itemize}
			\item Лицензирование.
			\item Кредитно-финансовая политика.
			\item Создание правовой базы для рыночной экономики.
			\item Создание технических условий, которым должен отвечать товар.
		\end{itemize}
		
		\item \textbf{Смешанная}. Экономическая система, в которой сочетаются элементы рыночной и командной экономик.
		
		\item \textbf{Командная}. Государство контролирует природные и производственные ресурсы, труд людей.
		
		Что производить и по какой цене продавать тоже определяет государство. Производство в такой системе осуществляется в соответствие с палном.
		
		Преимущества:
		\begin{enumerate}
			\item Государство имеет возможность сконцентрировать все силы и средства для решения какой-либо важной государственной проблемы: огромные мобилизационные возможности.
			\item Государство гарантирует людям определенный уровень жизненных благ: такая система не знает безработицы.
		\end{enumerate}
		
		Недостатки:
		\begin{enumerate}
			\item Отсутствие стимула к выпуску качественных товаров.
			\item Никакой план не определит, что нужно людям.
			\item Отсутствие экономической свободы.
		\end{enumerate}
	\end{enumerate}
	
	\subsection{Национальные модели экономических систем}
	
	\subsubsection{Американская экономическая система}
	
	Основана на идеи индивидуальной свободы. Частная собственность охраняется государством, и частные предприятия составляют основу государственной экономики США.
	
	Расчитана на индивидуальную экономическую активность. В первую очередь средний американец расчитывает на свои активы, а не на помощь государства.
	
	\subsubsection{Скандинавская модель}
	
	Часто противопоставляют Американской. В отличии от крайне индивидуализированной Американской модели, Шведская представляет собой социально-ориентированную экономику.
	
	Особенностью является перераспределение национального дохода в пользу нуждающихся, с помощью высоких налогов (налоговые ставки доходят до 50\%).
	
	\subsubsection{Японская (Азиатская) модель}
	
	Государство участвует в деятельности в качестве координатора. Оно предоставляется планы развития по отраслям и регионам. Но эти планы носят не обязательный, а рекомендательный характер.
	
	Перешла к современному обществу и экономике не отказавшись от традиций, а приспособив их к современным условиям.
	
	\subsection{Макро-экономические показатели}
	
	\begin{definition}[Марко-экономика]
		Исследует отдельные экономические субъекты.
	\end{definition}
	
	\subsubsection{Показатели}
	
	\begin{enumerate}
		\item ВВП (валовый внутренний продукт). Вся продукция, произведенная в стране за определенный период времени (обычно за год).
		\item ВНП (валовый национальный продукт). То же, что и ВВП, но с учетом предприятий, которые находятся на территории страны, даже если они не принадлежат этой стране.
		\item ЧВП = ВВП $-$ амортизация. Амортизация~--- ценность оборудования, затрата на изнашивание и тд.
		\item ЧНП = ВНП $-$ амортизация
	\end{enumerate}
	
	\subsection{Типы экономического роста}
	
	\subsubsection{Экстенсивный экономический рост}
	
	\begin{definition}
		Количественный рост. Происходит за счет привлечения дополнительных ресурсов (сырья, капитала). Такой рост не меняет средняя производительность труда.
	\end{definition}
	
	\subsubsection{Интенсивный экономический рост}
	
	\begin{definition}
		Качественный рост. Увеличение темпов роста, за счет роста качества, внедрения новых технологий, повышения квалификации. Важным фактором такого роста явлется научно-технический прогресс.
	\end{definition}
	
	\subsection{Экономический цикл}
	
	\begin{definition}
		Период подъемов и падения развития экономики.
		
		Состоит из 4 фаз:
		\begin{enumerate}
			\item \textbf{Кризис}. Может возникать из-за: демографических волн, финансовых кризисов, колебаний мировых цен на золото. Но если говорить о экономическом кризис, от зачастую это кризис перепроизводства. Кризис~--- падение всех экономических показателей.
			\item \textbf{Депрессия} (или стагнация). Падения уже нет, падать некуда, но и роста тоже нет. Все основные экономические показатели замирают на одном уровне. На этом этапе начинается технологическое обновление производства.
			\item \textbf{Оживление}. Обнаруживает тенденцию к экономическому росту. Начинается с отраслей, производящих оборудование.
			\item \textbf{Подъем} (или экспансия). Фаза роста всех экономических показателей. Высший подъем называется <<экономический бум>> или <<процветание>>; обычно не продолжительный период; после достижения начинается снижение.
		\end{enumerate}
	\end{definition}
	
	\subsubsection{История кризисов}
	
	\paragraph{Первый технический кризис}
	
	Произошел в Англии, в 18 веке. Произошел из-за самого раннего вступления в промышленность.
	
	\paragraph{Первый международный кризис}
	
	Произошел в 1857 году. Причины связали с социальными потрясениями.
	
	\paragraph{Великая депрессия}
	
	Экономический кризис, начавшийся в Америке. Длился с 1929 по 1933 года. Полностью был преодолен только после Второй Мировой Войны.
	
	\paragraph{Кризис 70-ых годов}
	
	Произошел из-за поднятия цен на нефть в 4 раза главным экспортером нефти с ближнего востока. Началась инфляция. Развитые страны мира столкнулись с нехваткой нефти.
	
	\paragraph{Конец 80-ых годов}
	
	Мировая социалистическая система прекратила существования в начале 90-ых годов. Страны вставали на путь рыночной экономики. Получил название <<трансформационный кризис>>.
	
	\paragraph{Вывод}
	
	
	\begin{enumerate}
		\item Кризисы превратились из национальных/локальных в мировые.
		\item Сокращается период циклических колебаний.
		\item К закономерным признакам экономики добавился фактор случайности.
	\end{enumerate}
	
	\subsection{Хозяйственная деятельность}
	
	\begin{definition}
		Использование хозяйствующими субъектами экономических ресурсов с целью создания благ, удовлетворяющих потребности.
	\end{definition}
	
	\begin{definition}[Потребности]
		Главная побудительная сила, стимулирующая хозяйственную деятельность.
	\end{definition}
	
	\begin{definition}[Блага]
		Средства удовлетворения потребностей.
		
		Делятся на:
		
		\begin{itemize}
			\item Природные.
			\item Экономические (произведенные).
		\end{itemize}
		
		Также бывают:
		
		\begin{itemize}
			\item Взаимозаменяемые.
			\item Взаимодополняемые.
		\end{itemize}
		
		Свойства:
		
		\begin{enumerate}
			\item По мере удовлетворений потребностей полезность благ снижается.
			\item Относительная редкость благ.
			\item Ценность благ~--- возможность обменять одни блага на другие.
		\end{enumerate}
	\end{definition}
	
	\subsection{Экономические ресурсы и доходы от них}
	
	\subsubsection{Факторы производства}
	
	\begin{definition}[Факторы производства]
		Источники экономических благ.
	\end{definition}
	
	\paragraph{Факторы}
	
	\begin{itemize}
		\item \textbf{Земля}~--- все природные ресурсы.
		
		\item \textbf{Труд}~--- использование физического и умственного труда человека в процессе создания благ.
		
		\item \textbf{Капитал}.
		
		\begin{itemize}
			\item 
			\item Производственный капитал; например станки и другое оборудование.
		\end{itemize}
		
		\item В $20$ веке был добавлен фактор \textbf{предпринимательских способностей}.
		
		\item В конце $20$ века также был добавлен информационный фактор. \textbf{Информация}~--- это знания, данные, технологии. Отличительная особенность информационного фактора~--- это его неисчерпаемость.
	\end{itemize}
	
	Используя или продавая факторы производства участники получают доход. И каждому фактору соответствует свой тип дохода:
	
	\begin{itemize}
		\item Земля~--- \textbf{рента}.
		\item Труд~--- \textbf{заработная плата}.
		\item Капитал~--- \textbf{процент от прибыли}.
		\item Предпринимательские способности~--- \textbf{прибыль}.
		\item Информация~--- \textbf{роялти}.
	\end{itemize}
	
	\subsubsection{Рынок}
	
	Для возникновения рынка необходим ряд условий:
	
	\begin{itemize}
		\item Общественное разделение труда.
		\item Обособление производства.
		\item Развитие товарно-денежных отношений. Потребность обмена приводит к появлению денег.
	\end{itemize}
	
	\paragraph{Структура рынка}
	
	\begin{itemize}
		\item \textbf{Рынок товаров и услуг} (потребительский рынок). На этом рынке действуют все предприятия, осуществляющие продажу товаров. Основные уровни: товарные биржи, торговые компании.
		\item \textbf{Рынок факторов производства}. Сделки по покупке и продаже сырья и оборудования. Частью этого рынка является рынок земли.
		\item \textbf{Финансовый рынок}. Состоит из двух частей:
		\begin{itemize}
			\item Кредитный (денежный) рынок.
			\item Рынок ценных бумаг (фондовый рынок).
		\end{itemize}
		\item \textbf{Рынок труда}. Продается и покупается особый вид товара, который называется <<рабочая сила>>.
	\end{itemize}
	
	\paragraph{Функции рынка}
	
	\begin{enumerate}
		\item \textbf{Посредничество}. Связывает производителей товаров и потребителей.
		\item \textbf{Ценообразование}. Рынок устанавливает равновесную цену, те цену, которая устраивает всех.
		\item \textbf{Информативная}. Предоставление информации о том, что где продается и по какой цене можно купить тот или иной товар.
		\item \textbf{Оздоровительная}. На рынке побеждают только те фирмы, которые выпускают товар, пользующийся спросом.
	\end{enumerate}
	
	\paragraph{Слабости рынка}
	
	\begin{enumerate}
		\item Рынок не создает достаточное количество общественных благ.
		\item Рынок не будет разрешать социальные проблемы, заниматься трудоустройство безработных, социальным обеспечением малоимущих и инвалидов и тому подобным.
		\item Вопросы социального неравенства. Рынок устроен по принципу выживает сильнейший, поэтому сильные участники рынка будут процветать, а слабе будут только проседать. И если государство не будет учавствовать в экономической жизни, то проблемы будут накапливаться, что рано или поздно приведет к кризису. Механизмы вмешательства государства в рыночную экономику:
		\begin{itemize}
			\item Политика мониторизма.
			\item Фискальная политика. 
			\item Правовое регулирование.
		\end{itemize}
	\end{enumerate}
	
	\section{Виды рынков и их классификации}
	
	\subsection{По территориальному принципу}
	
	\begin{itemize}
		\item Местный.
		\item Региональный.
		\item Национальный.
		\item Мировой.
	\end{itemize}
	
	\subsection{По типу борьбы}
	
	\begin{itemize}
		\item \textbf{Совершенная конкуренция}. Характеризуется большим количество производителей и потребителей; не регулируемым ценообразованием; свобода вступления и выхода всех участников; полная информированность.
		\item \textbf{Несовершенная конкуренция}.
		
		Формы:
		
		\begin{itemize}
			\item Монополия.
			\item Олигополия.
			\item Монопсония.
		\end{itemize}
	\end{itemize}
	
	\subsection{Плюсы и минусы конкурентной борьбы}
	
	\subsubsection{Плюсы}
	
	\begin{itemize}
		\item Остаются эффективные фирмы.
		\item Производства хорошего и качественного товара.
		\item Хорошие технологии.
	\end{itemize}
	
	\subsubsection{Минусы}
	
	\begin{itemize}
		\item Создает барьер для новичков.
		\item Угроза разорения социально значимых отраслей экономики.
	\end{itemize}
	
	\section{Спрос и предложение}
	
	\begin{definition}[Спрос]
		Желание и возможность приобрести товар по существующей на рынке цене.
		
		Экзогенный спрос~--- спрос, вызванный внешними причинами.
		
		Эндогенный спрос~--- спрос, который возникает внутри общества.
	\end{definition}
	
	\subsection{Факторы и условия спроса}
	
	\begin{itemize}
		\item Цена.
		\item Цена и количество комплиментарных товаров.
		\item Ожидания потребителей.
		\item Вкусы и предпочтения потребителей.
		\item Рост или снижение числа потребителей.
	\end{itemize}
	
	\begin{definition} [Закон спроса и предложения]
		Цена увеличивается при увеличении спроса и наоборот.
	\end{definition}
	
	\subsection{Не ценовые факторы предложения}
	
	\begin{itemize}
		\item Выход на рынок новых фирм.
		\item Изменение стоимости производства товара.
		\item Внешние факторы: природные катастрофы, войны, экономические санкции.
	\end{itemize}
	
	\begin{definition}[Цена равновесия]
		Пересечение кривых спроса и предложения от цены. Цена, которая в данный момент устраивает и продавца, и покупателя. Если цена ниже цены равновесия, то спрос растет, а предложение падает. Если цена выше цены равновесия, то предложение растет, а спрос падает.
	\end{definition}
	
	\section{Деньги}
	
	\subsection{Теории происхождения денег}
	
	Существует два основных подхода к объяснению того, как появились деньги:
	
	\begin{itemize}
		\item \textbf{Рационалистическая (договорная)}~--- деньги возникли как результат соглашения (договора) между людьми, которые осознали, что для обмена необходим специальный инструмент. Все, что участвует в обмене, должно быть сопоставимо.
		\item \textbf{Эволюционная} (А. Смит, Д. Рикардо, К. Маркс) — деньги появились естественным образом в результате длительного процесса развития обмена, помимо воли людей. 
	\end{itemize}
	
	\textbf{Этапы развития обмена:}
	\[ \text{Бартер (Т -- Т)} \to \text{Товарные деньги (скот, меха)} \to \text{Золото/Серебро} \to \text{Монеты} \to \text{Бумажные деньги} \]
	
	\subsection{Формы и ценность денег}
	
	\begin{itemize}
		\item \textbf{Металлические деньги (полноценные)}~--- обладали внутренней стоимостью, равной их номиналу.
		\item \textbf{Бумажные деньги (фиатные)}~--- деньги, чья реальная стоимость (стоимость бумаги) минимальна. Их выпускает государство, и они держатся на доверии к нему.
		\item \textbf{Золотой стандарт}~--- система, при которой бумажные деньги можно было свободно обменять на определенное количество золота. \textit{В настоящее время отменен во всем мире.}
	\end{itemize}
	
	\begin{definition}[Инфляция]
		Ситуация, при которой денег в обращении выпускается больше, чем производится товаров и услуг. Деньги обесцениваются.
	\end{definition}
	
	\begin{definition}[Дефляция]
		Ситуация, при которой товаров производится больше, чем выпускается денег. Покупательная способность денег растет.
	\end{definition}
	
	\subsection{Функции денег}
	
	\begin{enumerate}
		\item \textbf{Мера стоимости}~--- использование денег для измерения и сравнения стоимости различных товаров и услуг. 
		
		\begin{itemize}
			\item Позволяет соизмерять ценность товаров между собой.
			\item Связана с понятием \textbf{ППС} (Паритет покупательной способности).
		\end{itemize}
		
		\item \textbf{Средство обращения}~--- деньги выступают мимолетным посредником в акте купли-продажи.
		
		\begin{itemize}
			\item Формула: $T - D - T$ (Товар~--- Деньги~--- Товар). Это удобнее прямого обмена ($T - T$).
			\item \textbf{Ликвидность}~--- способность актива быстро превращаться в деньги без потери стоимости. Деньги обладают \textit{абсолютной ликвидностью}.
		\end{itemize}
		
		\item \textbf{Средство платежа}~--- использование денег при выплатах, не предполагающих сиюминутного получения товара (налоги, зарплаты, штрафы, оплата долгов).
		
		\begin{itemize}
			\item Формула: $D - \text{отсрочка} - T$.
			\item На этой функции основан \textbf{кредит} (банковский и коммерческий).
			\item \textbf{Виды банковского кредита}: краткосрочный ($< 1$ года), среднесрочный ($1-5$ лет), долгосрочный ($> 5$ лет).
		\end{itemize}
		
		\item \textbf{Средство накопления (сбережения)}~---изъятые из обращения деньги используются для переноса покупательной способности из настоящего в будущее.
		
		\begin{itemize}
			\item Накопления могут формировать граждане, фирмы и государство.
		\end{itemize}
		
		\item \textbf{Мировые деньги}~--- использование валюты для проведения международных расчетов (сегодня это чаще всего доллар, евро, юань или золото).
	\end{enumerate}
	
	\subsection{Финансовая и налоговая система}
	
	\begin{definition}[Финансовая система]
		Складывается в процессе использования денег. Функционирует на основе распределения и перераспределения денежных средств.
		
		Финансы разделяются на первичные и вторичные:
		\begin{itemize}
			\item \textbf{Первичные}. Средства, образующиеся у предприятия в результате его хозяйственной деятельности. Делятся между собственниками факторов производства: работниками (труд), предпринимателями (предпринимательские способности), собственниками земли (земля).
			\item \textbf{Вторичные}. Образуются в результате перераспределения доходов, которое осуществляется с помощью налогов.
		\end{itemize}
	\end{definition}
	
	\begin{definition}[Налоги]
		Обязательные платежи частных лиц и организаций в бюджеты разных уровней. Совокупность собираемых государством налогов называется налоговой системой.
	\end{definition}
	
	\subsubsection{Функции налогов}
	
	\begin{enumerate}
		\item \textbf{Фискальная}. Сбор денег для наполнения государственной казны.
		\item \textbf{Распределительная}. Перераспределение средств между отраслями экономики и социальными группами.
		\item \textbf{Учетная}. Осуществление учета денежных средств государством.
		\item \textbf{Стимулирующая}. Поддержка приоритетных отраслей или ограничение нежелательной экономической деятельности.
	\end{enumerate}
	
	\subsubsection{Виды налогов}
	
	\paragraph{По уровню постановления (в РФ)}
	
	\begin{itemize}
		\item \textbf{Федеральные}. Поступают напрямую в государственную казну. Примеры: НДС, НДФЛ, налог на добычу полезных ископаемых, акцизы.
		\item \textbf{Региональные}. Поступают в бюджеты субъектов федерации. Примеры: налог на имущество организаций, транспортный налог, налог на игорный бизнес.
		\item \textbf{Местные}. Идут в бюджеты муниципальных образований. Примеры: земельный налог, налог на имущество физических лиц, торговый сбор.
	\end{itemize}
	
	\paragraph{По способу (методу) взимания}
	
	\begin{itemize}
		\item \textbf{Прямые}. Взимаются непосредственно с дохода или имущества налогоплательщика. Зависят от уровня благосостояния граждан.
		\item \textbf{Косвенные}. Включаются в цену товаров и услуг. Формально их можно «не платить», если не покупать определенный товар. Косвенные налоги проще собирать, они зависят от уровня цен.
	\end{itemize}
	
	\subsubsection{Налоговые системы}
	
	\begin{enumerate}
		\item \textbf{Прогрессивная}. Ставка налога увеличивается по мере роста дохода (богатые платят больше).
		\item \textbf{Регрессивная}. Ставка налога снижается по мере роста доходов (основная нагрузка ложится на бедных).
		\item \textbf{Пропорциональная}. Ставка налога едина и не зависит от величины дохода.
	\end{enumerate}
	\begin{note}
		В России долгое время действовала пропорциональная система ($13\%$), однако с $2021$ года начался переход к элементам прогрессии, а с $2025$ года введена полноценная прогрессивная шкала НДФЛ.
	\end{note}
	
	\subsubsection{Подробная классификация налогов}
	
	\paragraph{Прямые налоги}
	
	\begin{itemize}
		\item \textbf{Индивидуальный подоходный налог (в РФ~--- НДФЛ)}. Ставка в мире варьируется от $0$ до $60\%$. Взимается либо по месту работы, либо через подачу налоговой декларации (если источников дохода несколько).
		\item \textbf{Налог на недвижимость (имущество)}.
		\item \textbf{Налог на транспортные средства}.
	\end{itemize}
	
	\paragraph{Косвенные налоги}
	
	\begin{itemize}
		\item \textbf{НДС (Налог на добавленную стоимость)}. Введен в России в $1992$ году. Основная ставка~--- $20\%$, льготная (лекарства, детские товары)~--- $10\%$. Отсутствует в США и Канаде. 
		\begin{note}
			Добавленная стоимость~--- это разница между выручкой и стоимостью закупленных материалов. Конечным плательщиком НДС всегда является потребитель.
		\end{note}
		\item \textbf{Акцизы}. Налоги на определенные виды товаров (алкоголь, табак, топливо). Часто называются «налогами на грехи». Могут составлять до $90\%$ от цены товара.
		\item \textbf{Таможенная пошлина}. Взимается при перемещении товаров через границу (импорт/экспорт).
		\item \textbf{Гербовый сбор}. Сбор за оформление официальных документов.
	\end{itemize}
	
	\subsubsection{Принципы налогообложения}
	
	\begin{person}[Адам Смит]
		Классик экономической теории, сформулировавший основные принципы справедливых налогов.
	\end{person}
	
	\begin{enumerate}
		\item \textbf{Принцип равенства}. Все граждане равны перед законом; налоги не могут устанавливаться по неэкономическим мотивам (раса, религия).
		\item \textbf{Принцип обязательности}. Неотвратимость уплаты налогов всеми субъектами.
		\item \textbf{Принцип удобства}. Налоги должны взиматься в такое время и таким способом, которые наиболее удобны для плательщика.
		\item \textbf{Принцип стабильности}. Налоговая система должна быть устойчивой и не меняться слишком часто.
		\item \textbf{Принцип справедливости}. Налоговая нагрузка должна соответствовать возможностям плательщика.
	\end{enumerate}
	
	\subsection{Права и обязанности налогоплательщика}
	
	\subsubsection{Права налогоплательщика}
	
	\begin{itemize}
		\item Право на получение информации о налогах, сроках и изменениях в законодательстве.
		\item Право присутствовать при проведении выездных налоговых проверок.
		\item Право обжаловать решения налоговых органов в суде.
		\item Право на налоговые льготы и вычеты (например, возврат части НДФЛ при оплате обучения или лечения).
	\end{itemize}
	
	\subsubsection{Обязанности налогоплательщика}
	
	\begin{itemize}
		\item Своевременно и в полном объеме уплачивать налоги.
		\item Предоставлять налоговую декларацию (обязательно для лиц с несколькими источниками дохода).
		\item Вести учет доходов и предоставлять документацию (для юридических лиц).
	\end{itemize}
	
	\section{Государственный бюджет}
	
	\begin{definition}
		Государственный бюджет~--- финансовый план государства на определенный период. В отличие от личного бюджета, принимается как закон.
	\end{definition}
	
	\noindent Проект бюджета составляется \textbf{Правительством РФ}, а принимается в качестве закона \textbf{Государственной Думой}.
	
	\subsection{Составляющие бюджета}
	
	\paragraph{Доходы бюджета}
	\begin{itemize}
		\item Налоги на прибыль предприятий и доходы граждан.
		\item Косвенные налоги и акцизы.
		\item Пошлины, штрафы и неналоговые сборы.
		\item Доходы от использования государственной собственности.
	\end{itemize}
	
	\paragraph{Расходы бюджета}
	
	\begin{itemize}
		\item Социальная сфера (образование, медицина).
		\item Оборонная мощь государства.
		\item Содержание государственного аппарата.
		\item Поддержка экономики и науки.
	\end{itemize}
	
	\subsection{Состояние бюджета}
	
	\begin{enumerate}
		\item \textbf{Сбалансированный}. Доходы примерно равны расходам.
		\item \textbf{Профицитный}. Доходы превышают расходы (встречается редко).
		\item \textbf{Дефицитный}. Расходы превышают доходы.
	\end{enumerate}
	
	\subsection{Источники финансирования дефицита бюджета}
	
	\begin{itemize}
		\item \textbf{Внутренние}:
		\begin{itemize}
			\item Выпуск государственных ценных бумаг (облигаций) для привлечения средств граждан и организаций. Это ведет к росту внутреннего госдолга.
			\item Использование накопленных резервных фондов.
		\end{itemize}
		\item \textbf{Внешние}: кредиты у других стран или международных организаций (МВФ).
		\item \textbf{Денежная эмиссия}: выпуск новых денег центральным банком (опасен ростом инфляции).
	\end{itemize}
	
	\section{Ценные бумаги}
	
	\begin{definition}[Ценные бумаги]
		Финансовые документы, которые могут быть двух видов.
		
		\begin{enumerate}
			\item Документ, предоставляющий право владения собственностью. Такие документы назвают \textbf{акции}; закрепляют следующие права:
			\begin{enumerate}
				\item Право на получение дивидендов.
				\item Право на участие в управлении фирмой.
				\item Право на долю собственности после ликвидации фирмы.
			\end{enumerate}
			\item Документ, удостоверяющий займ. Такие документы назвают \textbf{долговые бумаги}; закрепляют следующие права:
			\begin{enumerate}
				\item Фиксированный процент и номинал.
			\end{enumerate}
			Также примерами долговых бумаг являются \textbf{чеки}, \textbf{банковские сертификаты}.
		\end{enumerate}
	\end{definition}
	
	\section{Инфляция}
	
	\begin{definition}[Инфляция]
		Устойчивый рост цен на товары и услуги, не подразумевающий рост их качества.
	\end{definition}
	
	\subsection{Виды инфляции}
	
	\begin{enumerate}
		\item \textbf{Ползучая инфляция.} Цены растут на $3\% - 5\%$ процентов в год. Достаточно комфортна и для потребителей, и для производителей.
		
		Небольшая инфляция стимулирует спрос.
		
		\item \textbf{Умеренная инфляция.} Темпы от $6\%$ до $10\%$ в год.
		
		\item \textbf{Высокая инфляция.} Темпы протекания от $10\%$ до $100\%$ в год.
		
		Такая инфляция опасна для экономики и создают нестабильность: люди не могут планировать свое будущее и расходы, инвесторы не могут планировать свою деятельность. Если ползучая и умеренная инфляция характерны для развитых стран, то высокая инфляция характерна для слаборазвитых стран.
		
		\item \textbf{Гипер инфляция.} Темпы роста от $100\%$ до $1000\%$ и более в год. При такой инфляции деньги перестают выполнять свою функцию, люди переходят на бартер.
		
		В $1992$ году, в России рост цен составил $2500\%$.
	\end{enumerate}
	
	\subsection{Причины}
	
	\subsubsection{Внутренние}
	
	\begin{enumerate}
		\item Черезмерная денежная эмиссия.
		\item Монополизм экономики.
		\item Высокие косвенные налоги.
		\item Большие социальные расходы государства.
		\item Психологические факторы (инфляционное ожидание).
	\end{enumerate}

    \subsection{Внешние}

    \begin{enumerate}
        \item Мировая инфляция (растут цена на импортные товары).
    \end{enumerate}
    
    \section{Мировая экономика}
    
    \begin{definition}[Мировая экономика]
    	Совокупность экономик отдельных стран, связанных между собой международными отношениями.
    \end{definition}
    
    \begin{itemize}
    	\item Международное разделение труда
    	\item Международное движение капитала и рабочей силы
    	\item Международные валютные отношения
    	\item Международная интеграция
    \end{itemize}
    
    \noindent
    Многие государства объединяются в экономические блоки, снимая преграды на пути движения товаров, капитала и рабочей силы~--- это называется \textbf{международной интеграцией}.
    
    \subsection{Международное разделение труда}
    
    \begin{definition}[Международное разделение труда]
    	Страны специализируются на определенном производстве.
    \end{definition}
    
    \subsubsection{Предпосылки}
    
    \begin{enumerate}
    	\item \textbf{Климат, география}. Наличие полезных ископаемых, благоприятные природные условия. Государство, располагающее природными ресурсами (нефть, газ), продает их тем странам, у которых таких ресурсов нет. Государство в благоприятной климатической зоне может специализироваться на сельскохозяйственной продукции.
    	\item \textbf{Уровень экономического и технологического развития}. Более важная предпосылка, чем первая. Развитые страны специализируются на производстве готовых изделий (особенно высокотехнологичных), развивающиеся~--- на производстве сырья и сельскохозяйственной продукции. Производить готовые изделия выгоднее~--- продавая их, государство продает высококвалифицированный труд.
    	\item \textbf{Сложившиеся традиции}. Например: Италия и Португалия~--- вина, Франция~--- парфюмерия и косметика, Германия~--- автомобили, Эквадор~--- бананы. У государств складывается репутация, позволяющая успешнее продвигать свои товары на мировом рынке.
    \end{enumerate}
    
    \begin{note}[Автаркия]
    	Замкнутая экономика, при которой государство теоретически обеспечивает себя всем необходимым самостоятельно, не участвуя в международном разделении труда. Экономически невыгодна. Пример~--- Северная Корея (опора на собственные силы), но объясняется это не экономическими, а идеологическими соображениями. С $1994$ года КНДР стала активнее участвовать в мировой торговле, так как это выгодно.
    \end{note}
    
    \subsection{Международная торговля}
    
    \begin{definition}[Международная торговля]
    	Важнейший элемент мировой экономики, товарообмен между странами. В разное время лидировали разные державы: XVI--XVII вв.~--- Голландия, XVIII--XIX вв.~--- Великобритания, затем~--- США.
    \end{definition}
    
    \begin{note}[Ранние взгляды на выгоду торговли]
    	Меркантилисты считали, что внешняя торговля создает богатство: нужно вывозить как можно больше товаров (обеспечивая приток золота), и ввозить как можно меньше. Так рассуждали примерно до XVIII века. После промышленной революции экономисты пришли к выводу, что торговля не \textit{создает} богатство, а \textit{перераспределяет} его~--- богатство создается в сфере производства. Адам Смит подверг критике идею о том, что богатство заключается в большом количестве золота.
    \end{note}
    
    \begin{definition}[Экспорт]
    	Товары/услуги, которые вывозятся из страны с целью их продажи.
    \end{definition}
    
    \begin{definition}[Импорт]
    	Товары/услуги, которые ввозятся в страну с целью продажи на внутреннем рынке.
    \end{definition}
    
    \begin{definition}[Сальдо торгового баланса]
    	Разница между стоимостью экспорта и импорта за определенный период времени (экспорт $-$ импорт).
    \end{definition}
    
    \begin{note}
    	США постоянно имеют \textbf{отрицательное} сальдо~--- импорт больше экспорта. Россия, наоборот, вывозит больше товаров, чем ввозит~--- сальдо чаще \textbf{положительное}. Оптимально сбалансированное сальдо: при постоянном притоке в страну денег из-за экспорта происходит инфляция, что нежелательно.
    \end{note}
    
    \subsection{Теории преимущества во внешней торговле}
    
    \subsubsection{Теория абсолютного преимущества}
    
    \begin{person}[Адам Смит]
    	Ценил разделение труда и считал, что для граждан любой страны выгоднее покупать заграничные товары, если они дешевле отечественных.
    \end{person}
    
    \begin{definition}[Абсолютное преимущество]
    	Одни страны могут производить определенные товары с меньшими издержками, чем другие. Если это возможно, странам следует специализироваться на торговле именно тем товаром, который они производят более эффективно.
    \end{definition}
    
    \begin{note}
    	Теоретически можно выращивать бананы за полярным кругом, но это невыгодно из-за огромных издержек (теплицы, энергия, труд). Проще и дешевле выращивать их в благоприятных условиях, например в Эквадоре~--- он имеет абсолютное преимущество в торговле бананами. Россия имеет абсолютное преимущество по нефти и газу.
    
    	Абсолютное преимущество необязательно связано с климатом~--- оно может быть обусловлено владением технологиями производства и высококвалифицированной рабочей силой.
    \end{note}
    
    \subsubsection{Теория сравнительных (относительных) преимуществ}
    
    \begin{person}[Давид Рикардо]
    	Ответил на вопрос, что делать государству, если у него нет абсолютных преимуществ ни по одному товару.
    \end{person}
    
    \begin{definition}[Сравнительное преимущество]
    	Соотношение между издержками производства данного товара и издержками производства других товаров в этой стране должно быть ниже, чем соотношение издержек этого же товара в других странах.
    \end{definition}
    
    \begin{note}[Пример]
    	Взаимовыгодная торговля Англии и Португалии: Англия поставляет сукно, Португалия~--- вино. Это выгодно обеим сторонам: сукно в Англии качественное (хорошие овцы), а в Португалии благоприятный климат для виноделия. Обе страны теоретически могут производить и вино, и сукно, но издержки будут выше~--- невыгодно. Нужно найти товар, производимый с меньшими издержками относительно других товаров, и торговать именно им.
    \end{note}
    
    \begin{note}[Недостаток обеих теорий]
    	Обе теории (абсолютного и сравнительного преимущества) учитывают только один фактор производства~--- труд, и только им определяют цену товара, не учитывая другие ценообразующие факторы. Со временем эти теории были дополнены.
    \end{note}
    
    \subsubsection{Теория жизненного цикла продукта}
    
    \begin{person}[Вернон, вторая половина XX века]
    	Спрос на большинство товаров связан с модой, вкусами и предпочтениями покупателей, рекламой, уровнем доходов. Любой товар имеет определенный жизненный цикл, и структура международной торговли этим товаром меняется по мере прохождения им стадий этого цикла.
    \end{person}
    
    \subsubsection{Модель технологического разрыва}
    
    \begin{person}[Познер]
    	Преимущество во внешней торговле связано с внедрением новых технологий в производство. Пока какая-либо корпорация или государство владеет определенными технологиями, а конкуренты~--- нет, у страны есть преимущество на мировом рынке.
    \end{person}
    
    \subsection{Государственная политика в области международной торговли}
    
    \begin{itemize}
    	\item \textbf{Протекционизм}~--- государственная политика, при которой государство ограничивает импорт и стимулирует экспорт. Идея сформулирована меркантилистами.
    	\item \textbf{Свободная торговля (free trade)}~--- либерализация внешней торговли: страна открывает внутренний рынок для иностранных товаров, капиталов, рабочей силы.
    \end{itemize}
    
    \noindent
    В реальной жизни оба подхода сочетаются в разных пропорциях в зависимости от экономической политики государства.
    
    \subsubsection{Инструменты протекционизма}
    
    \begin{itemize}
    	\item \textbf{Тарифные меры}~--- таможенная пошлина, государственный налог на товары, пересекающие границу (делает товар дороже). Может быть источником дохода для развивающихся стран; часто заявляется как поддержка отечественного производителя.
    	\item \textbf{Нетарифные ограничения}:
    	\begin{itemize}
    		\item административный запрет на импорт (эмбарго, санкции);
    		\item количественные ограничения (квотирование);
    		\item импортные стандарты (экологичность, безопасность товара).
    	\end{itemize}
    \end{itemize}
    
    \subsubsection{Виды государственного регулирования торговли}
    
    \begin{enumerate}
    	\item \textbf{Селективный протекционизм}~--- избирательный, направлен на отдельные страны и товары.
    	\item \textbf{Отраслевой протекционизм}~--- защищает отдельные отрасли хозяйства (чаще сельское хозяйство).
    	\item \textbf{Коллективный протекционизм}~--- осуществляется объединением стран против других стран.
    	\item \textbf{Скрытый протекционизм}~--- использование нетаможенных методов.
    \end{enumerate}
    
    \begin{note}
    	За последнее время роль таможенных тарифов ослабла, а роль нетаможенных барьеров (санитарные стандарты, технические нормы, экологические стандарты)~--- возросла.
    \end{note}
    
    \subsection{Экономическая интеграция}
    
    \begin{definition}[Экономическая интеграция]
    	Процесс взаимопроникновения экономик отдельных стран в единый хозяйственный комплекс. Происходит потому, что это намного выгоднее для стран-участниц.
    \end{definition}
    
    \subsubsection{Уровни объединения}
    
    \begin{enumerate}
    	\item \textbf{Микроуровень}~--- на уровне отдельных фирм, компаний (корпоративный уровень).
    	\item \textbf{Макроуровень}~--- межгосударственные соглашения.
    \end{enumerate}
    
    \subsubsection{Причины интеграции}
    
    \begin{itemize}
    	\item \textbf{Географические}. Интеграционные процессы охватывают страны, входящие в один регион~--- интеграция является одновременно и регионализацией экономик.
    	\item \textbf{Единый уровень экономического развития}. Нужно, чтобы у стран совпадали темпы роста экономики, ВВП на душу населения, уровень заработной платы. Наиболее эффективна интеграция богатых стран между собой~--- бедным и богатым странам объединяться невыгодно, так как это будет поглощение, а не развитие.
    	\item \textbf{Взаимодополняемость}. Если страны производят одни и те же товары, они скорее конкуренты, а не партнеры~--- объединяться попросту невыгодно.
    	\item \textbf{Политическая воля}. Готовность и желание политических лидеров стран к объединению.
    \end{itemize}
    
    \begin{note}[Демонстрационный эффект и эффект домино]
    	\textbf{Демонстрационный эффект}~--- объединившиеся страны добиваются экономических успехов, и не вошедшие в объединение страны стремятся к нему присоединиться.
    
    	\textbf{Эффект домино} (обратный эффект)~--- страны, не вошедшие в объединение, испытывают экономические проблемы и тоже стремятся вступить в объединение.
    \end{note}
    
    \subsubsection{Этапы интеграции}
    
    \begin{itemize}
    	\item \textbf{Вширь}~--- количественное увеличение числа стран-участниц.
    	\item \textbf{Вглубь}~--- увеличение уровня (глубины) объединения.
    \end{itemize}
    
    \begin{note}[Преференциальная торговля]
    	Этап, предшествующий интеграции: соседние страны предоставляют друг другу преференции~--- таможенные преимущества.
    \end{note}
    
    \subsubsection{Формы (ступени) интеграции}
    
    \begin{enumerate}
    	\item \textbf{Зона свободной торговли (ЗСТ)}. Отмена тарифных и нетарифных ограничений для перемещения товаров внутри определенной зоны. Страны при этом сохраняют собственную торговую политику по отношению к третьим странам, но предоставляют преимущества друг другу. Пример: НАФТА~--- североамериканское соглашение о свободной торговле (США, Канада, Мексика).
    	
    	\item \textbf{Таможенный союз (ТС)}. Наряду с функциями ЗСТ добавляется согласованная торговая политика в отношении третьих стран (формируется единая внешняя торговая граница). Пример: Союз России, Беларуси и Казахстана (с $2007$ года). Крупнейший таможенный союз~--- Европейский союз.
    	
    	\item \textbf{Общий рынок}. Наряду с функциями ТС добавляется беспрепятственное перемещение всех факторов производства внутри союза (капитал, рабочая сила). Для этого необходимы соответствующие законодательные условия~--- происходит частичное объединение законодательств. Пример: Европейская экономическая зона (EEA, $1994$)~--- предусматривала свободное движение товаров, капиталов, услуг и людей (<<четыре свободы>>), а также единую политику в области защиты окружающей среды, прав потребителей, регулирования деятельности компаний и социальной политики.
    	
    	\item \textbf{Экономический валютный союз (ЭВС)}. Добавляется единая экономическая политика, вводится единая валюта. Чтобы вступить в такой союз, государства должны соответствовать определенным критериям: сохранять низкий уровень инфляции в течение долгого времени, не допускать чрезмерного бюджетного дефицита, не проводить односторонних девальваций.
    	
    	\item \textbf{Политический союз}. Высшая ступень интеграции~--- политическое и экономическое объединение, а не только экономическое. Пример: Европейский союз ($1992$ год, $27$ государств)~--- единое гражданство, парламент, единая оборонная политика, система правосудия, центральный банк. Не является государством, но обладает большим количеством его признаков.
    \end{enumerate}
    
    \subsection{Международная валютно-финансовая система}
    
    \begin{definition}[Международная валютно-финансовая система]
    	Закрепленная международными договорами форма валютно-финансовых отношений, которая обслуживает международное движение товаров и капиталов.
    \end{definition}
    
    \begin{definition}[Международный валютный рынок (МВР)]
    	Сфера экономических отношений, где осуществляется покупка и продажа золота, иностранной валюты и краткосрочных кредитов (до года).
    \end{definition}
    
    \begin{definition}[Международный финансовый рынок]
    	Более широкое понятие, включающее в себя МВР. Состоит из валютного рынка и рынка капитала (средние~--- $5-7$ лет, и долгосрочные~--- свыше $7$ лет кредиты).
    	
    	Основные субъекты: частный бизнес, государства и международные финансовые организации (\textbf{МВФ}, \textbf{МБРР}).
    \end{definition}
    
    \subsubsection{Международные финансовые организации}
    
    \begin{note}[МВФ~--- Международный валютный фонд]
    	Создан после Второй мировой войны, в $1945$ году. Цель~--- обеспечить стабильность валютно-финансовых отношений между всеми государствами-участниками.
    	
    	Предоставляет кредиты участникам, но не просто так, а для покрытия бюджетного дефицита. Это связывается с выполнением определенных условий: государство обязуется сократить государственные расходы (в т.ч. социальные), уменьшить госсубсидии, сдерживать рост заработной платы, проводить политику, нацеленную на ликвидацию дефицита бюджета собственными силами.
    \end{note}
    
    \begin{note}[МБРР~--- Международный банк реконструкции и развития]
    	Тоже был создан после Второй мировой войны, чтобы помогать экономикам, пострадавшим от войны. Сегодня оказывает экономическую помощь развивающимся государствам, входящим в его состав, предоставляет долгосрочные кредиты (до $20$ лет) под небольшие проценты (не более $10\%$).
    	
    	Также существует \textbf{Европейский МБРР (ЕБРР)}~--- создан позднее, в $1990$-е годы, для проведения рыночных реформ в странах Восточной Европы после их ухода от социалистического режима. Сейчас специализируется на кредитовании новых технологий, экологических компаний и инноваций.
    \end{note}
    
    \begin{note}[Неформальные финансовые объединения]
    	Их целью также является решение кредитно-валютных проблем государств. Крупнейшие~--- \textbf{Парижский} и \textbf{Лондонский} клубы.
    	
    	\begin{itemize}
    		\item \textbf{Парижский клуб}~--- объединение $19$ государств, занимается отсрочками кредитов, списанием и реструктуризацией долгов. Россия участвует в заседаниях клуба и как должник, и как кредитор (с $1997$ года).
    		\item \textbf{Лондонский клуб}~--- объединение частных банков, где также координируется политика долгов.
    	\end{itemize}
    \end{note}
    
    \subsubsection{Мировая валютная система}
    
    \begin{note}
    	Для взаиморасчетов между странами используется валюта (как функция денег: посредничество в международных экономических отношениях, мировое средство обращения и платежа). В роли мировых денег в разное время использовалось золото, затем~--- наиболее устойчивая мировая валюта: фунт стерлингов (весь XIX век и до Первой мировой войны), а сейчас~--- американский доллар.
    \end{note}
    
    \begin{definition}[Национальная валюта]
    	Законное средство платежа данного государства.
    \end{definition}
    
    \begin{definition}[Иностранная валюта]
    	Средство платежа, которое может продаваться и покупаться на мировом рынке, но не являться законным в данном государстве.
    \end{definition}
    
    \begin{definition}[Резервная валюта]
    	Иностранная валюта, в которой банки государств предпочитают накапливать и хранить свои валютные резервы.
    \end{definition}
    
    \begin{note}[Структура резервных валют]
    	К резервной валюте относятся: доллар ($50\%$ от всех мировых валютных сбережений), евро ($20\%$), далее с большим отрывом идут британский фунт ($5\%$), швейцарский франк ($4-5\%$) и японская йена ($2\%$).
    \end{note}
    
    \begin{definition}[Свободно конвертируемая валюта]
    	Валюта стран, где нет ограничений по обмену валюты. Такие государства принимают на себя обязательства перед Международным валютным фондом не вводить ограничения по обмену.
    	
    	К ним относятся $18$ стран, $5$ из которых~--- обладатели резервной валюты, а остальные~--- кандидаты на ее обладание.
    \end{definition}
    
    \begin{definition}[Частично конвертируемая валюта]
    	Валюта государств, которая имеет ограниченное хождение за их пределами. Примеры: турецкая лира, российский рубль (актуально только для Белоруссии).
    \end{definition}
    
    \begin{definition}[Неконвертируемая валюта]
    	Валюта стран, где запрещен ее обмен на иностранную. Пример: северокорейская вона.
    \end{definition}
    
    \begin{definition}[Валютный курс]
    	Цена одной валюты по отношению к другой. На валютном рынке устраиваются торги, по итогам которых определяется курс той или иной валюты. Курс может падать или повышаться в зависимости от спроса и предложения.
    \end{definition}
    
    \begin{note}[Валютная интервенция]
    	Государство частично может повлиять на курс своей валюты с помощью валютной интервенции, осуществляемой ЦБ. Если государство не хочет допустить падения своей валюты на рынке, оно повышает продажи (и наоборот). Такая система называется \textbf{системой гибкого курса валютного регулирования}.
    \end{note}
    
    \section{Микроэкономика}
    
    \begin{definition}[Микроэкономика]
    	Раздел экономической науки, рассматривающий деятельность конкретных экономических агентов (потребителей, производителей, продавцов товаров и услуг). Задача микроэкономики~--- объяснить, как на уровне этих субъектов принимаются рациональные решения и что влияет на их принятие в условиях ограниченности ресурсов.
    \end{definition}
    
    \begin{note}
    	Ресурсы, которыми располагают и потребитель, и производитель, всегда ограничены: покупатель не может купить всё, что хочет (ограничен бюджетом), производитель не может производить всё, что хочет,~--- он должен производить то, что востребовано рынком, и что ему позволяют имеющиеся ресурсы. Именно в условиях ограниченности ресурсов важно действовать рационально.
    \end{note}
    
    \subsection{Рациональное экономическое поведение производителей}
    
    \begin{definition}[Производитель]
    	Тот, кто формирует предложение товаров и услуг (фирмы, частные лица, государство). Производитель соединяет факторы производства (труд, земля, капитал) с предпринимательской инициативой с целью получения прибыли.
    \end{definition}
    
    \begin{note}
    	Главная цель производителя~--- получить максимальную прибыль при минимальных затратах; в этом состоит суть рационального подхода в микроэкономике для производителя. Максимальную прибыль можно получить по-разному: например, обманывая потребителей (продавая некачественный товар) или партнёров (нечестно ведя дела). Однако в долгосрочной перспективе построить бизнес на обмане невозможно.
    \end{note}
    
    \begin{definition}[Субъекты и объекты предпринимательства]
    	\textbf{Субъекты}~--- те, кто ведёт бизнес (частные лица, фирмы, товарищества, государство).\\
    	\textbf{Объекты}~--- то, чем занимаются субъекты, сама предпринимательская деятельность (производственная, товарозакупочная, операции с ценными бумагами и т.\,д.).
    \end{definition}
    
    \begin{note}
    	Субъекты могут образовывать или не образовывать юридическое лицо~--- самостоятельного носителя прав и обязанностей.
    \end{note}
    
    \subsubsection{Юридические лица}
    
    \paragraph{Признаки юридического лица}
    \begin{itemize}
    	\item Независимость от входящих в него частных лиц (лица приходят и уходят, а фирма остаётся).
    	\item Имущественная самостоятельность (имущество на правах собственности, счёт в банке).
    	\item Наличие официального наименования, отличающего юрлицо от других.
    	\item Организационное единство (наличие устава, государственной регистрации, учредительных документов).
    	\item Самостоятельная ответственность по обязательствам (государство не отвечает по обязательствам юрлиц).
    \end{itemize}
    
    \paragraph{Виды юридических лиц по цели деятельности}
    \begin{itemize}
    	\item \textbf{Коммерческие}~--- извлечение прибыли является основной целью деятельности.
    	\item \textbf{Некоммерческие}~--- не ставят получение прибыли основной целью. Они могут получать прибыль (например, платные услуги в поликлинике), но это не основная их задача: такие организации финансируются из бюджета и создаются для других целей (государственные и бюджетные организации, волонтёрские фонды).
    \end{itemize}
    
    \begin{note}
    	Физические лица тоже могут участвовать в предпринимательской деятельности, действуя от своего имени; они занимаются такой деятельностью с момента регистрации в качестве \textbf{ИП} (индивидуального предпринимателя). К ним применяются те же правила, что и к остальным субъектам предпринимательства.
    \end{note}
    
    \subsubsection{Виды предпринимательства}
    
    \begin{itemize}
    	\item \textbf{Производственное}~--- производство товаров, услуг и информации.
    	\item \textbf{Коммерческое}~--- операции и сделки по перепродаже товаров и услуг частным лицам и организациям.
    	\item \textbf{Финансовое}~--- перепродажа, но объектами выступают не товары, а деньги, валюта, ценные бумаги.
    	\item \textbf{Страховое}~--- предприниматель получает страховой взнос, который возвращается вместе со страховой премией при наступлении страхового случая.
    	\item \textbf{Посредническое}~--- посредник сводит продавца и покупателя и получает за это комиссию (процент).
    \end{itemize}
    
    \subsubsection{Формы предпринимательства по масштабу деятельности}
    
    \begin{description}
    	\item[Малый бизнес.] От $15$ до $100$ работников (до $200$ человек в промышленности и строительстве).
    	
    	\begin{note}[Франчайзинг]
    		От франц. \textit{франшиза}~--- льгота. Система мелких предприятий, заключающих контракт на пользование брендом и маркой крупной фирмы. Франчайзер (крупная фирма) предоставляет не только бренд, но и оборудование, технологии, обучает специалистов, получая за это плату~--- \textbf{роялти} ($2$--$7\%$ с оборота).
    	\end{note}
    	
    	\begin{note}[Венчурные фирмы]
    		От англ. \textit{venture}~--- авантюра. Коммерческие предприятия, строящие бизнес на нововведениях (новых технологиях или продукции). Рискуют прогореть, но имеют шанс обогатиться в случае успеха. Создаются в инновационных отраслях, финансируются венчурными фондами. Такой бизнес возник в США после Второй мировой войны и быстро стал популярен во всём мире.
    	\end{note}
    	
    	Преимущества малого бизнеса: гибкость (может подстраиваться под конъюнктуру рынка, быстро переориентироваться), меньшие затраты, легче открыть и закрыть, важное социально-политическое значение (малый бизнес~--- основа среднего класса).
    	
    	Недостатки: как правило, более низкая производительность труда, неспособность внедрять новые технологии и покупать новое оборудование (вынужден экономить), меньше возможностей для повышения квалификации сотрудников~--- отсюда недолговечность: легко открыть, легко и закрыть.
    	
    	\item[Средний бизнес.] От $100$ до $250$ работников. Также непрочен, так как приходится конкурировать одновременно и с малым, и с крупным бизнесом, не имея преимуществ ни того, ни другого. Поэтому либо перерастает в крупный, либо уходит с рынка~--- кроме тех фирм, что нашли устойчивую нишу и своего постоянного клиента (например, товары для людей с инвалидностью).
    	
    	\item[Крупный бизнес.] До нескольких тысяч работников. Более высокая устойчивость: может применять современную технику (есть средства), выпускать дешёвую массовую продукцию.
    	
    	Недостатки: негибкость, сложность управления, необходимость больших капиталовложений, которые бывает сложно найти.
    \end{description}

	\subsection{Организационно-правовые формы предприятий}

	\begin{definition}[Частное (индивидуальное) предприятие]
		Предприятие, принадлежащее одному человеку. Занимается предпринимательской деятельностью без образования юридического лица, капитал, как правило, невелик. Владелец несет неограниченную имущественную ответственность: отвечает по долгам всем своим имуществом.
	\end{definition}

	\begin{definition}[Товарищество (партнерство)]
		Объединение двух или более лиц (граждан или организаций), действующее на основании договора, без образования юридического лица. Создается, как правило, для реализации одноразовых проектов. Участники принимают решения совместно и несут личную ответственность за ведение дел.
	\end{definition}

	\paragraph{Виды товариществ}
	\begin{enumerate}
		\item \textbf{Полное товарищество.} Все участники несут полную ответственность: при разорении они отвечают перед кредиторами не только вкладами, но и личным имуществом. Может быть изъято все имущество, выходящее за рамки жизненной необходимости.
		\item \textbf{Коммандитное товарищество (товарищество на вере).} Наряду с одним или несколькими полными участниками есть вкладчики, которые отвечают только в размере своего вклада. Пример названия: <<Иванов, сыновья и компания>> (слова <<и компания>> означают вкладчиков~--- товарищей на вере).
	\end{enumerate}

	\subsubsection{Хозяйственные общества}

	\begin{definition}[Хозяйственное общество]
		Наиболее распространенная форма предпринимательства. Учреждается одним или несколькими физическими или юридическими лицами. Основной документ~--- \textbf{устав}, в котором указаны название, юридический адрес, размер и доли уставного капитала, цели и виды деятельности.
	\end{definition}

	\paragraph{Виды хозяйственных обществ}
	\begin{enumerate}
		\item \textbf{Общество с ограниченной ответственностью (ООО).} Формируется на основе заранее определенных вкладов (паев) участников. Участники рискуют только в размере своего вклада.
		\item \textbf{Общество с дополнительной ответственностью (ОДО).} Участники отвечают по долгам не только вкладами, но и дополнительно своим имуществом, в кратном размере к стоимости вкладов.
		\item \textbf{Открытое акционерное общество (ОАО).} Может свободно продавать свои акции на рынке ценных бумаг, эмитировать их и привлекать инвестиции. Число участников не ограничено.
		\item \textbf{Закрытое акционерное общество (ЗАО).} Число участников ограничено ($50$ человек), права свободной эмиссии акций нет. Акции распространяются только среди акционеров. Образуются, как правило, из небольших предприятий (например, бывших колхозов).
	\end{enumerate}

	\begin{note}
		С $2014$ года вместо деления на ОАО и ЗАО в законе используется деление на публичные (ПАО) и непубличные акционерные общества (АО). По смыслу ПАО соответствует ОАО, а АО~--- ЗАО.
	\end{note}

	\begin{definition}[Акционерное общество]
		Общество, капитал которого разделен на равные доли~--- \textbf{акции}. Акция свидетельствует о внесении вклада в капитал, делает владельца совладельцем и дает право:
		\begin{itemize}
			\item на получение дохода из общей прибыли~--- \textbf{дивиденда};
			\item на участие в управлении предприятием: акционер голосует пакетом акций на собрании акционеров.
		\end{itemize}
		Если пакет акций составляет менее $50\%$, акционер называется \textbf{миноритарным}, если более $50\%$~--- \textbf{мажоритарным} (владельцем).
	\end{definition}

	\subsubsection{Другие формы}

	\begin{definition}[Производственный кооператив]
		Объединение физических лиц (не менее $5$) для совместной хозяйственной деятельности. Основа членства~--- личные взносы (паи) и личное участие в деятельности.
	\end{definition}

	\begin{definition}[Унитарное предприятие]
		От лат. \textit{unitas}~--- единство. Основной принцип~--- неделимость имущества: оно не является собственностью предприятия, администрация лишь управляет им. Собственник~--- государство (федеральные или региональные власти). Пример сфер: оборона, здравоохранение.
	\end{definition}

	\subsection{Доход, прибыль и издержки}

	\noindent
	Главная цель предприятия~--- получение прибыли.

	\begin{definition}[Доход]
		Вся выручка предприятия за определенный период времени.
	\end{definition}

	\noindent
	Не все средства остаются у предприятия: часть идет на зарплату сотрудникам, часть~--- на закупку ресурсов, часть~--- на налоги и так далее. Эти затраты называются \textbf{издержками производства}.

	\begin{definition}[Прибыль]
		Сумма, оставшаяся у предприятия: доход $-$ издержки.
	\end{definition}

	\begin{note}
		Доход всегда неотрицателен, а прибыль может быть отрицательной (если издержки больше дохода). Поэтому предприятие должно уметь считать свои издержки.
	\end{note}

	\paragraph{Способы классификации издержек}
	\begin{itemize}
		\item По затратам (экономические издержки): явные и неявные.
		\item По отношению к величине производства: постоянные и переменные.
	\end{itemize}

	\subsubsection{Явные и неявные издержки}

	\begin{definition}[Явные издержки]
		Затраты предприятия на оплату ресурсов, не являющихся его собственностью: закупка сырья, зарплата работникам, арендная плата. Это затраты в пользу поставщиков факторов производства.
	\end{definition}

	\begin{definition}[Неявные (альтернативные) издержки]
		Затраты на оплату собственных ресурсов предприятия. Это доход, от которого предприятие отказывается, используя ресурс сам. Примеры: использование собственного помещения вместо сдачи его в аренду, вложение собственных денег в дело вместо их размещения под процент.
	\end{definition}

	\subsubsection{Постоянные и переменные издержки}

	\noindent
	Рост производства приводит к росту издержек. Чтобы понять пределы расширения предприятия, издержки делят на постоянные и переменные.

	\begin{definition}[Постоянные издержки]
		Затраты, которые фирма несет независимо от объема выпускаемой продукции. Примеры: арендная плата за помещение, налог на имущество, выплаты по займам, затраты на оборудование.
	\end{definition}

	\begin{definition}[Переменные издержки]
		Затраты, зависящие от объема выпускаемой продукции: при расширении производства они растут, при сокращении~--- уменьшаются. Примеры: затраты на сырье, оплата труда, транспортные услуги.
	\end{definition}

	\begin{note}
		Такое деление применимо только на коротком периоде времени. В долгосрочной перспективе все издержки становятся переменными.
	\end{note}

	\section{Социология}

	\subsection{Социальные нормы и ценности}

	\begin{definition}[Социальные нормы]
		Выработанные обществом образцы поведения людей в определённых обстоятельствах. Существуют в виде привычек, традиций, нравственных правил и законов.
	\end{definition}

	\begin{enumerate}
		\item \textbf{Привычки}~--- личные или семейные образцы поведения.
		\item \textbf{Традиции}~--- образцы поведения, принятые в обществе.
		\item \textbf{Нравственные правила}~--- нормы, установленные общественным мнением и имеющие обязывающий характер (например, уступать место в транспорте беременным и пожилым).
		\item \textbf{Законы}~--- нормы, обеспеченные государственным принуждением.
	\end{enumerate}

	\begin{note}
		Социальные нормы условны: они действуют только для тех групп, для которых предназначены. Одни правила соблюдаются в семье, другие~--- в дружеской компании, третьи рассчитаны на поведение в большой группе (например, на стадионе). Закон универсален и действует для общества в целом.
	\end{note}

	\begin{definition}[Аномия]
		Упадок системы социальных норм. Возникает во время социальных кризисов, когда ломаются устоявшиеся нормы.
	\end{definition}

	\begin{person}[Эмиль Дюркгейм]
		Ввёл термин <<аномия>>.
	\end{person}

	\begin{definition}[Социальные ценности]
		Убеждения людей в значимости каких-то благ и целей, к которым они стремятся (например, свобода, значимость которой признаёт большинство людей).
	\end{definition}

	\begin{definition}[Система ценностей]
		Определение важности социальных благ и целей, которое создаёт для себя каждый человек и общество в целом.
	\end{definition}

	\begin{note}
		Социология изучает изменения системы ценностей. Они неизбежны в жизни каждого человека и общества: люди взрослеют, набираются опыта, получают образование, и то, что было важно раньше, перестаёт быть таковым. По данным социологических опросов, на первом месте в личной системе ценностей у молодёжи стоит свобода, у людей зрелого возраста~--- материальное благополучие, у пожилых~--- здоровье и социальная стабильность. Кризисы могут менять систему ценностей всего общества.
	\end{note}

	\subsection{Типы культуры}

	\paragraph{По источнику и аудитории}
	\begin{itemize}
		\item \textbf{Элитарная}~--- рассчитана на узкий круг ценителей и знатоков, на духовную и интеллектуальную элиту. В её рамках осуществляется поиск нового, она свободна от коммерции. Аналогия с фундаментальной наукой: открытие $\to$ прикладная наука. Высокое искусство находит что-то новое, после чего это приходит в массовую культуру.
		\item \textbf{Массовая}~--- продаётся и покупается; её ценность определяется продаваемостью. Это не значит, что элитарная культура хороша, а массовая плоха.
		\item \textbf{Народная}~--- создаётся народом, характерно отсутствие автора. Распространяется в устной форме (сказки, фольклор, частушки).
	\end{itemize}

	\paragraph{По отношению к господствующей системе}
	\begin{itemize}
		\item \textbf{Доминирующая культура}~--- преобладающая в обществе система.
		\item \textbf{Субкультура}~--- находится внутри доминирующей и не противостоит ей, занимает нишу рядом (молодёжные и этнические субкультуры).
		\item \textbf{Контркультура}~--- система, создающаяся, чтобы бросить вызов доминирующей культуре (в том числе политический).
	\end{itemize}

	\subsection{Социальные группы}

	\begin{definition}[Социальная группа]
		Устойчивая совокупность индивидов, объединённая общими признаками: интересами, целями, взаимодействием.
	\end{definition}

	\begin{note}[Социальная группа и социальная категория]
		Реальные социальные группы существуют объективно, независимо от социологов: люди в них взаимодействуют друг с другом и осознают свою принадлежность. \textbf{Социальные категории}~--- условные группы, которые искусственно выделяются социологами (телезрители Первого канала, некурящие, слабослышащие).
	\end{note}

	\subsubsection{Виды групп}

	\paragraph{По размеру}
	\begin{itemize}
		\item \textbf{Малые} (от $2$ человек до нескольких десятков). Примеры: семья, друзья, учебный класс. Характерно личное общение и психологическая общность: люди не просто знакомы, а имеют контакты, взаимные ожидания, общие эмоции. Поддерживать такие отношения с большим числом людей невозможно, поэтому численность малой группы не превышает $40$ человек, оптимальный размер~--- около $10$.
		\item \textbf{Большие} (макрогруппы): школа, работники предприятия, население города, социальные классы, нации. Личное взаимодействие редкое, связи опосредованные.
	\end{itemize}

	\paragraph{По типу связи}
	\begin{itemize}
		\item \textbf{Первичные} (семья, группа сверстников и друзей). Каждый член~--- индивидуальность, которую никто не может заменить. Важнейший фактор социализации. Неустойчивы: целостность зависит от каждого члена (ссора, развод~--- и группа распадается).
		\item \textbf{Вторичные}. Связи имеют функциональный характер, группа устойчивее, участники легко заменимы (ушёл один учитель~--- пришёл другой). Важна система социальных ролей, а не индивидуальность.
	\end{itemize}

	\paragraph{По организации}
	\begin{itemize}
		\item \textbf{Формальные}~--- имеют статус, закреплённый в документах (семья, зарегистрированная в ЗАГСе, школьный класс, производственный коллектив).
		\item \textbf{Неформальные}~--- складываются стихийно и не закреплены документами (компания друзей, влюблённая пара).
	\end{itemize}

	\paragraph{По социальному положению}
	\begin{itemize}
		\item Демографические (половозрастные).
		\item Этнические (по национальности).
		\item Территориальные (живущие на одной территории, с похожим образом жизни: москвичи, горожане, селяне).
		\item Профессиональные.
	\end{itemize}

	\begin{definition}[Референтная группа]
		Группа-эталон: её члены относят себя к ней или хотят к ней относиться. Человек соотносит с ней своё поведение.
	\end{definition}

	\begin{note}
		Референтной может быть группа, в которой человек состоит (гордится своей национальностью и старается вести себя соответственно), и группа, в которую он только хочет попасть (ребёнок подражает взрослым). Если группа не значима для человека, она не референтна (пацифист в армии не ассоциирует себя с ней). Большинство референтных групп~--- плод воображения: человек идеализирует их, видя только лучшие качества. Чем выше социальная зрелость человека, тем выше его требования к референтной группе (для молодого учёного это научное сообщество, для человека без образования~--- криминальная среда и <<понятия>>).
	\end{note}

	\subsection{Этнические общности}

	\begin{definition}[Этническая общность (этнос)]
		Устойчивая группа людей, исторически сложившаяся на определённой территории и способная к самовоспроизводству (физическому, социальному, культурному, экономическому). Люди в ней осознают отличие от других этносов и своё единство, дорожат им, то есть обладают этническим самосознанием.
	\end{definition}

	\paragraph{Исторические типы этнических общностей}
	\begin{enumerate}
		\item \textbf{Род}~--- исторически первая общность: люди, связанные кровным родством и общим предком. Возник из первобытного человеческого стада. Для стада были характерны промискуитет и родовой брак, то есть неупорядоченные связи, которые вели к наследственным болезням и высокой смертности. Выжить при такой системе было невозможно, поэтому запрещается инцест (часто с помощью религии), и происходит переход к \textbf{экзогамии}~--- запрету браков между близкими родственниками. В истории рода два периода:
		\begin{itemize}
			\item \textbf{Матриархат}: родство ведётся по материнской линии; объединяет не только общий предок, но и общее хозяйство. Экономика присваивающая, главную роль играет собирательство, главная роль в нём и в организации принадлежит женщине.
			\item \textbf{Патриархат}: в хозяйстве и родовых отношениях преобладают мужчины. Возникает при разделении труда, с появлением земледелия, скотоводства и охоты, требовавших значительных физических усилий. Складывается семья как экономическая ячейка общества, что ведёт к разложению родового строя.
		\end{itemize}
		\item \textbf{Племя}~--- общность, возникающая при объединении нескольких родов. Кровнородственный принцип сменяется территориальным, община из родовой становится соседской. Разделение труда приводит к появлению излишков, социальному неравенству и делению общества на управляющих и управляемых. Появляются племенное самоуправление, вожди, жрецы, военные вожди, язык, общая религия и мифология.
		\item \textbf{Народность}~--- появляется в период складывания государства; первые этнические общности государственной эпохи, характерны для рабовладельческих обществ. В Европе процесс развернулся позднее, во время и после эпохи Великого переселения народов.
		\item \textbf{Нация}~--- понятие скорее политическое, чем этническое. Государственное образование, в котором люди, даже говорящие на разных языках и исповедующие разные религии, ощущают политическое и культурное единство. Появляется только в Новое время, когда на смену подданству (господству одних над другими) приходит гражданство~--- равенство перед единым для всех законом. Нацию обычно рассматривают как согражданство, принадлежность к стране, а не к этносу (русский или бурят~--- всё равно россиянин).
	\end{enumerate}

	\begin{definition}[Гражданская нация]
		Этнополитическое сообщество, которое сознаёт свою уникальность и идентичность, дорожит принадлежностью к государству и лежит в основе государственности.
	\end{definition}

	\subsection{Социальная структура общества}

	\subsubsection{Социальная дифференциация и стратификация}

	Эти понятия описывают социальное устройство общества с разных позиций.

	\begin{definition}[Социальная дифференциация]
		Горизонтальное деление общества на группы. Число групп постоянно растёт, главная движущая сила~--- разделение труда: деятельность людей специализируется всё уже, и с новыми профессиями появляются новые различия.
	\end{definition}

	\begin{definition}[Социальная стратификация]
		Вертикальное деление общества на слои, связанное с социальным неравенством. Неравенство~--- это сам факт существования богатых и бедных, а стратификация~--- система, которая делает его видимым: показывает, на какие слои делится общество и как они различаются (богатство, доступ к власти, престиж, образование).
	\end{definition}

	\subsubsection{Исторические формы стратификации}

	\begin{enumerate}
		\item \textbf{Рабство}~--- одни люди захватывали других и принуждали к бесплатному труду; раб~--- <<одушевлённая собственность>>.
		\item \textbf{Касты}~--- строго замкнутые слои некоторых традиционных обществ. Переход между кастами и внутри них невозможен, никакие успехи не меняют положение человека. Возникновение объясняется религией: в индуизме верят в перерождение по закону воздаяния. Основные касты (варны) в традиционном индийском обществе:
		\begin{itemize}
			\item \textbf{брахманы} (жрецы)~--- высшая каста, белая одежда; выше правителей;
			\item \textbf{кшатрии} (воины)~--- красный цвет;
			\item \textbf{вайшьи}~--- торговля, земледелие, скотоводство;
			\item \textbf{шудры}~--- слуги, рабочие и рабы.
		\end{itemize}
		За пределами общественных отношений находятся \textbf{неприкасаемые}: на них распространялись самые строгие ограничения (к ним нельзя было приближаться и даже наступать на их тень). После Второй мировой войны кастовые ограничения смягчились, но традиционные взгляды в Индии всё ещё сильны.
		\item \textbf{Сословия}~--- относительно замкнутые слои феодального общества. Движение по сословной лестнице затруднено, но возможно (благодаря браку или личным заслугам). Права и обязанности закреплены обычаем и передаются по наследству. Характерна иерархия: высшие сословия (духовенство и дворянство) и низшие.
		\item \textbf{Классы и страты.}
	\end{enumerate}

	\begin{note}
		С точки зрения социологии касты и сословия формальны: принадлежность к ним чётко определена по рождению и закону. Классы и страты неформальны: их искусственно выделяют учёные для анализа социально-экономических процессов.
	\end{note}

	\subsubsection{Классовая теория}

	\begin{definition}[Классы]
		Большие социальные группы, различающиеся отношением к собственности и средствам производства.
	\end{definition}

	\begin{person}[Авторы теории классов]
		Гизо, Тьерри, Адам Смит, Давид Рикардо; позднее теорию доработал Карл Маркс. По Марксу, отношение к собственности (владение) определяет положение человека в обществе.
	\end{person}

	Поскольку частная собственность возникла очень давно, первые классы~--- рабы и рабовладельцы. Далее:
	\begin{itemize}
		\item в феодальном обществе~--- феодалы и зависимые от них крестьяне;
		\item в капиталистическом~--- буржуазия и пролетариат (антагонистическая пара);
		\item в социалистическом~--- рабочие и крестьяне (не антагонисты).
	\end{itemize}

	\begin{note}[Прослойка]
		Между классами существует прослойка. Она не является классом, так как у неё нет средств производства; она может обслуживать идеологические потребности классов (интеллигенция).
	\end{note}

	\begin{note}[Критика классовой теории]
		Теория подвергается критике и теряет актуальность: она преувеличивает значение экономики и преуменьшает значение культуры и политики. На рубеже веков появилась теория социальной стратификации.
	\end{note}

	\subsubsection{Теория социальной стратификации}

	\begin{person}[Макс Вебер]
		Добавил к экономическому критерию власть, престиж, уровень образования и доход: именно они определяют положение человека в обществе.
	\end{person}

	Выделяют три основных класса:
	\begin{enumerate}
		\item \textbf{Высший}: верхний высший (выходцы из старых богатых семей) и нижний высший (недавно разбогатевшие).
		\item \textbf{Средний}: верхний средний (люди с престижными профессиями~--- адвокаты, врачи) и нижний средний (люди с рабочей квалификацией и стабильной зарплатой).
		\item \textbf{Низший}: нищие, малоимущие.
	\end{enumerate}

	\begin{definition}[Андеркласс]
		Опустившиеся люди, утратившие все социальные связи (бездомные).
	\end{definition}

	\begin{definition}[Маргинальные слои]
		От лат. \textit{marginalis}~--- находящийся на краю. Группы людей, утративших социальный статус и оказавшихся неспособными ориентироваться в новой социальной ситуации; находятся между двумя стратами.

		Пример: сокращённый научный сотрудник вынужден торговать на рынке и чувствует себя некомфортно в новой среде.
	\end{definition}

	\begin{note}
		Маргинальность~--- скорее психологическое понятие. Она бывает индивидуальной и коллективной (беженцы во время войн и кризисов).
	\end{note}

	\subsubsection{Средний класс}

	Средний класс обеспечивает социальную стабильность: чем он больше, тем стабильнее общество.

	\paragraph{Признаки среднего класса}
	\begin{itemize}
		\item Средний для данного общества уровень благосостояния.
		\item Стабильные доходы.
		\item Высокий уровень образования или профессиональная квалификация.
		\item Высокий уровень вертикальной мобильности (желание и возможность делать карьеру).
	\end{itemize}

	\paragraph{Как средний класс обеспечивает стабильность}
	\begin{enumerate}
		\item При стабильном положении он сам заинтересован в политической стабильности и сохранении правопорядка, поэтому риск социальных конфликтов снижается.
		\item Он~--- основной потребитель и поддерживает спрос.
		\item Он делает возможными технический прогресс и квалифицированный труд.
		\item Он обеспечивает равновесие в обществе, снижая противостояние между богатыми и бедными.
	\end{enumerate}

	\subsubsection{Социальное неравенство и бедность}

	Социальное неравенство можно представить в виде шкалы: на одном полюсе богатые, на другом~--- бедные с минимальным доступом к экономическим благам.

	\begin{definition}[Абсолютная бедность]
		Состояние, при котором человек не может удовлетворить базовые потребности (еда, жильё) или обеспечивает лишь своё биологическое существование.
	\end{definition}

	\begin{definition}[Относительная бедность]
		Невозможность поддерживать приличные жизненные стандарты (обед в ресторане, отпуск, автомобиль).
	\end{definition}

	\begin{note}
		Бедность~--- также психологическое состояние человека.
	\end{note}

	\begin{definition}[Порог бедности]
		Доход на душу населения, обеспечивающий социальное выживание человека (транспорт, коммунальные услуги, одежда). В России называется \textbf{прожиточным минимумом}.
	\end{definition}

	\begin{definition}[Масштаб бедности]
		Доля населения, живущего ниже порога бедности, в общем населении страны (в процентах).
	\end{definition}

	\begin{definition}[Децильный коэффициент]
		Разрыв доходов между $10\%$ самых богатых и $10\%$ самых бедных: показывает, во сколько раз самые богатые богаче самых бедных.
	\end{definition}

	\begin{itemize}
		\item Оптимальным считается значение около $4{,}7$ (как в развитых странах), его стараются поддерживать.
		\item Значение $8$ считается порогом, за которым возникает почва для потенциальных беспорядков.
		\item По лекции: Скандинавия~--- $3{,}4$, Германия~--- $5{,}7$, США~--- $15$ (исключение: там богатым не принято завидовать, богатые~--- экономический стимул), Россия~--- $25{,}3$.
	\end{itemize}

	\begin{person}[Питирим Сорокин]
		Для измерения социального неравенства ввёл шкалу и профиль социальной стратификации. \textbf{Шкала} показывает, на какие слои делится общество, \textbf{профиль}~--- удельный вес каждого слоя.

		Стабильность общества связана с профилем стратификации. Вытягивание профиля (много богатых и много бедных) создаёт угрозу социальных взрывов. Уплотнение профиля (уравнивание доходов, нет ни богатых, ни бедных) тоже нежелательно: общество лишается стимула к развитию. Поэтому так важен средний класс.
	\end{person}

	\subsection{Социализация}

	\begin{definition}[Социализация]
		Развитие человека в процессе жизни в обществе; усвоение им норм, принятых в обществе. Благодаря социализации человек учится вести себя в разных ситуациях, быть частью общества и становится личностью.
	\end{definition}

	\begin{note}
		Личность формируется только в человеческом обществе. Дети-<<маугли>> не могут стать полноценными личностями: это феральные люди, утратившие способность приобретать социальные навыки (социализация началась слишком поздно). Около $70\%$ личности формируется в детстве.
	\end{note}

	\subsubsection{Первичная социализация}

	Усвоение норм и ценностей ребёнком; начинается с рождения и длится до подросткового возраста ($11$--$13$ лет). Этапы:
	\begin{enumerate}
		\item \textbf{Адаптация}: обучение жестам, обращению с предметами и нормам морали, формирование представлений о мире и социальных ролях. Главный механизм~--- подражание ребёнка взрослым.
		\item \textbf{Обособление и идентификация}: преимущественно школьная социализация (воспитание и обучение одновременно). Человек осознаёт себя личностью.
	\end{enumerate}

	\begin{definition}[Агенты и институты социализации]
		\textbf{Агенты}~--- люди, отвечающие за воспитание. \textbf{Институты}~--- учреждения, влияющие на процесс социализации.
	\end{definition}

	\begin{itemize}
		\item Агенты первичной социализации: родители, семья, друзья, учителя (личные отношения).
		\item Институты первичной социализации: семья (и агент, и институт), школа.
	\end{itemize}

	\subsubsection{Вторичная социализация}

	Начинается, когда человек выходит за рамки семьи и школы (начало работы, служба в армии, поступление, женитьба). Идёт всю жизнь, так как человек постоянно сталкивается с новыми ситуациями. Она характеризуется:
	\begin{itemize}
		\item адаптацией к новым условиям: человек осваивает новые правила поведения и социальные роли (муж или жена, руководитель);
		\item развитием самостоятельного критического мышления: в детстве нормы воспринимаются как единственно возможные, а взрослый сам выбирает, какие нормы и ценности ему подходят.
	\end{itemize}

	Вторичная социализация сложнее, так как человек учится вести себя ещё и в зависимости от ситуации.
	\begin{itemize}
		\item Агенты: люди, связанные формально-деловыми отношениями (представители администрации, вуза, работодателя, государства; медиаперсоны, блогеры, ведущие).
		\item Институты: государство и его органы, система образования, СМИ.
	\end{itemize}

	\subsubsection{Теория Эриксона}

	\begin{person}[Эрик Эриксон]
		Американский психолог, автор теории кризисной социализации. В течение жизни человек переживает $8$ кризисов, каждый из которых служит основанием для нового этапа.
	\end{person}

	\begin{enumerate}
		\item \textbf{Первый год жизни}~--- удовлетворение основных потребностей (забота, пища, внимание, ласка). Формируется чувство безопасности и доверие к миру и людям.
		\item \textbf{Ранний возраст} ($1{,}5$--$4$ года)~--- формирование автономии и независимости: ребёнок начинает ходить и говорить, ощущает самостоятельность, но не умеет контролировать действия и эмоции, его приучают к гигиене. Агенты: семья, детский сад, сверстники.
		\item \textbf{Игровой возраст} ($4$--$6$ лет)~--- ребёнок убеждён, что он личность, предъявляет права, требует внимания, познаёт мир через игру (развиваются инициатива, социальные роли). Если родители подавляют инициативу, это плохо сказывается на развитии.
		\item \textbf{Школьный возраст} ($6$--$11$ лет)~--- формируется отношение к труду и навыки школьной дисциплины. Если ребёнок не справляется, он чувствует себя неудачником.
		\item \textbf{Отрочество} ($11$--$20$ лет)~--- поиск образцов для подражания и собственной идентичности, места в жизни; отсюда подростковые бунты.
		\item \textbf{Начало зрелости} ($20$--$25$ лет)~--- поиск близости (духовной и физической). Если человек её не находит, он замыкается в себе, и ему трудно строить отношения в семье.
		\item \textbf{Средний возраст} ($30$--$50$ лет)~--- наиболее плодотворный период, важно максимально реализоваться в обществе. Если это удаётся, человек гармоничен и удовлетворён; если нет, начинаются метания и необъяснимые поступки (кризис среднего возраста).
		\item \textbf{Старость}~--- если человек испытывает удовлетворение от прожитой жизни, он спокойно принимает старость.
	\end{enumerate}

	\subsubsection{Факторы социализации}

	\begin{itemize}
		\item \textbf{Биологическая наследственность} (разные таланты, готовность развиваться и меняться)~--- не определяющий фактор.
		\item \textbf{Внешние факторы}:
		\begin{itemize}
			\item \textit{мегафакторы}~--- планета, мир, глобальные процессы, влияющие на всё человечество;
			\item \textit{макрофакторы}~--- уровень государства и общества, мораль, ценности, традиции;
			\item \textit{мезофакторы}~--- условия социализации, различающиеся по типу местности, где живут люди;
			\item \textit{микрофакторы}~--- агенты и институты социализации, индивидуальная история.
		\end{itemize}
	\end{itemize}

	\begin{definition}[Ресоциализация]
		Усвоение новых норм и ценностей взамен старых.
	\end{definition}

	\begin{definition}[Десоциализация]
		Утрата навыков социального поведения. Может происходить из-за проблем с психикой, алкоголя, длительного тюремного заключения.
	\end{definition}

	\begin{note}
		Отклонения в социализации называются девиацией.
	\end{note}

	\subsection{Социальная мобильность}

	\begin{definition}[Социальная мобильность]
		Перемещение индивидов и социальных групп в социальной структуре общества, ведущее к изменению их социального статуса.
	\end{definition}

	\begin{person}[Питирим Сорокин]
		Автор теории социальной мобильности (книга <<Социальная мобильность>>, $1927$).
	\end{person}

	\subsubsection{Типы мобильности}

	\begin{itemize}
		\item \textbf{Горизонтальная}~--- перемещение, не меняющее социального положения (переход в другую возрастную группу, смена работы на равнозначную должность).
		\item \textbf{Вертикальная}~--- перемещение между слоями, качественно меняющее положение (кассир стал директором). Возможна как вверх, так и вниз.
		\item \textbf{Индивидуальная}~--- перемещение одного человека; \textbf{групповая}~--- перемещение группы (шахта закрылась, шахтёры осваивают новые профессии).
		\item \textbf{Межпоколенная}~--- изменение статуса детей по отношению к родителям; \textbf{внутрипоколенная}~--- карьера в течение жизни одного человека.
	\end{itemize}

	\begin{definition}[Открытое и закрытое общество]
		Общество с высокой мобильностью называется \textbf{открытым}, с низкой~--- \textbf{закрытым}. В самых закрытых (кастовых) мобильность невозможна, в сословных она есть, но затруднена.
	\end{definition}

	\subsubsection{Каналы (лифты) социальной мобильности}

	Способы и пути, позволяющие людям перемещаться между слоями. По Сорокину: семья, армия, церковь, образование.

	\begin{itemize}
		\item Семья задаёт изначальный статус: легче сделать карьеру из влиятельной семьи, чем из нищей провинциальной.
		\item Образование: с хорошим образованием легче сделать карьеру.
		\item Армия, партии, собственность и богатство.
		\item В традиционном обществе: семья, армия, церковь (священник мог дослужиться до митрополита).
	\end{itemize}

	\begin{note}
		В современном обществе важнейшим каналом мобильности является образование; также важны собственность и семья.
	\end{note}

	\subsubsection{Количественные показатели мобильности}

	Введены для сравнительного изучения мобильности по регионам, отраслям и странам.

	\begin{itemize}
		\item \textbf{Объём}~--- количество людей, переместившихся в вертикальном или горизонтальном направлении.
		\item \textbf{Дистанция}~--- условная величина, на которую индивиду удалось подняться или опуститься. \textit{Нормальная дистанция} (шаг)~--- на одну--две ступеньки (из капитана в майоры, из кассира в старшие кассиры). \textit{Ненормальная}~--- неожиданный взлёт на вершину или такое же резкое падение.
	\end{itemize}

	\subsection{Социальные статусы и роли}

	\begin{person}[Роберт Мертон]
		Внёс наибольший вклад в ролевую теорию личности.
	\end{person}

	\begin{definition}[Социальный статус]
		Позиция, которую человек занимает в системе социальных связей; определяется полом, возрастом, профессией и т.\,д.
	\end{definition}

	\begin{definition}[Статусный набор]
		Совокупность статусов одного человека (например, мужчина, муж, отец).
	\end{definition}

	\subsubsection{Виды статусов}

	\paragraph{По способу получения}
	\begin{itemize}
		\item \textbf{Предписанные}~--- получены с рождения (национальность, пол, <<чей-то сын>>); заслуг человека в этом нет.
		\item \textbf{Приобретённые (достигнутые)}~--- предполагают усилия человека (профессор).
		\item \textbf{Смешанные}~--- сочетают оба признака. Пример: статус гражданина (им можно родиться или стать).
	\end{itemize}

	\paragraph{По значимости}
	\begin{itemize}
		\item \textbf{Основные}~--- определяют положение человека в обществе: профессия, род занятий, возраст (для детей). Ребёнка в первую очередь воспринимают как ребёнка, даже если он известный актёр; взрослого~--- по профессии. Основные статусы определяют круг общения и манеру поведения. Директора банка воспринимают прежде всего как директора, хотя он ещё муж, сын, болельщик.
		\item \textbf{Неосновные}~--- влияют на поведение в конкретной ситуации (например, статус покупателя).
	\end{itemize}

	\paragraph{По масштабу группы}
	\begin{itemize}
		\item \textbf{Личные}~--- положение в малой группе (семья, компания).
		\item \textbf{Социально-групповые}~--- положение в большой группе (профессия, национальность).
	\end{itemize}

	\begin{note}[Престиж]
		Существует иерархия статусов (престиж): одни ценятся выше других (академик~--- выше выпускника ПТУ). Эта иерархия не зафиксирована в документах, но существует для каждого человека: для одних статусы связаны с работой, для других~--- с семейной жизнью.
	\end{note}

	\subsubsection{Социальные роли}

	\begin{definition}[Социальная роль]
		Действия людей, соответствующие их социальным статусам; ожидаемое поведение, предписанное статусом. Статус фиксирует позицию человека в обществе, а роль создаёт динамичную картину взаимодействия людей. От врача ждут врачебной помощи, от мастера сервисного центра~--- ремонта прибора.
	\end{definition}

	\begin{note}[Ролевые ожидания и ролевое поведение]
		Ожидаемое поведение идеально, оно исходит из теоретической ситуации. Его нужно отличать от реального ролевого поведения, которое может сильно отличаться от идеального:
		\begin{itemize}
			\item по условиям реализации (плохое настроение, личные причины);
			\item по способностям (не все могут быть хорошими врачами);
			\item по пониманию социальной роли (у разных людей оно различается);
			\item по низкой социальной ответственности.
		\end{itemize}
		Если человек не оправдывает ожиданий, к нему применяют санкции: плохого работника могут уволить, у плохих родителей забирают родительские права. Социальные роли осваиваются в процессе социализации (ролевое обучение~--- приобретение соответствующих качеств и навыков).
	\end{note}

	\begin{definition}[Ролевой набор]
		По Мертону, совокупность ролей, соответствующих одному статусу. Статусу супруга соответствуют роли мужа, добытчика, защитника; статусу директора~--- организатора, психолога, администратора.
	\end{definition}

	\paragraph{Ролевые конфликты}
	\begin{itemize}
		\item \textbf{В пределах одной роли}: роль включает несовместимые требования (муж не зарабатывает денег, но хороший отец).
		\item \textbf{Между несколькими ролями}: мать-одиночка и работница. На работе ждут, что она будет всё время уделять работе, в семье ждут внимания.
	\end{itemize}

	Такой выбор может приводить личность к внутренним конфликтам и отказу выполнять роль.

	\paragraph{Способы снижения ролевой напряжённости}
	\begin{itemize}
		\item \textbf{Рационализация ролей}: человек ищет негативные стороны нежелательных для себя ролей (домохозяйка, выбравшая семью, начинает вспоминать оставленную работу как скучную и бесполезную).
		\item \textbf{Разделение ролей}: временное изъятие из сознания нежелательной роли (женщина не защитила диссертацию, поэтому пока не думает о материнстве).
	\end{itemize}

	\subsection{Социальное поведение и социальный контроль}

	\paragraph{Формы социального поведения}
	\begin{itemize}
		\item \textbf{Естественное}~--- направлено на достижение конкретных целей, эгоистично, обеспечивает потребности без заботы об окружающих (ребёнок кричит, когда голоден). Естественное поведение взрослых~--- Эрос и Танатос (любовь и агрессия, по Фрейду). Такая жизнь опасна для общества, поэтому оно естественное поведение ограничивает, допуская только договорённости.
		\item \textbf{Ритуальное}~--- невыгодно для индивида, но именно благодаря ему общество существует и воспроизводится: помощь маленьким, слабым, больным, передача знаний следующему поколению.
	\end{itemize}

	Общество заинтересовано в ритуальном поведении, но не может отменить естественное, так как оно выгодно. Его задача~--- преобразовать естественное поведение в ритуальное с помощью поощрений и наказаний, то есть социального контроля.

	\begin{definition}[Социальный контроль]
		Контроль поведения людей с помощью социальных норм и санкций.
	\end{definition}

	\paragraph{Виды социального контроля}
	\begin{itemize}
		\item \textbf{Внутренний (самоконтроль)}: человек сам регулирует поведение, согласуя его с общепринятыми нормами. \textbf{Интериоризация}~--- глубокое усвоение норм, поэтому при их нарушении человек чувствует вину.
		\item \textbf{Внешний}: совокупность агентов и институтов, гарантирующих соблюдение норм. В традиционных обществах~--- традиции, табу, неписаные правила; в современном~--- законы, инструкции и правила. Поддерживается институтами образования, армии, полиции, налоговыми службами.
	\end{itemize}

	\paragraph{Методы социального контроля}
	\begin{enumerate}
		\item \textbf{Формальный}: осуществляют официальные организации и должностные лица (полицейские, судьи). Санкции: позитивные~--- медали, ордена, дипломы, премии; негативные~--- штрафы и т.\,п.
		\item \textbf{Неформальный}: осуществляют члены малых групп (родители, друзья). Позитивные санкции~--- похвала, негативные~--- осуждение.
	\end{enumerate}

	\paragraph{Функции социального контроля}
	Регулирующая, охранительная, контролирующая.

	\begin{note}
		Чем лучше развит внутренний контроль, тем меньше приходится прибегать к внешнему.
	\end{note}

	\subsection{Девиантное поведение}

	\begin{definition}[Девиантное поведение]
		От лат. \textit{deviatio}~--- отклонение. Поведение, отклоняющееся от общепринятых норм морали.
	\end{definition}

	\begin{definition}[Делинквентное поведение]
		Поведение, не соответствующее нормам закона, то есть преступное.
	\end{definition}

	\begin{note}
		Девиантное поведение условно: в одних условиях оно считается отклоняющимся, в других допустимым (однополые браки в России не допускаются, а на Западе разрешены). Делинквентное поведение абсолютно: воровать и убивать недопустимо ни в одном государстве.
	\end{note}

	\paragraph{Виды девиации}
	\begin{itemize}
		\item Чаще всего девиация направлена в негативную сторону, и для общества, и для самой личности.
		\item \textbf{Антисоциальная}~--- направлена вовне (хулиганство, домашнее насилие).
		\item \textbf{Аддиктивная}~--- направлена на саморазрушение (алкоголизм, игромания, наркомания).
		\item Иногда девиация имеет положительную направленность: избыточное трудолюбие (трудоголизм, возможный в ущерб семье), религиозное подвижничество, героизм.
	\end{itemize}

	Итак, девиация~--- состояние конфликта личности с общественными интересами и моралью. Рассмотрим причины.

	\subsubsection{Биологические причины}

	Самое раннее объяснение (более $100$ лет назад).

	\begin{person}[Чезаре Ломброзо]
		Врач, установивший связь между преступным поведением и физическими чертами: преступный тип можно опознать по строению тела (низкий лоб, сильно развитые надбровные дуги). Таким людям свойственна недостаточная чувствительность и отсутствие сочувствия, что сближает преступников с <<дикарями>>. Преступный тип~--- результат деградации в процессе эволюции.
	\end{person}

	\begin{person}[Уильям Шелдон]
		Считал, что на преступное поведение влияет строение тела, и различал три типа:
		\begin{itemize}
			\item \textbf{эндоморфы}~--- умеренная полнота, округлое тело, общительны, легко ладят с людьми;
			\item \textbf{эктоморфы}~--- хрупкие, стройные, склонны к самоанализу;
			\item \textbf{мезоморфы}~--- склонны к беспокойству, сильные; именно они склонны к девиации.
		\end{itemize}
	\end{person}

	\subsubsection{Психологические причины}

	\begin{person}[Зигмунд Фрейд]
		Девиация связана с психологическим отклонением в личности. Таким людям свойственно чувство вины: они сами воспринимают свои действия как преступные и из-за склонности к разрушению даже хотят быть пойманными.

		Влечение к разрушению возникает из-за неправильной социализации. Самоограничения формируются в детстве (около $70\%$ личности), и важный момент первичной социализации~--- формирование \textbf{сверх-Я (суперэго)}, внутренней цензуры: системы запретов, моральных норм, традиций, представлений о допустимом. У некоторых людей из-за особых отношений с родителями или сверстниками этих самоограничений нет. Насильниками и преступниками чаще становятся те, кого унижали или оскорбляли сверстники.
	\end{person}

	\begin{note}
		Общее объяснение~--- недостаточно развитое суперэго и искажённая социализация. Эта теория объясняет девиантность лучше предыдущей, но недостаточно.
	\end{note}

	\subsubsection{Социокультурные причины}

	\paragraph{Теория аномии Дюркгейма}

	\begin{person}[Эмиль Дюркгейм]
		Французский социолог и философ. В $1897$ году опубликовал книгу <<Самоубийство>>, в которой доказывал, что самоубийства имеют не психологические, а социологические причины. Главная причина~--- аномия (от <<а>>~--- отрицание и греч. \textit{номос}~--- закон): отсутствие правил и норм.

		Нормы регулируют поведение, люди знают, как себя вести и чего ждать от других. Во время войн, кризисов и переломных эпох привычные нормы разрушаются и ставятся под сомнение, люди теряются, и возникает много суицидов и попыток. Аномия связана с нарушением общественного порядка и затрагивает прежде всего молодёжь. Теория критиковалась, но считается классикой социологии.
	\end{person}

	\paragraph{Теория аномии Мертона}

	\begin{person}[Роберт Мертон]
		Большинство американцев выражают жизненный успех богатством; признание получили только определённые пути его достижения (образование $\to$ хорошая работа $\to$ богатство). Но некоторые группы (бедняки, мигранты и др.) тоже приняли богатство как цель, но не могут разбогатеть легально (например, из-за худшего образования), что толкает их к нетрадиционным путям.

		\textbf{Причина девиации~--- разрыв между общепринятыми целями и средствами их достижения.} Мертон выделил $5$ типов реакции, из которых $4$ девиантные.
	\end{person}

	\begin{enumerate}
		\item \textbf{Конформизм}~--- цель и средство совпадают с общепринятыми (желание разбогатеть $\to$ образование $\to$ хорошая работа $\to$ заработок). Не девиация.
		\item \textbf{Инновация}~--- согласие с общепринятой целью, но отрицание общепринятых средств (путь обогащения слишком сложный, долгий, дорогой). Прямого нарушения закона может не быть, но человек пренебрегает моралью и балансирует на грани закона: авантюристы, игроки, спекулянты, коррупционеры.
		\item \textbf{Ритуализм}~--- отрицание цели и преувеличение значения средств. Цель считается недостижимой, но средства сохраняются как традиция (человек десятилетиями работает на нелюбимой работе, потому что так принято).
		\item \textbf{Ретретизм}~--- бегство от действительности, отказ и от целей, и от средств (алкоголики, наркоманы, люмпены; движение хиппи).
		\item \textbf{Бунт}~--- отрицание целей и средств и замена их принципиально новыми (революционеры-коммунисты отрицали богатство, их целью было справедливое бесклассовое общество).
	\end{enumerate}

	\begin{note}
		В стабильном обществе господствует конформный путь, в нестабильном возникает аномия. Социокультурное объяснение девиации~--- конфликт культурных ценностей.
	\end{note}

	\paragraph{Теория стигматизации}

	\begin{person}[Говард Беккер]
		Девиация определяется не самим поведением, а групповой оценкой, которую дают человеку другие. К девианту применяют санкции, так как он нарушает нормы.
	\end{person}

	\begin{itemize}
		\item \textbf{Первичная девиация}: человек время от времени нарушает нормы, но окружающие не придают этому значения, и сам себя он девиантом не считает (иногда выпивает).
		\item \textbf{Вторичная девиация}: на человека ставится клеймо девианта, отношение общества меняется, возникает социальная изоляция, и он сам начинает воспринимать себя как девианта. Так образуется спираль девиации.
	\end{itemize}

	\paragraph{Культурный перенос девиации}

	\begin{person}[Габриэль Тард]
		Люди становятся девиантами, потому что девиантная среда становится для них референтной группой, которой они подражают (стремление подростков из неблагополучных семей подражать бандитам). Чаще всего склонны к девиации \textbf{группы риска} (молодёжь). Образцы девиантного поведения усваиваются тем чаще, чем чаще человек с ними сталкивается и чем он моложе.
	\end{person}

	\subsection{Социальные конфликты}

	\begin{definition}[Конфликт]
		Противостояние сторон. Наука, изучающая причины, условия протекания и способы разрешения конфликтов,~--- \textbf{конфликтология}. Она близка к социологии, но отдельная наука: социология изучает общество и социальное взаимодействие в целом, а конфликтология сосредоточена только на конфликтах (на стыке социологии, психологии, политологии и др.).
	\end{definition}

	\subsubsection{Теории происхождения конфликтов}

	\begin{person}[Герберт Спенсер]
		Один из основателей социологии и конфликтологии, представитель натуралистического направления: социальная эволюция~--- часть общебиологической. Конфликт~--- проявление естественного отбора, который действует и в природе, и в обществе (слабые обречены). Перенос биологических законов на общество называется \textbf{социал-дарвинизмом}. Конфликты неизбежны и даже полезны, так как способствуют эволюции, но разрушительных революционных конфликтов следует избегать.
	\end{person}

	\begin{person}[Марксисты]
		Главный источник конфликтов~--- противоречие между классами. Класс эксплуататоров живёт за счёт труда эксплуатируемых, которые недовольны, отсюда конфликт. Конфликт~--- временное состояние: при коммунизме нет классов и нет конфликтов.
	\end{person}

	\begin{person}[Льюис Козер]
		Американский социолог, классик современной науки. Причины конфликтов:
		\begin{enumerate}
			\item дефицит ресурсов (есть в любом обществе);
			\item перераспределение доходов (несправедливое);
			\item власть и статус (люди стремятся к более высокому положению);
			\item переоценка ценностей.
		\end{enumerate}
		Пока существует человечество, будет существовать социальная напряжённость, которая временами перерастает в конфликты. Как и Спенсер, Козер считает конфликты в определённой мере полезными: они снимают напряжённость, выявляют скрытые противоречия, помогают найти компромисс. Конфликт~--- <<выпускной клапан>>, переводящий скрытое напряжение в открытую форму и предотвращающий более масштабный социальный взрыв.

		Роль конфликтов в открытом и закрытом обществе разная. В закрытом обществе конфликт имеет разрушительный характер (нет возможности открыто выразить интересы), в открытом существуют механизмы защиты интересов (демократические процедуры), поэтому конфликты скорее полезны и способствуют развитию.
	\end{person}

	\begin{note}
		Причина конфликта в том, что разные группы преследуют несовместимые интересы, связанные с общим предметом (материальным благом, моральными ценностями и т.\,д.).
	\end{note}

	\subsubsection{Сферы конфликтов}

	\begin{itemize}
		\item \textbf{Экономические}~--- разногласия из-за материальных благ.
		\item \textbf{Политические}~--- разные интересы в сфере политики.
		\item \textbf{Культурные}~--- несовпадение ценностей, разные представления о мире.
		\item \textbf{Социальные}~--- порождены социальным неравенством.
	\end{itemize}

	\subsubsection{Структура конфликта}

	\begin{itemize}
		\item \textbf{Объект}~--- ценность или благо, за которое идёт борьба (канал, который супруги хотят смотреть,~--- телевизор).
		\item \textbf{Субъекты}~--- участники: \textit{основные} (муж и жена) и \textit{неосновные} (свидетели, подстрекатели, пособники, посредники).
		\item \textbf{Инцидент}~--- повод для начала конфликта (муж переключил канал).
	\end{itemize}

	\begin{note}
		Конфликт не обязательно протекает в открытой форме: он может стать замороженным и перейти в скрытую форму.
	\end{note}

	\paragraph{Последствия конфликта}
	\begin{itemize}
		\item \textbf{Конструктивные}~--- совместное преодоление конфликта (пара купила второй телевизор).
		\item \textbf{Деструктивные}~--- затяжная обида и взаимные претензии приводят, например, к разводу.
	\end{itemize}

	\subsubsection{Виды социальных конфликтов}

	\paragraph{По количеству участников}
	\begin{itemize}
		\item \textbf{Внутриличностные} (например, ролевой конфликт: женщина разрывается между семьёй и работой; родители хотят, чтобы сын пошёл в вуз, а он хочет стать слесарем).
		\item \textbf{Межличностные}~--- разногласия между двумя и более членами одной группы (между друзьями).
		\item \textbf{Межгрупповые}~--- между группами с несовместимыми целями (социальные, межнациональные, религиозные).
	\end{itemize}

	\paragraph{По направленности}
	\begin{itemize}
		\item \textbf{Горизонтальные}~--- между людьми, не находящимися в подчинении друг у друга.
		\item \textbf{Вертикальные}~--- между людьми, находящимися в подчинении.
		\item \textbf{Смешанные}~--- участвуют и те, и другие.
	\end{itemize}

	\paragraph{По источнику возникновения}
	\begin{itemize}
		\item \textbf{Объективно обусловленные}~--- вызваны причинами вне воли людей, можно изменить только ситуацию (двое претендуют на одну должность или на одну пару обуви).
		\item \textbf{Субъективно обусловленные}~--- связаны с эмоциями и личными причинами сторон.
	\end{itemize}

	\paragraph{По характеру протекания}
	\begin{itemize}
		\item \textbf{Созидательные (конструктивные)}~--- основаны на уважении взаимных интересов и наиболее полезны (научная дискуссия, политическая борьба).
		\item \textbf{Разрушительные (деструктивные)}~--- участники не слышат друг друга и не стремятся к согласию, оппонента воспринимают как врага, цель~--- разрушить то, что не устраивает; любые проявления физического насилия (семейные ссоры, драки).
	\end{itemize}

	\paragraph{По длительности}
	\begin{itemize}
		\item \textbf{Кратковременные}.
		\item \textbf{Затяжные}~--- вызваны психологическими или объективными противоречиями, которые трудно разрешить.
	\end{itemize}

	\subsubsection{Способы разрешения конфликтов}

	\begin{enumerate}
		\item \textbf{Избегание}~--- физический или психологический уход от конфликта. Не обязательно худший вариант, но конфликт не устраняется, а лишь замораживается.
		\item \textbf{Откладывание}~--- не равно избеганию: временная сдача позиций ради накопления сил и аргументов, чтобы позже вернуться к спору с новыми возможностями.
		\item \textbf{Переговоры}~--- наилучший способ: позволяют избежать силы, достичь взаимопонимания и найти компромисс. Если стороны не доверяют друг другу, переговоры ведутся с посредником. Иногда помогает применение власти или закона.
	\end{enumerate}

	\paragraph{Функции социальных конфликтов}
	\begin{itemize}
		\item \textbf{Негативная}: конфликт в основном разрушает социальную систему и создаёт стресс.
		\item \textbf{Позитивные}:
		\begin{enumerate}
			\item Информативная: конфликт сигнализирует о проблеме, он не возникает просто так, и причины нужно устранять.
			\item Стимулирующая: чтобы не допустить повторения конфликтов, принимаются новые законы, создаются институты, общество развивается.
			\item Созидательная: конфликты помогают группе сплотиться, осознать единство и устранить противоречия.
		\end{enumerate}
	\end{itemize}

	\paragraph{Условия разрешения конфликта}
	\begin{enumerate}
		\item Своевременное выявление причин, пока конфликт не зашёл слишком далеко.
		\item Анализ причин возникновения.
		\item Определение целей и интересов сторон.
		\item Совместная заинтересованность сторон в преодолении (без согласия конфликт не разрешится).
		\item Совместный поиск путей преодоления (напрямую или через посредника; главное~--- общая воля).
	\end{enumerate}

	\subsection{Межнациональные конфликты}

	\begin{definition}[Межнациональный (межэтнический) конфликт]
		Столкновение представителей разных этнических групп, чаще всего между разными народами одного государства (но не обязательно). Самый опасный вид конфликта.
	\end{definition}

	Может протекать скрыто (недовольство наплывом мигрантов), затем вспыхивать беспорядками и снова переходить в вялотекущую форму. Может вырасти до региональных масштабов (Израиль и Палестина) или глобальных (Первая мировая война началась с конфликта сербов и австрийцев).

	\paragraph{Особенности}
	\begin{itemize}
		\item Стороны разделены по национальному признаку.
		\item Участники ищут поддержку в среде, близкой по национальному признаку.
		\item Как правило, у таких конфликтов есть явные или скрытые политические мотивы.
	\end{itemize}

	\paragraph{Виды}
	\begin{itemize}
		\item \textbf{Этнополитический}~--- столкновение политических интересов на национальной почве; национальные интересы часто служат прикрытием политических (Чеченский конфликт в России).
		\item \textbf{Межконфессиональный}~--- на религиозной почве (католики и протестанты).
		\item \textbf{Этнотерриториальный}~--- претензии разных народов на одну территорию.
	\end{itemize}

	\subsubsection{Национальная политика}

	Форм и видов политики по предотвращению межнациональных конфликтов очень много.

	\begin{itemize}
		\item \textbf{Концепция <<плавильного котла>>}~--- характерна для США: представители разных народов превращаются в единую американскую нацию. Если человек принимает ценности американского общества, он становится американцем независимо от национальности.
		\item \textbf{Этническая миксация}~--- объединение разных племён и народов в одну группу (латиноамериканцы).
		\item \textbf{Ассимиляция}~--- слияние одного народа с другим: народ утрачивает национальное самосознание, язык и идентичность, заимствуя их у другого. Бывает \textit{естественной} (добровольное постепенное слияние культур при смешанных браках, длительном соседстве) и \textit{насильственной} (государственная политика подавления языков и культур этнических меньшинств).
	\end{itemize}

	\subsubsection{Разрушительные практики и идеологии}

	\begin{itemize}
		\item \textbf{Национализм}~--- противопоставление одной нации другой. Бывает \textit{политический} (в нём как бы нет ничего плохого) и \textit{этнический} (утверждение превосходства одной нации).
		\item \textbf{Шовинизм}~--- угнетение малых наций большими. Название от имени Николя Шовена, солдата армии Наполеона, прославившегося восхвалением наполеоновской политики и враждебностью к другим нациям.
		\item \textbf{Геноцид}~--- шовинизм и дискриминация в самом концентрированном виде; тяжкое преступление.
	\end{itemize}

	\begin{definition}[Сепаратизм]
		От лат. \textit{separatus}~--- отдельный. Стремление к отделению части государства. Принимает более мягкие формы (автономизация) или жёсткие (полное отделение по национальному признаку). Примеры: шотландцы, чеченцы.
	\end{definition}

	\begin{definition}[Ирредентизм]
		От итал. \textit{irredento}~--- неосвобождённый. Стремление этнической общности отделиться от одного государства и присоединиться к другому, этнически близкому. Классический пример~--- движение за присоединение к Италии территорий с итальянским населением (Трентино, Триест).
	\end{definition}

	\subsection{Социальные институты}

	\begin{definition}[Социальные институты]
		Исторически сложившиеся устойчивые формы совместной деятельности, направленные на удовлетворение жизненно важных потребностей. Основа~--- социальные нормы. Примеры: институты семьи, брака, армии, образования.
	\end{definition}

	\begin{note}
		За каждым институтом стоит длительная история и важная базовая потребность общества, без которой оно не может нормально существовать. Потребность в передаче знаний порождает институт образования (правила, учреждения, методики, набор ролей), потребность в безопасности~--- институт армии.
	\end{note}

	\begin{definition}[Институционализация]
		Становление социального института. Этапы:
		\begin{enumerate}
			\item Возникновение потребности, осознаваемой как общественная (а не частная), для удовлетворения которой нужна совместная деятельность.
			\item Формирование общих целей в ходе совместной деятельности.
			\item Появление норм и правил.
			\item Формирование системы санкций за нарушение правил.
			\item Создание системы статусов и ролей, охватывающей всех участников.
		\end{enumerate}
	\end{definition}

	\begin{person}[Герберт Спенсер]
		Ввёл понятие <<социальный институт>>. Как сторонник натуралистического подхода, сравнивал общество с биологическим организмом, у которого есть функции, поддерживающие жизнедеятельность; эти функции и есть социальные институты: семья, государство, религия, образование, промышленность, торговля (государство~--- порядок, армия~--- безопасность, семья~--- воспроизводство, образование~--- культурная преемственность). Этот подход разделяет большинство социологов.
	\end{person}

	\begin{person}[Торстейн Веблен]
		Рассматривал социальные институты как деятельность по удовлетворению потребностей, а не как функции.
	\end{person}

	\subsubsection{Классификация институтов}

	\paragraph{По назначению}
	\begin{itemize}
		\item Воспроизводство членов общества (семья, государство).
		\item Социализация членов общества (семья, образование).
		\item Производство и распределение (промышленность, сельское хозяйство).
		\item Управление и контроль (государство, право, религия).
	\end{itemize}

	\paragraph{По масштабу деятельности}
	\begin{itemize}
		\item \textbf{Основные}~--- удовлетворяют базовые потребности (общественный порядок, продолжение рода, воспитание, передача знаний). Пример: государство, включающее институты монархии, республики и т.\,д.
		\item \textbf{Неосновные}~--- решают частные задачи. Пример: брак внутри семьи (семья возможна и без брака, поэтому брак не основной).
	\end{itemize}

	\paragraph{По внутренней структуре}
	\begin{itemize}
		\item \textbf{Простые}~--- без внутреннего деления (семья).
		\item \textbf{Сложные}~--- состоят из нескольких простых.
	\end{itemize}

	\paragraph{По сфере деятельности}
	Политические, экономические, социально-культурные, семейные, религиозные.

	\subsubsection{Функции социальных институтов}

	Главная задача институтов~--- сохранять социальную стабильность; остальные функции вытекают из неё.

	\begin{itemize}
		\item \textbf{Регулятивная}: нормы и правила регулируют поведение, обеспечивают социальный контроль и стабильность.
		\item \textbf{Интегративная}: объединяет людей, обеспечивает взаимодействие и взаимосвязи.
		\item \textbf{Трансляция опыта и социализация}: передача знаний, опыта и ценностей от поколения к поколению.
		\item \textbf{Воспроизводство}: прежде всего у семьи; физическое и социальное воспроизведение членов общества.
		\item \textbf{Удовлетворение потребностей}: производство благ, обеспечение стабильности и безопасности.
	\end{itemize}

	\begin{note}
		Институты складываются очень медленно, консервативны и меняются медленно: это самая стабильная часть общественной жизни. Поэтому консервативные партии берут их под защиту, а любая революция предполагает слом или радикальное изменение институтов. Пока институты действуют, в обществе царит порядок.
	\end{note}

	\subsubsection{Функции и дисфункции (по Мертону)}

	\begin{person}[Роберт Мертон]
		Социальный институт может быть функциональным, нефункциональным или дисфункциональным.
	\end{person}

	\begin{definition}[Функция]
		Работа института, идущая на благо общества, поддерживающая его стабильность и увеличивающая его адаптивность (способность реагировать на вызовы среды и сохранять порядок).
	\end{definition}

	\begin{definition}[Дисфункция]
		Всё, что разрушает общество и уменьшает его адаптивность. Пример: полиция создана для поддержания порядка, но если полицейские начинают покрывать группировки и брать взятки, они наносят обществу ущерб; так же чиновники, занимающиеся коррупцией наряду с управленческими функциями.
	\end{definition}

	\begin{note}
		Функции и дисфункции бывают \textbf{явными} и \textbf{скрытыми} (скрытыми чаще бывают дисфункции, но и функции могут: если институт создан с одной целью, а со временем начинает выполнять несвойственные функции~--- санитарный контроль используется во внешнеполитических целях и запрещает импорт <<нелицензированных>> товаров). В разных условиях авторитет институтов повышается или понижается (в индустриальном обществе в спокойное время растёт авторитет образования, в информационном~--- падает авторитет традиционных СМИ в пользу интернета). Если авторитет или легитимность института снижается, его называют \textbf{нефункциональным}, что свидетельствует об \textbf{институциональном кризисе}.
	\end{note}

	\subsection{Семья и брак}

	\begin{note}
		Семью изучают не только социология, но и право, демография и др., чаще всего рассматривая её как малую социальную группу и как социальный институт. \textbf{Семья как институт}~--- отношения семьи и государства, выполнение социальных функций, встроенность в общество, зависимость от политического режима и экономики. \textbf{Семья как малая группа}~--- внутренние отношения, относительная самостоятельность.
	\end{note}

	\begin{definition}[Семья]
		Малая социальная группа, члены которой связаны брачными или родственными отношениями, общим бытом и взаимной ответственностью. Понятие шире брака: рождение и воспитание детей и совместное хозяйство возможны и без официальной регистрации.
	\end{definition}

	\begin{definition}[Брак]
		Исторически сложившийся социально регулируемый союз мужчины и женщины, который устанавливает их права и обязанности по отношению друг к другу, к детям и к обществу. Брак возник позже семьи и регулирует имущественные и правовые положения супругов и членов семьи.
	\end{definition}

	\begin{note}
		В России в $2020$ году в Конституции закреплено, что брак~--- союз мужчины и женщины; в ряде стран разрешены однополые браки.
	\end{note}

	\begin{person}[Платон]
		Давал философское толкование брачному союзу: в мифе об андрогинах (двуполых совершенных существах) боги, позавидовав их совершенству, разделили их пополам, поэтому брак~--- влечение к утраченной цельности.
	\end{person}

	\begin{person}[Льюис Генри Морган]
		Американский историк и социолог, автор книги <<Древнее общество>> ($1877$); считается основателем науки о семье. Впервые применил исторический метод: семейные отношения включены в общий ход истории. С его точки зрения, все семьи складывались примерно одинаково (моногенизм). Этапы развития брачно-семейных отношений сменяли друг друга от группового брака до моногамии:
		\begin{enumerate}
			\item \textbf{Групповой брак (промискуитет)}, характерный для первобытного стада: беспорядочные отношения между всеми. Затем родовой строй с первой формой~--- матриархатом (происхождение по материнской линии; собирательство развито лучше охоты и земледелия, им занимались в основном женщины, которые управляли хозяйством и родом).
			\item \textbf{Парная семья}~--- исторически первая, непрочная, часто временная форма, допускавшая внебрачные связи; своего хозяйства нет, оно ещё родовое. Парный брак~--- переход от группового к моногамному. С совершенствованием орудий труда и разделением на скотоводство, земледелие и охоту приходит патриархат: нужны физические силы, роль мужчины растёт.
			\item \textbf{Моногамная семья}~--- современный тип: союз двух людей с общим хозяйством и собственностью, которая начинает передаваться по наследству. Со временем становится нормой для человечества.
		\end{enumerate}
	\end{person}

	\begin{note}
		Семья значительно улучшила положение женщины в древнем обществе и помогла решить три взаимосвязанные проблемы: женскую, детскую и демографическую. В первобытном обществе женщина полностью зависела от мужчин (постоянная беременность и вскармливание, невозможность самой добывать пищу). При парном браке о детях заботятся оба супруга, мужчина обеспечивает женщину всем необходимым, снижается детская смертность, растёт рождаемость, и население стремительно увеличивается.
	\end{note}

	\subsubsection{Формы брака}

	\begin{itemize}
		\item \textbf{Моногамия}~--- союз одного мужчины и одной женщины.
		\item \textbf{Полигамия (многожёнство)}~--- союз одного мужчины с несколькими женщинами; узаконена в некоторых странах арабского Востока.
		\item \textbf{Полиандрия}~--- союз одной женщины с несколькими мужчинами; бывает в Индии и Тибете.
	\end{itemize}

	\begin{note}
		Эти формы возникли из-за нехватки земли: её нельзя дробить между сыновьями, так как через несколько поколений участки измельчали бы, и нужно сохранять землю в одних руках. Для христианского и постхристианского общества наиболее характерен моногамный брак, основанный на равенстве полов.
	\end{note}

	\subsubsection{Типология семьи}

	\paragraph{По структуре родственных связей}
	\begin{itemize}
		\item \textbf{Простая (нуклеарная)}~--- не более двух поколений: супруги без детей или с неженатыми (незамужними) детьми.
		\item \textbf{Сложная (расширенная)}~--- несколько поколений: прародители, родители, дети и другие родственники.
	\end{itemize}

	В традиционном аграрном обществе обычны многопоколенные семьи: большое хозяйство требует много рабочих рук и разделения обязанностей, поэтому большие семьи поощрялись. С урбанизацией такие семьи становятся неудобными (нет места, нет смысла), карьера и ритм жизни не способствуют многодетности, семьи мельчают. Типичная нуклеарная семья: мама, папа и двое детей.

	\paragraph{По количеству детей}
	\begin{itemize}
		\item Малодетные ($1$--$2$ ребёнка).
		\item Среднедетные ($3$--$4$).
		\item Многодетные ($5$ и более).
		\item Бездетные.
		\item Неполные~--- с детьми, но без одного из родителей.
	\end{itemize}

	\subsubsection{Семейные статусы и роли}

	Семейный социальный статус (супружеский, детский, родительский) предписан, а правила ролевого поведения устанавливаются семьёй индивидуально и зависят от модели социализации в данном обществе.

	\begin{itemize}
		\item \textbf{Традиционная семья} патриархальна: строгое распределение обязанностей, сильное различие старших и младших, мужчин и женщин по полномочиям (иерархия).
		\item \textbf{Современная (эгалитарная) семья}: равные права супругов, условное распределение обязанностей, оба супруга работают и обеспечивают семью, равные права в воспитании детей, роли пересекаются.
	\end{itemize}

	Оба типа распространены, но у эгалитарной больше перспектив, и такой тип отношений предпочтителен.

	\subsubsection{Социальные функции семьи}

	\begin{itemize}
		\item \textbf{Репродуктивная}~--- воспроизводство биологического вида.
		\item \textbf{Воспитательная}~--- первичная социализация личности.
		\item \textbf{Экономическая}~--- у любой семьи есть бюджет, который нужно планировать.
		\item \textbf{Хозяйственно-бытовая}~--- распределение обязанностей.
		\item \textbf{Эмоционально-психологическая}~--- поддержка, понимание.
		\item \textbf{Рекреационная (досуговая)}~--- отпуск, выходные, отдых после работы.
		\item \textbf{Социально-статусная}~--- предоставление социальных ролей; дети получают первичный социальный статус.
		\item \textbf{Первичный социальный контроль}.
	\end{itemize}

	\subsubsection{Проблемы современной семьи}

	Кризис традиционной семьи связан с изменением гендерных ролей: до $1970$-х годов женщины были преимущественно домохозяйками, затем стали строить карьеру, иногда отказываясь от семьи ради работы. Распространились альтернативные формы брака:
	\begin{itemize}
		\item \textbf{Пробный брак}~--- временное совместное проживание без оформления отношений, до появления детей.
		\item \textbf{Открытый (гостевой) брак}~--- признание права партнёра на самостоятельную жизнь, партнёры живут раздельно и время от времени встречаются. Английский писатель-анархист Уильям Годвин считал брак способом эксплуатации женщины и отрицал его.
	\end{itemize}

	\begin{note}
		В таких формах есть разумные доводы, но это негативная тенденция: она формирует представление о семейной жизни как о временной и необязательной, ведёт к отказу от взаимной ответственности, а брак как раз предполагает её. Альтернативные формы брака~--- пример дисфункции. К браку нужно относиться ответственно.
	\end{note}

	\subsection{Молодёжь как социальная группа}

	\begin{definition}[Молодёжь]
		Понятие рассматривают в трёх смыслах: \textbf{демографическом}~--- возрастная группа; \textbf{социологическом}~--- особенности положения молодёжи в обществе; \textbf{социально-психологическом}~--- самоощущение молодёжи. Возрастные границы: $15$--$29$ лет.
	\end{definition}

	\begin{note}
		Нижняя граница связана с половым созреванием, окончанием основной школы и началом работы или профессионального образования. Границы условны и исторически подвижны. В традиционном обществе взрослели рано (детство заканчивалось в $5$--$7$ лет, в $12$--$14$ уже создавали семью): социализация была проще, а продолжительность жизни меньше. В современном обществе из-за роста значимости образования границы отодвинулись, верхняя граница доходит до $29$--$30$ лет, когда у человека уже есть своя семья.
	\end{note}

	Молодёжь социально неоднородна: городская, сельская, учащаяся, рабочая. Рабочая молодёжь начинает трудовую жизнь раньше (около $16$ лет), так как появляются материальные возможности для создания семьи, и рабочие семьи образуются рано. У среднего класса сначала долгая учёба, затем поиск работы и только потом семья. Молодость у одних групп заканчивается раньше, чем у других.

	\subsubsection{Молодёжь с социально-психологической точки зрения}

	Это время формирования личности и выработки собственного опыта: осваиваются новые роли (студент, работник, семейная роль), определяется место в жизни, это хорошее время для карьерного роста. В одних вопросах молодёжь проявляет зрелость и ответственность, в других остаётся инфантильной.

	\begin{definition}[Инфантилизм]
		Нежелание и неумение развиваться (от лат. \textit{infans}~--- ребёнок); восприятие мира по законам детства, когда мир~--- магазин игрушек для собственных удовольствий.
	\end{definition}

	\begin{note}
		Ответственность~--- главный признак взросления. Взросление в социально-психологическом плане~--- формирование ответственности за себя и окружающих.
	\end{note}

	\subsubsection{Основные проблемы молодёжи}

	Связаны с переходным состоянием и нестабильностью.

	\begin{enumerate}
		\item \textbf{Экономические}: недостаточная обеспеченность, отсутствие работы и стабильного заработка, отсутствие собственного жилья, зависимость от родителей.
		\item \textbf{Духовные и психологические}. Важнейшая~--- проблема <<отцов и детей>>, конфликт поколений. Его причины:
		\begin{itemize}
			\item Разные типы культуры. Старшее поколение мало заинтересовано в переменах: оно достигло определённого положения и благополучия, жизнь налажена, поэтому ему характерны бытовой и политический конформизм и потребность в стабильности. Молодым нужны перемены, они открыты новым знакомствам, увлечениям, деятельности.
			\item Старшее поколение идеализирует свой опыт и считает его единственно правильным, и в мелочах (вкусы, мода, жаргон), и в серьёзном (политические предпочтения, культурные ценности). Поэтому молодёжь воспринимается не как носитель альтернативного опыта, а как объект воздействия.
		\end{itemize}
	\end{enumerate}

	\begin{person}[Толкотт Парсонс]
		В социологии до $1960$-х годов преобладала его теория, согласно которой молодёжь~--- объект социального воздействия: ею нужно управлять, а недовольство считать девиацией и устранять запретительными мерами. Так было не только в СССР, но и на Западе.
	\end{person}

	\begin{note}
		Отношение изменилось в конце $1960$-х годов после молодёжных движений в Америке и Европе (хиппи, <<дети цветов>>). Анализ показал, что молодёжные протесты имеют важные социальные причины: они поднимали проблемы общества в целом (социальное неравенство, половая и расовая дискриминация, бездуховность), просто молодёжь в силу возраста и психологии реагирует острее. Вывод: молодёжь~--- не только объект воспитания, но и носитель нового социального опыта, которого нет у старшего поколения. Общество стремится предоставить молодёжи каналы вертикальной мобильности.
	\end{note}

	\subsubsection{Социализация молодёжи}

	\begin{itemize}
		\item \textbf{Направленная (системная)}~--- организуется общественной системой, институтами и агентами социализации (школа, вуз, армия).
		\item \textbf{Стихийная (ненаправленная)}~--- со стороны сверстников, молодёжных групп, СМИ, интернета.
		\item \textbf{Самовоспитание}~--- свидетельствует, что личность достаточно развита, чтобы работать над собой самостоятельно.
		\item \textbf{Экстремальная}~--- в кризисных, необычных условиях (тюремное заключение, война, тяжёлые жизненные обстоятельства). Всё индивидуально, зависит от человека; кризисом может стать даже переход в другую школу.
	\end{itemize}

	\subsubsection{Молодёжная субкультура}

	\begin{definition}[Молодёжная субкультура]
		Субкультура, создаваемая самими молодыми людьми для себя; поэтому она закрыта, эзотерична, непонятна посторонним. Её задача~--- отгородиться от мира взрослых и создать собственный стиль, моду, сленг. Если она противостоит доминирующей культуре, то становится контркультурой.
	\end{definition}

	\paragraph{Общие черты}
	\begin{itemize}
		\item Стремление к обособлению от основного мира, мира взрослых, общества в целом и других молодёжных групп.
		\item Желание принадлежать к общности, где находят понимание, общие интересы и поддержку (у рокеров одни интересы, у футбольных фанатов другие), свой стиль поведения, сленг.
		\item Влияние глобализации и интернета: субкультуры формируются под воздействием мировых (прежде всего западных, а также восточных, например корейской моды) культурных тенденций.
	\end{itemize}
