	\section{Механика}
	\begin{definition}
		Механическое движение~--- изменение пространственного положения тела относительно других тел с течением времени.
	\end{definition}
	\begin{definition}
		При поступательном движении прямая проведенная через любые две точки внутри тела остается параллельна сама себе.
	\end{definition}
	\begin{definition}
		При вращательном движении каждая точка тела вращается по своей окружности, центры этих окружностей лежат на одной прямой, прямая называется осью вращения.
	\end{definition}
	\begin{statement}
		Любое движение~--- сумма этих двух движений.
	\end{statement}
	\begin{definition}
		Колебательное движение~--- движение, повторяющееся с той или иной точностью во времени.
	\end{definition}

	\subsubsection{Движение под углом горизонта}
	Тело брошено с высоты $h$ под углом $\alpha$ со скоростью $V_0$.
	\begin{enumerate}
		\item $V_x = V_0 \cos\alpha$
		\item $x = V_0 \cos\alpha t$
		\item $V_y = V_0 \sin\alpha - gt$
		\item $y = h_0 + V_0 \sin\alpha t - \frac{gt^2}{2}$
	\end{enumerate}
	I. Траектория.
	\begin{quote}
		$t = \frac{x}{V_0 \cos\alpha}$.\\
		$y = h_0 + V_0 \sin\alpha \frac{x}{V_0 \cos\alpha} - \frac{g}{2} \cdot \frac{x^2}{V_0^2 \cos^2\alpha}$.\\
		$y = h_0 + x \tg\alpha - \frac{gx^2}{2V_0^2\cos^2\alpha}$.\\
	\end{quote}
	II. $H_{max}$: $V_{y} = 0$.
	\begin{quote}
		$0 = V_0 \sin\alpha - gt_{\text{падения}}$.\\
		$t_{\text{падения}} = \frac{V_0 \sin\alpha}{g}$.\\
		$H_{max} = h_0 + V_0 \sin\alpha \cdot \frac{V_0 \sin\alpha}{g} - \frac{g}{2} \cdot \frac{V_0^2 \sin^2\alpha}{g^2}$.\\
		$H_{max} = h_0 + \frac{V_0^2 \sin^2\alpha}{2g}$.\\
	\end{quote}
	III. $t_{\text{полета}}$: $y = 0$.
	\begin{quote}
		$0 = h_0 + V_0 \sin\alpha t_{\text{полета}} - \frac{gt_{\text{полета}}^2}{2}$.\\
		$\frac{gt_{\text{полета}}^2}{2} - V_0 \sin\alpha t_{\text{полета}} - h_0 = 0$.\\
		$t_{\text{полета}} = \frac{V_0 \sin\alpha + \sqrt{V_0^2\sin^2\alpha + 2gh_0}}{g}$.\\
	\end{quote}
	IV. Дальность полета: $L$.
	\begin{quote}
		$L = x(t_{\text{полета}}) = V_0 \cos\alpha t_{\text{полета}}$.\\
		$L = \frac{V_0^2 \sin2\alpha}{g}$.\\
	\end{quote}
	V. Конечная скорость.
	\begin{quote}
		$V_{\text{y к}} = V_0 \sin\alpha - gt_{\text{полета}} = V_0 \sin\alpha - g \cdot \frac{V_0 \sin\alpha}{g} - g \cdot \frac{\sqrt{V_0^2 \sin^2\alpha + 2gh_0}}{g}$.\\
		$V_{\text{x к}} = V_0 \cos\alpha$.\\
		$V_{\text{y к}} = -\sqrt{V_0^2 \sin^2\alpha + 2gh_0}$.\\
		$V_{\text{к}} = \sqrt{V_{\text{x к}}^2 + V_{\text{y к}}^2} = \sqrt{V_0^2 \cos^2\alpha + V_0^2 \sin^2\alpha + 2gh_0}$.\\
		$V_{\text{к}} = \sqrt{V_0^2 + 2gh_0}$.\\
	\end{quote}
	VI. Угол падения ($\beta$).
	\begin{quote}
		$\cos\beta = \frac{V_x}{V_{\text{к}}} = \frac{V_0 \cos\alpha}{\sqrt{2gh_0 + V_0^2}}$.
	\end{quote}

		Инерция~--- способность тела сохранять скорость при отсутствие внешнего воздействия.
	\end{definition}
	\begin{definition}
		Три закона Ньютона:
		\begin{enumerate}[1.]
			\item Существуют инерциальные системы отсчета (ИСО). ИСО --- те системы отсчета, в которых если на тело не действуют силы или их действие скомпенсировано, то тело движется равномерно и прямолинейно или покоится.
			\item $\sum\vec{F} = m\vec{a}$.\\
			Инертность --- свойство тела, которое заключается в том, что для изменения скорости тела необходимо время.
			\item При взаимодействие двух тел возникает две силы. Эти две силы приложены к двум разным телам, равным по модулю, противоположны по направлению, лежат на одной прямой, имеют одну природу (гравитационная, электромагнитная, сильная, слабая).
		\end{enumerate}
	\end{definition}
	\begin{statement}[Ограничения на законы]
		Работают только для скоростей много меньших скоростей света, в инерциальных системах счисления и с ненулевыми массами.
	\end{statement}
	\begin{statement}
		Полезная информация:
		\begin{enumerate}
			\item Тело стоит на платформе, платформа движется вверх с ускорением $\vec{a}$, у тела масса $m$, то $P = m \cdot (g + a)$.
			\item Тело стоит на платформе, платформа движется вниз с ускорением $\vec{a}$, у тела масса $m$, то $P = m \cdot (g - a)$.
		\end{enumerate}
	\end{statement}

	\begin{theorem}[О движении центра масс]
		Центр масс тела движется таким образом, как будто он точка массой $m_{\text{общ}}$ и все силы приложены к этой точке.
	\end{theorem}
	\begin{proof}
		$\frac{\varDelta (\varDelta m_i \vec{V}_i)}{\varDelta t} = \frac{\varDelta \vec{p}_i}{\varDelta t} = \sum\limits_{i + k} \vec{F}_{ik} + \sum\limits_i \vec{F}_{\text{внеш}}$\\
		$\frac{\sum\limits_i \varDelta \varDelta m_i \vec{V}_i}{\varDelta t} = \vec{F}_{\text{внеш}}$\\
		$\frac{\varDelta \sum\limits_i \varDelta m_i \vec{V}_i}{\varDelta t} = \vec{F}_{\text{внеш}}$\\
		$r_c' = V_c = \frac{\sum\limits_i \varDelta m_i V_i}{m}$\\
		$\varDelta \frac{m \vec{V}_c}{\varDelta t} = \vec{F}_{\text{внеш}}$\\
		$m \vec{a}_c = \vec{F}_{\text{внеш}}$\\
		$\Rightarrow$ Если внешних сил не действует, то центр масс покоится, если покоился, или движется по инерции, если двигался по инерции.
	\end{proof}

	\subsection{Теплоемкость}
	\begin{definition}[Теплоемкость]
		$c = \dfrac{Q}{m \varDelta T} = \dfrac{A + \varDelta U}{m \varDelta T}$
	\end{definition}
	\begin{enumerate}
		\item Изохорный процесс.\\
		$V = const \Rightarrow A = 0$\\
		$c = \dfrac{\varDelta U}{m \varDelta T} = \dfrac{\frac{i}{2} \nu R \varDelta T}{\nu M \varDelta T} = \dfrac{\frac{i}{2} R}{M}$\\
		$c_{M_V} = c \cdot M = \dfrac{iR}{2}$
		\item Изобарный процесс.\\
		$p = const$\\
		$c = \dfrac{\varDelta U + A}{m \varDelta T} = \dfrac{\frac{i}{2} R \varDelta T \nu + p \varDelta V}{m \varDelta T} = \dfrac{\frac{i}{2} R \varDelta T \nu + \nu R \varDelta T}{m \varDelta T} = \left( \dfrac{i}{2} + 1 \right) \dfrac{R}{M}$\\
		$c_{M_p} = \left( \dfrac{i}{2} + 1 \right) R = c_{M_V} + R$~--- соотношение Майера.\\
		$\gamma = \dfrac{c_{M_p}}{c_{M_V}} = \dfrac{i + 2}{i}$~--- коэффициент Пуассона.
		\item Изотермический процесс.\\
		$c = \dfrac{Q}{m \varDelta T} = \dfrac{A}{m \varDelta T}$\\
		$T = const \Rightarrow \varDelta T = 0 \Rightarrow \varDelta U = 0 \Rightarrow c \rightarrow \infty$
		\item Адиабатный процесс.\\
		$Q = 0$\\
		$c = \dfrac{Q}{m \varDelta T} = 0$
	\end{enumerate}

	\paragraph{Закон Кулона}
	
	Кулон открыл, что $F \sim q_1$, $F \sim q_2$, $F \sim \dfrac{1}{r^2}$. Позднее была получена формула:
	$$
	F = \dfrac{k |q_1| |q_2|}{\varepsilon r^2}, \, k = 9 \cdot 10^9 \, \dfrac{\text{Н} \cdot \text{м}^2}{\text{Кл}^2},
	$$
	$k$~--- электрическая постоянная, $\varepsilon$~--- диэлектрическая проницаемость.
	
	Условия для соблюдения закона:
	\begin{itemize}
		\item тела~--- точечные,
		\item тела не двигаются.
	\end{itemize}

	\paragraph{Теория Друде} Уравнение движения свободного электрона в электрическом поле из классической теории электропроводности Друде, где связываются сила поля, дрейфовая скорость, время и длина свободного пробега электрона: $E \cdot \overline{e} = m \cdot a = \dfrac{m V_{max}}{\tau} = \dfrac{m V_{max} u}{\lambda}$, где $u = \sqrt{\dfrac{3KT}{m_e}} = \sqrt{\dfrac{3 \cdot 1.38 \cdot 10^{-23} \, \frac{\text{Дж}}{\text{К}} \cdot 300 \, \text{К}}{9.1 \cdot 10^{-31} \, \text{кг}}} \approx 1.168 \cdot 10^5 \, \dfrac{\text{м}}{\text{с}}$, $\tau = \dfrac{\lambda}{u}$.
	
	$E e = \dfrac{m u}{\lambda} \cdot \dfrac{2 I}{e n S} \Rightarrow j = E \cdot \dfrac{e^2 \lambda n}{2 m u} = E \sigma \Rightarrow I = \dfrac{\sigma S}{d} U \Rightarrow I = GU$.

	\section{Колебания}
	
	\subsection{Гармонические колебания}
	
    \begin{definition}[Гармонические колебания]
    	Колебания называются гармоническими, если возвращающая сила (а значит, и ускорение) пропорциональна смещению тела от положения равновесия и направлена против него.
    \end{definition}
	
	\subsection{Уравнение колебаний}
	
	\subsubsection{Пружинный маятник (горизонтальный)}
	
	Груз массой $m$ лежит на гладкой горизонтальной поверхности и прикреплен к пружине жесткости $k$; ось $Ox$ направлена вдоль движения, начало~--- в положении равновесия.
	
	\[ \vec{F}_{\text{упр}} + \vec{N} + m\vec{g} = m\vec{a} \]
	
	\[ Ox: \ F_{\text{упр }x} = ma_x \]
	
	\[-kx = ma_x\]
	
	\[ \omega_0^2 = \dfrac{k}{m} \text{~--- циклическая частота} \]
	
	\[ -\omega_0^2 x = a_x \]
	
	\subsubsection{Пружинный маятник (вертикальный)}
	
	Груз массой $m$ подвешен на пружине жесткости $k$. В положении равновесия пружина растянута на $|\varDelta l|$, при этом $F_{\text{упр}0} = k|\varDelta l|$, $\vec{mg} \downarrow$. Отклоним груз от положения равновесия на $\varDelta x$, тогда полное удлинение пружины равно $\varDelta x + |\varDelta l|$.
	
	\[ Ox: \ mg - k(\varDelta x + |\varDelta l|) = ma_x \]
	
	\[ \text{Так как } mg = k|\varDelta l| \text{, получаем } -kx = ma_x \text{, где } x = \varDelta x\]
	
	\begin{statement}
		Период колебаний пружинного маятника: $T = 2\pi \sqrt{\dfrac{m}{k}}$.
	\end{statement}
	
	\subsubsection{Математический маятник}
	
	Груз на нити длиной $l$ отклонен на угол $\alpha$ от вертикали ($\alpha$~--- мал), дуга $s = l\alpha \Rightarrow \alpha = \dfrac{s}{l}$. Проекция второго закона Ньютона на касательное направление:
	
	\[ -mg\sin\alpha = ma_{\tau} \]
	
	\[ \sin\alpha \approx \alpha \text{ (при малых углах)} \Rightarrow -g\alpha = a_{\tau} \]
	
	\[ -g\dfrac{s}{l} = a_{\tau} \]
	
	\[ \omega_0^2 = \dfrac{g}{l} \]
	
	\[ -\omega_0^2 s \approx a_{\tau} \]
	
	\begin{statement}
		Период колебаний математического маятника: $T = 2\pi \sqrt{\dfrac{l}{g}}$.
	\end{statement}
	
	\subsubsection{Обобщение}
	
	\begin{statement}
		Во всех трех случаях получено уравнение вида $a = -\omega_0^2 x$~--- это и есть уравнение гармонических колебаний, где $\omega_0$~--- циклическая (собственная) частота системы, $T = \dfrac{2\pi}{\omega_0}$.
	\end{statement}
	
    \subsubsection{Решение уравнения гармонических колебаний}
    
    \begin{statement}
    	Функция $x(t) = x_m \sin (\omega_0 t)$ является решением уравнения $x'' = -\omega_0^2 x$.
    \end{statement}
    \begin{proof}
    	$x = x_m \sin (\omega_0 t)$ \\
    	$x' = x_m \omega_0 \cos (\omega_0 t) =: V_x$ \\
    	$x'' = -x_m \omega_0^2 \sin (\omega_0 t) = -\omega_0^2 x$
    \end{proof}
    
    \begin{statement}[Общее решение]
    	Общим решением уравнения гармонических колебаний является
    	$$x = x_m \sin (\omega_0 t + \varphi_0),$$
    	где $x_m$~--- амплитуда, $\varphi_0$~--- начальная фаза. Проверка подстановкой полностью аналогична случаю $\varphi_0 = 0$ выше.
    \end{statement}
    
    \subsubsection{Нахождение амплитуды и начальной фазы по начальным условиям}
    
    \begin{problem}
    	Дано: $x_0$, $v_0$, $\omega_0$ ($v_{0x} > 0$). Найти: $x_m$, $\varphi_0$.
    \end{problem}
    
    \begin{solution}
    	Подставим $t = 0$ в $x(t)$ и $x'(t)$:
    	$$x_0 = x_m \sin (\varphi_0), \qquad v_0 = x_m \omega_0 \cos (\varphi_0).$$
    	Возведем оба равенства в квадрат и сложим:
    	$$x_0^2 + \frac{v_0^2}{\omega_0^2} = x_m^2(\sin^2(\varphi_0) + \cos^2(\varphi_0)) = x_m^2 \; \Rightarrow \; x_m = \sqrt{x_0^2 + \frac{v_0^2}{\omega_0^2}}.$$
    	Разделим первое равенство на второе:
    	$$\frac{x_0 \omega_0}{v_0} = \tg (\varphi_0) \; \Rightarrow \; \varphi_0 = \arctg \left( \frac{x_0 \omega_0}{v_0} \right).$$
    \end{solution}

	\subsubsection{Вывод периода через закон сохранения энергии}

	\begin{statement}
		Пусть полная механическая энергия системы, совершающей одномерное движение, имеет вид
		\[
			E = Ax^2 + B(\dot x)^2 = \mathrm{const},
		\]
		где $A, B > 0$~--- постоянные. Тогда система совершает гармонические колебания с циклической частотой $\omega_0 = \sqrt{A/B}$ и периодом
		\[
			T = 2\pi\sqrt{\frac{B}{A}}.
		\]
	\end{statement}
	\begin{proof}
		Продифференцируем $E$ по времени (энергия постоянна, поэтому $\dot E = 0$):
		\[
			2Ax\dot x + 2B\dot x\ddot x = 0.
		\]
		Сокращая на $2\dot x$ (случай $\dot x \equiv 0$ не рассматриваем), получаем $Ax + B\ddot x = 0$, то есть $\ddot x = -\dfrac{A}{B}x$~--- уравнение гармонических колебаний с $\omega_0^2 = A/B$.
	\end{proof}

	\paragraph{Пружинный маятник.}
	\[
		E = \frac{k}{2}x^2 + \frac{m}{2}V_x^2.
	\]
	Здесь $A = k/2$, $B = m/2$. Продифференцировав по времени напрямую:
	\[
		\frac{k}{2}\cdot 2xV_x + \frac{m}{2}\cdot 2V_xV_x' = 0 \quad \Longrightarrow \quad kx + mV_x' = 0 \quad \Longrightarrow \quad kx + ma_x = 0,
	\]
	что совпадает с уравнением $\ddot x = -\dfrac{k}{m}x$, полученным ранее из второго закона Ньютона. Отсюда
	\[
		T = 2\pi\sqrt{\frac{m/2}{k/2}} = 2\pi\sqrt{\frac{m}{k}}.
	\]

	\paragraph{Математический маятник.}
	\[
		E = \frac{mV^2}{2} + mgh,
	\]
	где $h$~--- высота подъёма груза над положением равновесия. При отклонении на малый угол $\alpha$ (дуга $x = l\alpha$):
	\[
		h = l - l\cos\alpha \approx l\cdot\frac{\alpha^2}{2} = \frac{x^2}{2l}
	\]
	(использовано $\cos\alpha \approx 1 - \dfrac{\alpha^2}{2}$ при малых $\alpha$). Тогда
	\[
		E = \frac{mV^2}{2} + \frac{mg}{2l}x^2,
	\]
	то есть $A = \dfrac{mg}{2l}$, $B = \dfrac{m}{2}$, откуда
	\[
		T = 2\pi\sqrt{\frac{m/2}{mg/2l}} = 2\pi\sqrt{\frac{l}{g}}.
	\]

	\subsection{Затухающие колебания}
	
	\begin{definition}[Затухающие колебания]
		Свободные колебания в системе, где, кроме возвращающей силы, действует сила сопротивления, пропорциональная скорости: $\vec{F}_{\text{сопр}} = -r\vec{V}$, где $r > 0$~--- коэффициент сопротивления, $[r] = \dfrac{\text{кг}}{\text{с}}$. Энергия при этом убывает, и амплитуда со временем уменьшается.
	\end{definition}
	
	\subsubsection{Уравнение затухающих колебаний}
	
	Рассмотрим пружинный маятник (вертикальный или горизонтальный~--- как и раньше, при отсчёте $x$ от положения равновесия сила тяжести из уравнения выпадает). Проекция второго закона Ньютона на ось $Ox$:
	\[
		ma_x = -kx - rV_x \quad \Longleftrightarrow \quad mx'' = -kx - rx'.
	\]
	Введём обозначения
	\[
		\omega_0 = \sqrt{\dfrac{k}{m}}, \qquad \beta = \dfrac{r}{2m}.
	\]
	Здесь $\omega_0$~--- собственная частота системы без затухания, $\beta$~--- коэффициент затухания, $[\beta] = \text{с}^{-1}$.
	
	\subsubsection{Решение уравнения}
	
	\begin{statement}
		Пусть $\beta < \omega_0$. Тогда функция
		\[
			x(t) = A_0 e^{-\beta t} \cos (\omega t + \varphi_0), \qquad \omega = \sqrt{\omega_0^2 - \beta^2},
		\]
		является решением уравнения $mx'' = -kx - rx'$ при любых $A_0$ и $\varphi_0$.
	\end{statement}
	\begin{proof}[Вывод]
		Будем искать решение в виде колебания с медленно меняющейся амплитудой:
		\[
			x = A(t) \cos (\omega t + \varphi_0),
		\]
		где $A(t)$ и $\omega$ нужно найти. Дифференцируем:
		\[
			x' = A' \cos (\omega t + \varphi_0) - A \omega \sin (\omega t + \varphi_0),
		\]
		\[
			x'' = A'' \cos (\omega t + \varphi_0) - 2A' \omega \sin (\omega t + \varphi_0) - A \omega^2 \cos (\omega t + \varphi_0).
		\]
		Подставим в $mx'' + rx' + kx = 0$ и сгруппируем слагаемые при $\sin$ и $\cos$:
		\[
			-\omega \sin (\omega t + \varphi_0) \bigl( 2mA' + rA \bigr) + \cos (\omega t + \varphi_0) \bigl( mA'' - mA\omega^2 + rA' + kA \bigr) = 0.
		\]
		Это равенство выполнено для всех $t$, если обращаются в нуль обе скобки:
		\[
			\begin{cases}
				2mA' + rA = 0, \\
				mA'' - mA\omega^2 + rA' + kA = 0.
			\end{cases}
		\]
		\textit{Первое уравнение} (с разделяющимися переменными, $A \ne 0$):
		\[
			2m \dfrac{dA}{dt} = -rA \quad \Longrightarrow \quad \dfrac{dA}{A} = -\dfrac{r}{2m}\,dt = -\beta\,dt \quad \Longrightarrow \quad \ln |A| = C - \beta t.
		\]
		Значит, $A(t) = A_0 e^{-\beta t}$, где $A_0 = e^C$ (или $-e^C$ для отрицательной амплитуды~--- знак поглощается фазой $\varphi_0$).
		
		\textit{Второе уравнение.} Из $A' = -\beta A$ получаем $A'' = -\beta A' = \beta^2 A$. Подставим в него $A' = -\beta A$ и $A'' = \beta^2 A$:
		\[
			m\beta^2 A - mA\omega^2 - r\beta A + kA = 0.
		\]
		Сократим на $A \ne 0$ и разделим на $m$, учитывая $\dfrac{k}{m} = \omega_0^2$ и $r = 2m\beta$:
		\[
			\beta^2 - \omega^2 - 2\beta^2 + \omega_0^2 = 0 \quad \Longrightarrow \quad \omega^2 = \omega_0^2 - \beta^2.
		\]
		Такое $\omega$ вещественно только при $\beta < \omega_0$, откуда и условие в утверждении. В явном виде:
		\[
			\omega = \sqrt{\dfrac{k}{m} - \dfrac{r^2}{4m^2}}.
		\]
	\end{proof}
	
	\begin{statement}[Итог]
		Затухающие колебания (при $\beta < \omega_0$) описываются законом
		\[
			\boxed{x(t) = A_0 e^{-\frac{r}{2m}t} \cos (\omega t + \varphi_0), \qquad \omega = \sqrt{\omega_0^2 - \beta^2} = \sqrt{\dfrac{k}{m} - \dfrac{r^2}{4m^2}}}
		\]
		Здесь $A_0$ и $\varphi_0$ определяются начальными условиями (как и в случае гармонических колебаний).
	\end{statement}
	
	\subsubsection{Характеристики затухающих колебаний}
	
	\begin{itemize}
		\item Амплитуда убывает по экспоненциальному закону: $A(t) = A_0 e^{-\beta t}$. За время $\tau = \dfrac{1}{\beta} = \dfrac{2m}{r}$ (время релаксации) амплитуда уменьшается в $e$ раз.
		\item Колебания не являются периодическими в строгом смысле (не повторяются точно), но условным периодом называют
		\[
			T = \dfrac{2\pi}{\omega} = \dfrac{2\pi}{\sqrt{\omega_0^2 - \beta^2}}.
		\]
		Он больше периода $T_0 = \dfrac{2\pi}{\omega_0}$ колебаний без затухания: сопротивление замедляет колебания.
		\item Отношение амплитуд, разделённых одним периодом, постоянно:
		\[
			\dfrac{A(t)}{A(t + T)} = e^{\beta T}.
		\]
		Его логарифм $\delta = \beta T$ называется логарифмическим декрементом затухания.
	\end{itemize}
	
	\begin{note}
		При $\beta \geqslant \omega_0$ (сильное сопротивление) выражение $\omega_0^2 - \beta^2$ неположительно, и решение вида $A(t)\cos(\omega t + \varphi_0)$ не существует~--- система колебаний не совершает. При $\beta > \omega_0$ решение имеет вид $x = e^{-\beta t}\bigl( C_1 e^{\gamma t} + C_2 e^{-\gamma t} \bigr)$, $\gamma = \sqrt{\beta^2 - \omega_0^2}$ (апериодическое движение); при $\beta = \omega_0$ (критическое затухание)~--- $x = (C_1 + C_2 t)e^{-\beta t}$. Оба случая нужно проверять подстановкой отдельно.
	\end{note}

	\begin{definition}[Волновой фронт]

	\subsection{Типы волн}

	\begin{itemize}
		\item \textbf{Продольные.} Колебания происходят вдоль направления распространения волны.
		\item \textbf{Поперечные.} Колебания происходят перпендикулярно направлению распространения волны.
	\end{itemize}

	\subsection{Движение частиц}

	\begin{statement}
		Происходит перенос энергии \textbf{без} переноса вещества.
	\end{statement}

	\begin{statement}
		Частица совершает \textbf{колебательное движение}. Оно характеризуется \textbf{периодом, частотой и амплитудой}.
	\end{statement}

	\begin{definition}[Длина волны]
		Расстояние, на которое распространилась волна за один период.

		\[
			\lambda = V \cdot T.
		\]
	\end{definition}

	\begin{statement}[Смещение точки от времени и координаты]
		\[
		S \left( x; t \right) = S_m \cdot \sin \left( \omega_0 \left( t - \frac{x}{v} \right) \right) = S_m \cdot \sin \left( 2 \pi \left( \frac{t}{T} - \frac{x}{\lambda} \right) \right)
		\]
	\end{statement}

	\subsection{Звук}

	\begin{definition}[Звук]
		Механическое колебание, которое происходит с частотой от $16$ Гц до $20000$ Гц.
	\end{definition}

	\begin{definition}[Высота звука]
		Высота звука зависит от частоты колебаний, чем больше частота, тем выше звук.
	\end{definition}

	\begin{statement}[Скорость поперечной волны]
		В натянутой струне (шнуре) с силой натяжения $F$, площадью сечения $S$ и плотностью $\rho$ скорость распространения поперечной волны равна
		\[
			V = \sqrt{\frac{F}{\rho S}}.
		\]
	\end{statement}
	\begin{proof}[Вывод]
		Перейдём в систему отсчёта, движущуюся вместе с волной со скоростью $V$. В ней профиль волны неподвижен, а вещество струны движется вдоль него со скоростью $V$. Рассмотрим малый участок у гребня: его можно считать дугой окружности радиуса $R$ с центральным углом $2\alpha$ (рис.~\ref{fig:transverse-wave}). Тогда участок движется по окружности со скоростью $V$ и центростремительным ускорением $a = \dfrac{V^2}{R}$.

		\begin{figure}[H]
			\centering
			\begin{tikzpicture}[scale=0.9, line join=round, line cap=round, >=Stealth]
				% Волна
				\begin{scope}[xshift=-7.6cm, yshift=0.2cm]

						% окружность кривизны в гребне: R = 1/y'' = 1/(0.6*1.8^2)

		На концы участка действуют силы натяжения $F$, направленные по касательным. Их горизонтальные составляющие компенсируются, а вертикальные складываются и направлены к центру окружности. Второй закон Ньютона в проекции на радиус:
		\[
			2F\sin\alpha = \varDelta m \cdot a = \varDelta m\,\frac{V^2}{R}.
		\]
		Масса участка длиной $\varDelta l = 2\alpha R$:
		\[
			\varDelta m = \rho\,\varDelta l\,S = \rho \cdot 2\alpha R \cdot S.
		\]
		Угол $\alpha$ мал, поэтому $\sin\alpha \approx \alpha$. Подставляем:
		\[
			2F\alpha = \rho \cdot 2\alpha R S \cdot \frac{V^2}{R} \quad \Longrightarrow \quad F = \rho S V^2 \quad \Longrightarrow \quad V = \sqrt{\frac{F}{\rho S}}.
		\]
	\end{proof}

	\begin{statement}[Скорость продольной волны]
		В стержне с модулем Юнга $E$ и плотностью $\rho$ скорость распространения продольной волны (малых деформаций) равна
		\[
			V = \sqrt{\frac{E}{\rho}}.
		\]
	\end{statement}
	\begin{proof}[Вывод]
		Пусть левый конец стержня сечением $S$ начинает двигаться с постоянной скоростью $u$ под действием постоянной силы $F$ (рис.~\ref{fig:longitudinal-wave}). За время $\varDelta t$ возмущение распространяется на расстояние $l = V\varDelta t$. Эта часть стержня сжата и движется со скоростью $u$, остальная часть ещё покоится. Левый конец за это время сместился на $\varDelta l = u\varDelta t$.

			\centering

			\label{fig:longitudinal-wave}
		\end{figure}

		Деформация сжатой части и сила по закону Гука ($\sigma = E|\varepsilon|$):
		\[
			|\varepsilon| = \frac{\varDelta l}{l} = \frac{u\varDelta t}{V\varDelta t} = \frac{u}{V}, \qquad F = \sigma S = E S\,\frac{\varDelta l}{l}.
		\]
		Импульс силы равен изменению импульса сжатой части (масса $\varDelta m = \rho S l$, скорость менялась от $0$ до $u$):
		\[
			F\varDelta t = \varDelta p \quad \Longrightarrow \quad E S\,\frac{\varDelta l}{l}\,\varDelta t = \rho S l u.
		\]
		Подставим $u = \dfrac{\varDelta l}{\varDelta t}$ и сократим на $S$ и $\varDelta l$:
		\[
			E\,\frac{\varDelta t}{l} = \rho\,\frac{l}{\varDelta t} \quad \Longrightarrow \quad E = \rho \left( \frac{l}{\varDelta t} \right)^2 = \rho V^2 \quad \Longrightarrow \quad V = \sqrt{\frac{E}{\rho}}.
		\]
	\end{proof}

	\subsubsection{Распространение звука}

	\begin{definition}[Модуль всестороннего сжатия]
		Отношение изменения давления к относительному изменению объёма, взятое с обратным знаком:
		\[
			K = -\dfrac{\varDelta p}{\varDelta V / V} = -V\,\dfrac{dp}{dV}, \qquad [K] = \text{Па}.
		\]
		При сжатии ($dV < 0$) давление растёт, поэтому $K > 0$.
	\end{definition}

	\begin{statement}[Скорость звука в среде]
		В газе или жидкости с модулем всестороннего сжатия $K$ и плотностью $\rho$ скорость звука равна
		\[
			V_{\text{зв}} = \sqrt{\dfrac{K}{\rho}}.
		\]
	\end{statement}
	\begin{proof}[Вывод]
		Повторим вывод для продольной волны в стержне. Пусть поршень в трубе сечением $S$ начинает двигаться с постоянной скоростью $u$. За время $\varDelta t$ сжатие распространяется на расстояние $l = V_{\text{зв}}\varDelta t$, а поршень смещается на $\varDelta l = u\varDelta t$. Сечение постоянно, поэтому относительное изменение объёма сжатого слоя равно относительному изменению его длины:
		\[
			\dfrac{|\varDelta V|}{V} = \dfrac{\varDelta l}{l} = \dfrac{u}{V_{\text{зв}}}, \qquad \varDelta p = K\,\dfrac{|\varDelta V|}{V} = K\,\dfrac{u}{V_{\text{зв}}}.
		\]
		Избыточное давление за время $\varDelta t$ разгоняет слой массой $\varDelta m = \rho S l$ от $0$ до $u$:
		\[
			\varDelta p\, S\,\varDelta t = \rho S l u = \rho S V_{\text{зв}}\varDelta t\, u \quad \Longrightarrow \quad \varDelta p = \rho V_{\text{зв}} u.
		\]
		Приравняем два выражения для $\varDelta p$ и сократим на $u$:
		\[
			K\,\dfrac{1}{V_{\text{зв}}} = \rho V_{\text{зв}} \quad \Longrightarrow \quad V_{\text{зв}} = \sqrt{\dfrac{K}{\rho}}.
		\]
	\end{proof}

	\begin{statement}[Скорость звука в газе]
		Если сжатия и разрежения в волне считать изотермическими, то (формула Ньютона)
		\[
			V_{\text{зв}} = \sqrt{\dfrac{RT}{M}}.
		\]
		Если считать их адиабатными, то (формула Лапласа)
		\[
			V_{\text{зв}} = \sqrt{\dfrac{\gamma RT}{M}},
		\]
		где $\gamma = \dfrac{i + 2}{i}$~--- коэффициент Пуассона (показатель адиабаты). Для воздуха ($i = 5$, $\gamma = \dfrac{7}{5}$):
		\[
			V_{\text{зв}} = \sqrt{\dfrac{7RT}{5M}}.
		\]
		Опыт согласуется с формулой Лапласа.
	\end{statement}
	\begin{proof}[Вывод]
		Найдём $K$ в каждом случае. При изотермическом процессе $pV = const$, откуда $p\,dV + V\,dp = 0$ и $K_T = -V\dfrac{dp}{dV} = p$. При адиабатном процессе $pV^\gamma = const$, откуда $V^\gamma dp + \gamma p V^{\gamma - 1} dV = 0$ и $K_{\text{ад}} = -V\dfrac{dp}{dV} = \gamma p$.

		Плотность газа выражается через уравнение Менделеева--Клапейрона: $\rho = \dfrac{m}{V} = \dfrac{\nu M}{V} = \dfrac{pM}{RT}$. Подставим в $V_{\text{зв}} = \sqrt{\dfrac{K}{\rho}}$:
		\[
			V_{\text{зв}} = \sqrt{\dfrac{p \cdot RT}{pM}} = \sqrt{\dfrac{RT}{M}}, \qquad V_{\text{зв}} = \sqrt{\dfrac{\gamma p \cdot RT}{pM}} = \sqrt{\dfrac{\gamma RT}{M}}.
		\]
	\end{proof}

	\begin{note}
		Процесс в звуковой волне адиабатный: сжатия и разрежения сменяют друг друга за время, много меньшее времени выравнивания температуры теплопроводностью, поэтому $Q \approx 0$. Формула Ньютона для этого неверна.
	\end{note}

	\begin{example}
		Воздух: $M = 29 \cdot 10^{-3}$ кг/моль, $T = 273$ К. Формула Ньютона даёт $V_{\text{зв}} \approx 280$ м/с, формула Лапласа~--- $V_{\text{зв}} \approx 331$ м/с, что совпадает с измеренным значением.
	\end{example}

	\begin{note}
		Скорость звука в газе растёт как $\sqrt{T}$ и не зависит от давления при фиксированной температуре. Она одного порядка с тепловой скоростью молекул: $V_{\text{зв}} = \sqrt{\dfrac{\gamma}{3}}\,\sqrt{\overline{\mathcal{V}^2}}$. В жидкостях $K$ значительно больше: для воды $K \approx 2.2 \cdot 10^{9}$ Па, $\rho = 10^3$ кг/м$^3$, откуда $V_{\text{зв}} \approx 1.5$ км/с.
	\end{note}

	\subsection{Интерференция волн}

	\begin{statement}[Принцип суперпозиции]
		При малых деформациях среды смещение каждой её точки равно сумме смещений, которые вызывала бы каждая из волн по отдельности. Волны проходят друг через друга, не изменяясь.
	\end{statement}

	\begin{definition}[Когерентные волны]
		Волны одинаковой частоты с постоянной во времени разностью фаз.
	\end{definition}

	\begin{definition}[Интерференция волн]
		Сложение когерентных волн, при котором в одних точках пространства колебания усиливаются, а в других~--- ослабляются, причём эта картина не меняется со временем.
	\end{definition}

	\begin{definition}[Разность хода]
		$\varDelta d = d_2 - d_1$, где $d_1$ и $d_2$~--- расстояния от точки наблюдения до источников.
	\end{definition}

	\begin{statement}[Условия максимума и минимума]
		Пусть два когерентных источника колеблются в одной фазе. Колебания в точке максимально усиливаются, если разность хода равна целому числу длин волн, и максимально ослабляются, если она равна нечётному числу полуволн:
		\[
			\left| \varDelta d \right| = k\lambda \quad (\text{максимум}), \qquad \left| \varDelta d \right| = (2k + 1)\,\dfrac{\lambda}{2} \quad (\text{минимум}), \qquad k = 0, 1, 2, \dots
		\]
	\end{statement}
	\begin{proof}[Вывод]
		Для простоты амплитуды обеих волн в точке считаем равными $S_m$. По принципу суперпозиции, используя формулу смещения точки от времени и координаты,
		\[
			S = S_m \sin \left( 2\pi \left( \dfrac{t}{T} - \dfrac{d_1}{\lambda} \right) \right) + S_m \sin \left( 2\pi \left( \dfrac{t}{T} - \dfrac{d_2}{\lambda} \right) \right) = 2 S_m \cos \left( \dfrac{\pi \varDelta d}{\lambda} \right) \sin \left( 2\pi \left( \dfrac{t}{T} - \dfrac{d_1 + d_2}{2\lambda} \right) \right),
		\]
		где использована формула $\sin a + \sin b = 2 \sin \dfrac{a + b}{2} \cos \dfrac{a - b}{2}$. Амплитуда результирующего колебания
		\[
			A = 2 S_m \left| \cos \left( \dfrac{\pi \varDelta d}{\lambda} \right) \right|.
		\]
		Она максимальна ($A = 2S_m$) при $\dfrac{\pi \left| \varDelta d \right|}{\lambda} = \pi k$, то есть $\left| \varDelta d \right| = k\lambda$, и равна нулю при $\dfrac{\pi \left| \varDelta d \right|}{\lambda} = \dfrac{\pi}{2} + \pi k$, то есть $\left| \varDelta d \right| = (2k + 1)\dfrac{\lambda}{2}$.
	\end{proof}

	\begin{note}
		Если источники колеблются в противофазе, условия максимума и минимума меняются местами. Стоячая волна~--- частный случай интерференции двух встречных волн одинаковой амплитуды (например, падающей и отражённой).
	\end{note}

	\begin{example}
		Два одинаковых динамика, колеблющихся в одной фазе, излучают звук частотой $\nu = 680$ Гц ($V_{\text{зв}} = 340$ м/с). Длина волны $\lambda = \dfrac{V_{\text{зв}}}{\nu} = 0.5$ м. В точке на расстояниях $4$ м и $4.25$ м от динамиков $\left| \varDelta d \right| = 0.25$ м $= \dfrac{\lambda}{2}$~--- это минимум, звук практически не слышен.
	\end{example}

	\subsection{Принцип Гюйгенса}

	\begin{statement}[Принцип Гюйгенса]
		Каждая точка, до которой дошёл волновой фронт, становится источником вторичных сферических волн. Фронт в момент $t + \varDelta t$ является огибающей (общей касательной поверхностью) вторичных волн, радиус которых равен $V\varDelta t$, где $V$~--- скорость волны.
	\end{statement}

	\begin{example}[Плоская волна]
		Вторичные волны от точек плоского фронта~--- полусферы радиуса $V\varDelta t$ (рис.~\ref{fig:huygens}). Их огибающая~--- плоскость, параллельная прежнему фронту и смещённая на $V\varDelta t$, то есть плоская волна остаётся плоской.

			\centering
			\begin{tikzpicture}[scale=1.2, line join=round, line cap=round, >=Stealth]
				\def\r{1.2}
				\foreach \y in {-1.6, -0.8, 0, 0.8, 1.6} {
					\fill (0, \y) circle (1.2pt);
					\draw[gray] (0, \y) ++(0, -\r) arc (-90:90:\r);
				}
				\draw[thick, blue!70!black] (0, -1.9) -- (0, 1.9);
				\draw[thick, red!70!black] (\r, -1.6) -- (\r, 1.6);
				\node[left] at (0, 1.5) {\small фронт в момент $t$};
				\node[right] at (\r, 1.0) {\small фронт в момент $t + \varDelta t$};
				\draw[->, thick] (-1.2, 0) -- (-0.3, 0) node[midway, above] {$\vec V$};
				\draw[<->] (0, -3.1) -- (\r, -3.1) node[midway, below] {$V\varDelta t$};
			\end{tikzpicture}
			\caption{Плоская волна: вторичные волны и огибающая}
			\label{fig:huygens}
		\end{figure}
	\end{example}

	\begin{note}
		Принцип Гюйгенса определяет только положение фронта. С его помощью выводятся законы отражения и преломления (см. ниже) и объясняется дифракция~--- отклонение волны от прямолинейного распространения у краёв препятствий и отверстий. Он не даёт амплитуду волны и не объясняет, почему не возникает волна, идущая назад: это решает уточнение принципа, которое пока не рассматривается.
	\end{note}

	\subsection{Отражение волн}

	\begin{definition}[Углы падения и отражения]
		Луч~--- линия, перпендикулярная волновым поверхностям и показывающая направление распространения волны. Угол падения $\alpha$ (отражения $\beta$)~--- угол между падающим (отражённым) лучом и перпендикуляром к границе, проведённым в точку падения.
	\end{definition}

	\begin{law}[Закон отражения]
		Падающий луч, отражённый луч и перпендикуляр к границе, проведённый в точку падения, лежат в одной плоскости, причём угол падения равен углу отражения:
		\[
			\alpha = \beta.
		\]
	\end{law}
	\begin{proof}[Вывод]
		Пусть плоская волна падает на плоскую границу, а рисунок выполнен в плоскости падения (отражённая волна лежит в ней по симметрии задачи). В момент, когда падающий фронт $AB$ касается границы в точке $A$, его другой край $B$ ещё не дошёл до границы (рис.~\ref{fig:reflection}). Он дойдёт до точки $C$ через время $\varDelta t = \dfrac{BC}{V}$, где $V$~--- скорость волны. За это время вторичная волна из точки $A$ пройдёт расстояние $AD = V\varDelta t = BC$, а отражённый фронт $CD$ будет касаться окружности радиуса $AD$ с центром в $A$.

	\end{proof}

	\begin{example}[Эхо]
		Ухо различает прямой и отражённый звук, если они приходят с запаздыванием не менее $0.1$ с. За это время звук проходит туда и обратно примерно $340 \cdot 0.1 = 34$ м, поэтому эхо слышно, если преграда находится дальше $17$ м.
	\end{example}

	\subsection{Преломление волн}

	\begin{definition}[Преломление]
		Изменение направления распространения волны при переходе из одной среды в другую, в которой скорость волны иная. Угол падения обозначим $\alpha_1$, угол преломления~--- $\alpha_2$ (угол между преломлённым лучом и перпендикуляром к границе), скорости волны в средах~--- $V_1$ и $V_2$.
	\end{definition}

	\begin{law}[Закон преломления]
		Падающий луч, преломлённый луч и перпендикуляр к границе, проведённый в точку падения, лежат в одной плоскости, причём
		\[
			\dfrac{\sin \alpha_1}{\sin \alpha_2} = \dfrac{V_1}{V_2} = n_{21},
		\]
		где $n_{21}$~--- относительный показатель преломления второй среды относительно первой.
	\end{law}
	\begin{proof}[Вывод]
		Пусть плоская волна падает на плоскую границу двух сред. В момент, когда падающий фронт $AB$ касается границы в точке $A$, его край $B$ ещё не дошёл до границы (рис.~\ref{fig:refraction}). Он дойдёт до точки $C$ через время $\varDelta t = \dfrac{BC}{V_1}$. По принципу Гюйгенса точка $A$ становится источником вторичной волны во второй среде, которая за это время пройдёт расстояние $AD = V_2\varDelta t$. Преломлённый фронт $CD$~--- касательная к окружности радиуса $AD$ с центром в $A$.

		\begin{figure}[H]
			\centering

				\pgfmathsetmacro{\s}{2.2}

				\pgfmathsetmacro{\ac}{\s/cos(\al)}

		\end{figure}

		Треугольники $ABC$ и $ADC$ прямоугольные с общей гипотенузой $AC$. Угол $BAC$ между фронтом и границей равен углу падения $\alpha_1$, а угол $ACD$~--- углу преломления $\alpha_2$ (их стороны перпендикулярны лучам и нормали). Поэтому
		\[
			\sin \alpha_1 = \dfrac{BC}{AC} = \dfrac{V_1\varDelta t}{AC}, \qquad \sin \alpha_2 = \dfrac{AD}{AC} = \dfrac{V_2\varDelta t}{AC} \quad \Longrightarrow \quad \dfrac{\sin \alpha_1}{\sin \alpha_2} = \dfrac{V_1}{V_2}.
		\]
	\end{proof}

	\begin{note}
		При переходе в среду с меньшей скоростью ($V_2 < V_1$) луч приближается к нормали ($\alpha_2 < \alpha_1$), с большей~--- удаляется от неё. Частота волны при переходе не меняется (она задана источником), поэтому из $\lambda = VT$ следует $\dfrac{\lambda_1}{\lambda_2} = \dfrac{V_1}{V_2}$: меняется длина волны.
	\end{note}

	\begin{statement}[Полное отражение]
		Пусть $V_2 > V_1$. При угле падения $\alpha_1 \geqslant \alpha_{\text{пр}}$, где
		\[
			\sin \alpha_{\text{пр}} = \dfrac{V_1}{V_2},
		\]
		преломлённой волны нет, и вся волна отражается от границы.
	\end{statement}
	\begin{proof}[Вывод]
		Из закона преломления $\sin \alpha_2 = \dfrac{V_2}{V_1}\sin \alpha_1$. При $\sin \alpha_1 > \dfrac{V_1}{V_2}$ получаем $\sin \alpha_2 > 1$, что невозможно.
	\end{proof}

	\begin{example}[Звук из воздуха в воду]
		Для воздуха $V_1 \approx 340$ м/с, для воды $V_2 \approx 1500$ м/с, поэтому $\sin \alpha_{\text{пр}} = \dfrac{340}{1500} \approx 0.23$, то есть $\alpha_{\text{пр}} \approx 13$\textdegree. Звук, падающий на воду под большим углом к нормали, почти полностью отражается.
	\end{example}

	\subsection{Поляризация волн}

	\begin{definition}[Поляризация]
		Свойство поперечной волны, состоящее в том, что колебания частиц происходят в плоскости, перпендикулярной направлению распространения, и упорядочены в ней определённым образом.
	\end{definition}

	\begin{note}
		Продольные волны (в том числе звук в газах и жидкостях) не поляризуются: колебания в них идут вдоль луча, и в перпендикулярной плоскости нет выделенного направления.
	\end{note}

	\begin{definition}[Плоская (линейная) поляризация]
		Колебания каждой частицы происходят вдоль одной прямой, перпендикулярной лучу. Плоскость, содержащую эту прямую и луч, называют плоскостью колебаний.
	\end{definition}

	\begin{definition}[Круговая поляризация]
		Конец вектора смещения частицы равномерно движется по окружности в плоскости, перпендикулярной лучу.
	\end{definition}

	\subsection{Энергия}

	\begin{definition}[Интенсивность волны, $I$, $\dfrac{\text{Вт}}{\text{м}^2}$]
		Энергия, которую волна переносит за единицу времени через единичную площадку, перпендикулярную направлению распространения:
		\[
			I = \dfrac{\varDelta W}{\varDelta S\,\varDelta t}.
		\]
	\end{definition}

	\begin{statement}[Интенсивность гармонической волны]
		Пусть бегущая гармоническая волна с амплитудой смещения $S_m$ и циклической частотой $\omega$ распространяется со скоростью $V$ в среде плотностью $\rho$. Тогда её интенсивность (среднее значение за период) равна
		\[
			I = \dfrac{\rho V \omega^2 S_m^2}{2}.
		\]
	\end{statement}
	\begin{proof}[Вывод]
		За время $\varDelta t$ волна проходит расстояние $V\varDelta t$, поэтому через площадку $\varDelta S$ за это время переносится энергия, заключённая в цилиндре с основанием $\varDelta S$ и высотой $V\varDelta t$. Масса вещества в нём $\varDelta m = \rho\,\varDelta S\,V\varDelta t$. Каждая частица колеблется гармонически с амплитудой $S_m$, то есть её максимальная скорость равна $\omega S_m$, а энергия колебаний~--- $\dfrac{m (\omega S_m)^2}{2}$. Следовательно,
		\[
			\varDelta W = \dfrac{\varDelta m\,\omega^2 S_m^2}{2} = \dfrac{\rho\,\varDelta S\,V\varDelta t\,\omega^2 S_m^2}{2}, \qquad I = \dfrac{\varDelta W}{\varDelta S\,\varDelta t} = \dfrac{\rho V \omega^2 S_m^2}{2}.
		\]
	\end{proof}

	\begin{note}
		Из вывода видно, что средняя плотность энергии волны $w = \dfrac{\rho \omega^2 S_m^2}{2}$ и $I = wV$: энергия переносится со скоростью волны. В бегущей волне кинетическая и потенциальная энергии элемента среды достигают максимума одновременно (при прохождении положения равновесия) и в среднем равны, по $\dfrac{w}{2}$ каждая. Этим бегущая волна отличается от маятника, у которого энергии переходят друг в друга.
	\end{note}

	\begin{corollary}[Интенсивность через амплитуду давления]
		Для звуковой волны с амплитудой избыточного давления $\varDelta p_m$
		\[
			I = \dfrac{\varDelta p_m^2}{2\rho V_{\text{зв}}}.
		\]
	\end{corollary}
	\begin{proof}
		Соотношение $\varDelta p = \rho V_{\text{зв}} u$ из вывода скорости звука связывает избыточное давление и скорость $u$ частиц среды в любой точке бегущей волны. Максимальная скорость частиц равна $\omega S_m$, поэтому $\varDelta p_m = \rho V_{\text{зв}} \omega S_m$ и
		\[
			I = \dfrac{\rho V_{\text{зв}} \omega^2 S_m^2}{2} = \dfrac{(\rho V_{\text{зв}} \omega S_m)^2}{2\rho V_{\text{зв}}} = \dfrac{\varDelta p_m^2}{2\rho V_{\text{зв}}}.
		\]
	\end{proof}

	\begin{statement}[Закон обратных квадратов]
		Пусть точечный источник мощностью $P$ излучает сферическую волну равномерно по всем направлениям в однородной среде без поглощения. Тогда на расстоянии $r$ от источника
		\[
			I = \dfrac{P}{4\pi r^2}, \qquad \text{то есть} \quad I \sim \dfrac{1}{r^2},
		\]
		а амплитуда смещения $S_m \sim \dfrac{1}{r}$.
	\end{statement}
	\begin{proof}
		Поглощения нет, поэтому через сферу радиуса $r$ за единицу времени проходит вся излучаемая энергия: $P = I \cdot 4\pi r^2$. При фиксированных $\rho$, $V$ и $\omega$ интенсивность пропорциональна $S_m^2$, откуда $S_m \sim \dfrac{1}{r}$.
	\end{proof}

	\begin{example}
		Всенаправленный источник звука мощностью $P = 0.1$ Вт создаёт на расстоянии $r = 10$ м интенсивность $I = \dfrac{0.1}{4\pi \cdot 10^2} \approx 8 \cdot 10^{-5}$ $\dfrac{\text{Вт}}{\text{м}^2}$. При удвоении расстояния интенсивность уменьшается в $4$ раза, а амплитуда смещения~--- в $2$ раза.
	\end{example}

		\subsection{Эффект Доплера}

	\begin{definition}[Эффект Доплера]
		Изменение частоты колебаний, воспринимаемой приёмником, при движении источника и (или) приёмника относительно среды, в которой распространяется волна.
	\end{definition}

	\begin{statement}[Формула Доплера]
		Пусть источник излучает волну частоты $\nu_0$, скорость волны в среде равна $V$, а источник и приёмник движутся вдоль соединяющей их прямой со скоростями $V_{\text{и}}$ и $V_{\text{п}}$ относительно среды. Тогда приёмник регистрирует частоту
		\[
			\nu = \nu_0\,\dfrac{V \pm V_{\text{п}}}{V \pm V_{\text{и}}}.
		\]
		В числителе берётся ``$+$'', если приёмник приближается к источнику, и ``$-$'', если удаляется. В знаменателе берётся ``$+$'', если источник удаляется от приёмника, и ``$-$'', если приближается. Формула верна при $V_{\text{и}} < V$ для приближающегося источника и $V_{\text{п}} < V$ для удаляющегося приёмника.
	\end{statement}
	\begin{proof}[Вывод]
		\textbf{Источник движется, приёмник неподвижен.} Пусть источник удаляется от приёмника со скоростью $V_{\text{и}}$ (рис.~\ref{fig:doppler}). Приёмник неподвижен, поэтому воспринимаемая частота и длина волны в среде связаны соотношением $\lambda = \dfrac{V}{\nu}$. Неподвижный источник излучал бы волну длиной $\lambda_0 = \dfrac{V}{\nu_0} = VT_0$, где $T_0 = \dfrac{1}{\nu_0}$.

		\begin{figure}[H]
			\centering
			\begin{tikzpicture}[scale=0.9, >=Stealth]
				\draw[thick] (-2, 0) -- (10.5, 0);
				\draw[domain=0:8, samples=120, smooth, thick] plot (\x, {0.7*cos(deg(2*pi*\x/8))});
				\fill (0, 0) circle (2pt) node[above left] {$A$};
				\fill (2, 0) circle (2pt) node[above right] {$B$};
				\fill (0, 0.7) circle (2pt);
				\fill (8, 0.7) circle (2pt) node[above right] {$C$};
				\fill (10, 0) circle (2pt) node[above] {П};
				\draw[->, thick, blue!70!black] (0, 1.3) -- (-1.6, 1.3) node[midway, above] {$\vec V_{\text{и}}$};
				\draw[dotted] (0, 0.7) -- (0, -2.6);
				\draw[dotted] (2, 0) -- (2, -1.2);
				\draw[dotted] (8, 0.7) -- (8, -2.6);
				\draw[<->] (0, -1) -- (2, -1) node[midway, below] {$V_{\text{и}}T_0$};
				\draw[<->] (2, -1) -- (8, -1) node[midway, below] {$\lambda_0$};
				\draw[<->] (0, -2.4) -- (8, -2.4) node[midway, below] {$\lambda$};
			\end{tikzpicture}
			\caption{Источник в точке $A$; период назад он был в точке $B$ и излучил гребень, дошедший до $C$}
			\label{fig:doppler}
		\end{figure}

		Скорость волны задаётся средой и не зависит от движения источника, поэтому за время $T_0$ гребень, испущенный в точке $B$, проходит расстояние $\lambda_0$ и оказывается в точке $C$. Источник за это время смещается на $V_{\text{и}}T_0$ от приёмника в точку $A$ и испускает следующий гребень. Расстояние между гребнями, то есть длина волны в среде,
		\[
			\lambda = \lambda_0 + V_{\text{и}}T_0.
		\]
		Подставим $\lambda = \dfrac{V}{\nu}$, $\lambda_0 = \dfrac{V}{\nu_0}$ и $T_0 = \dfrac{1}{\nu_0}$:
		\[
			\dfrac{V}{\nu} = \dfrac{V}{\nu_0} + \dfrac{V_{\text{и}}}{\nu_0} = \dfrac{V + V_{\text{и}}}{\nu_0} \quad \Longrightarrow \quad \nu = \dfrac{V\nu_0}{V + V_{\text{и}}} = \dfrac{\nu_0}{1 + \dfrac{V_{\text{и}}}{V}}.
		\]
		Если источник приближается к приёмнику, расстояние между гребнями не увеличивается, а уменьшается, и перед $V_{\text{и}}$ стоит ``$-$''.

		\textbf{Приёмник движется, источник неподвижен.} Теперь длина волны в среде равна $\lambda_0 = \dfrac{V}{\nu_0}$ и от приёмника не зависит. Пусть приёмник приближается к источнику со скоростью $V_{\text{п}}$. Тогда скорость волны относительно приёмника равна $V + V_{\text{п}}$, и он встречает гребни с частотой $\nu = \dfrac{V + V_{\text{п}}}{\lambda_0}$:
		\[
			\lambda_0 = \dfrac{V}{\nu_0} = \dfrac{V + V_{\text{п}}}{\nu} \quad \Longrightarrow \quad \nu = \nu_0\,\dfrac{V + V_{\text{п}}}{V}.
		\]
		Если приёмник удаляется, скорость волны относительно него равна $V - V_{\text{п}}$, и перед $V_{\text{п}}$ стоит ``$-$''.

		\textbf{Общий случай.} Длину волны в среде определяет движение источника: $\lambda = \dfrac{V \pm V_{\text{и}}}{\nu_0}$. Частоту, с которой приёмник встречает гребни, определяет его собственное движение: $\nu = \dfrac{V \pm V_{\text{п}}}{\lambda}$. Подставляя первое равенство во второе, получаем
		\[
			\nu = \nu_0\,\dfrac{V \pm V_{\text{п}}}{V \pm V_{\text{и}}}.
		\]
	\end{proof}

	\begin{note}
		Частные случаи:
		\begin{itemize}
			\item Приёмник неподвижен ($V_{\text{п}} = 0$): $\nu = \nu_0\,\dfrac{V}{V - V_{\text{и}}} > \nu_0$ при приближении источника и $\nu = \nu_0\,\dfrac{V}{V + V_{\text{и}}} < \nu_0$ при удалении.
			\item Источник неподвижен ($V_{\text{и}} = 0$): $\nu = \nu_0\,\dfrac{V + V_{\text{п}}}{V} > \nu_0$ при приближении приёмника и $\nu = \nu_0\,\dfrac{V - V_{\text{п}}}{V} < \nu_0$ при удалении.
		\end{itemize}
		Если скорости направлены не вдоль прямой, соединяющей источник и приёмник, вместо них берутся проекции скоростей на эту прямую (в момент испускания и в момент приёма соответственно).
	\end{note}

	\begin{note}
		Пусть источник и приёмник сближаются со скоростью $v$. Если движется источник, то $\nu = \nu_0\,\dfrac{V}{V - v}$, если приёмник~--- $\nu = \nu_0\,\dfrac{V + v}{V}$. Эти сдвиги различны, потому что для звука среда выделена и важно, что именно движется. При $v \ll V$ различие мало: оба множителя приближённо равны $1 + \dfrac{v}{V}$. Для света среды нет, и сдвиг зависит только от относительной скорости (по другой формуле).
	\end{note}

	\begin{example}[Сирена]
		Источник с частотой $\nu_0 = 1000$ Гц движется со скоростью $V_{\text{и}} = 20$ м/с мимо неподвижного наблюдателя, $V = 340$ м/с. При приближении $\nu = 1000 \cdot \dfrac{340}{340 - 20} \approx 1062$ Гц, при удалении $\nu = 1000 \cdot \dfrac{340}{340 + 20} \approx 944$ Гц. В момент прохождения мимо наблюдателя тон резко понижается.
	\end{example}

	\begin{example}[Отражение от движущейся преграды]
		Неподвижный источник частоты $\nu_0$ посылает звук на стену, которая приближается со скоростью $u$; приёмник находится рядом с источником. Стена как приёмник приближается к источнику ($V_{\text{п}} = u$), поэтому воспринимает частоту $\nu_1 = \nu_0\,\dfrac{V + u}{V}$. Отражённую волну излучает стена как источник частоты $\nu_1$, приближающийся к неподвижному приёмнику ($V_{\text{и}} = u$):
		\[
			\nu_2 = \nu_1\,\dfrac{V}{V - u} = \nu_0\,\dfrac{V + u}{V - u}.
		\]
		Разность частот отражённого и прямого звука $\nu_2 - \nu_0 = \nu_0\,\dfrac{2u}{V - u} \approx \dfrac{2u}{V}\,\nu_0$ при $u \ll V$ позволяет измерять скорость стены.
	\end{example}
